const B = EXT_BASE_012B;
const WEAPONS = [B + 19, B + 21, B + 22, B + 20];
interface Gate { x: number; y: number; t: number; life: number; tgt: Enemy | null; fired: boolean }

/** 繪製天門光輪 */
function ring(c: CanvasRenderingContext2D, x: number, y: number, r: number, t: number, a = 1, col = "#ffd24a") {
  c.save();
  c.translate(x, y);
  c.globalCompositeOperation = "lighter";
  c.globalAlpha = a;
  const gr = c.createRadialGradient(0, 0, r * 0.2, 0, 0, r * 1.3);
  gr.addColorStop(0, "rgba(255,250,220,0.9)");
  gr.addColorStop(0.55, "rgba(255,210,74,0.35)");
  gr.addColorStop(1, "rgba(255,210,74,0)");
  c.fillStyle = gr;
  c.beginPath(); c.arc(0, 0, r * 1.3, 0, 6.3); c.fill();
  c.strokeStyle = col;
  c.lineWidth = Math.max(2, r * 0.1);
  c.beginPath(); c.arc(0, 0, r, 0, 6.3); c.stroke();
  c.rotate(t * 1.2);
  c.lineWidth = Math.max(1.2, r * 0.05);
  c.strokeStyle = "#fff4c8";
  c.setLineDash([r * 0.3, r * 0.2]);
  c.beginPath(); c.arc(0, 0, r * 0.74, 0, 6.3); c.stroke();
  c.setLineDash([]);
  c.rotate(-t * 2.4);
  for (let i = 0; i < 8; i++) {
    c.rotate(Math.PI / 4);
    c.beginPath(); c.moveTo(r * 0.5, 0); c.lineTo(r * 0.64, 0); c.stroke();
  }
  c.restore();
}

/** 繪製武具（朝 +x 方向，尖端在前） */
function weapon(c: CanvasRenderingContext2D, kind: number, x: number, y: number, ang: number, s: number, a = 1) {
  c.save();
  c.translate(x, y);
  c.rotate(ang);
  c.scale(s, s);
  c.globalAlpha = a;
  c.shadowColor = "#ffd24a";
  c.shadowBlur = 16;
  const gold = "#ffd24a";
  const light = "#fff4c8";
  switch (kind % 5) {
    case 0: // 劍
      c.fillStyle = light; c.beginPath(); c.moveTo(70, 0); c.lineTo(10, -7); c.lineTo(-24, -5); c.lineTo(-24, 5); c.lineTo(10, 7); c.closePath(); c.fill();
      c.fillStyle = gold; c.fillRect(-30, -16, 8, 32); c.fillRect(-56, -4, 28, 8); c.beginPath(); c.arc(-58, 0, 6, 0, 6.3); c.fill();
      break;
    case 1: // 長槍
      c.fillStyle = "#c8902a"; c.fillRect(-70, -2.5, 100, 5);
      c.fillStyle = light; c.beginPath(); c.moveTo(80, 0); c.lineTo(30, -10); c.lineTo(24, 0); c.lineTo(30, 10); c.closePath(); c.fill();
      c.fillStyle = gold; c.fillRect(20, -6, 6, 12);
      break;
    case 2: // 斧
      c.fillStyle = "#c8902a"; c.fillRect(-60, -3, 90, 6);
      c.fillStyle = light; c.beginPath(); c.moveTo(34, 0); c.quadraticCurveTo(52, -34, 18, -30); c.quadraticCurveTo(24, -10, 8, -3); c.lineTo(8, 3); c.quadraticCurveTo(24, 10, 18, 30); c.quadraticCurveTo(52, 34, 34, 0); c.fill();
      break;
    case 3: // 戟
      c.fillStyle = "#c8902a"; c.fillRect(-66, -2.5, 96, 5);
      c.fillStyle = light; c.beginPath(); c.moveTo(78, 0); c.lineTo(34, -6); c.lineTo(34, 6); c.closePath(); c.fill();
      c.fillStyle = gold; c.beginPath(); c.moveTo(34, -4); c.quadraticCurveTo(38, -26, 14, -24); c.lineTo(20, -12); c.lineTo(30, -4); c.fill(); c.beginPath(); c.moveTo(34, 4); c.quadraticCurveTo(38, 26, 14, 24); c.lineTo(20, 12); c.lineTo(30, 4); c.fill();
      break;
    default: // 偃月刀
      c.fillStyle = "#c8902a"; c.fillRect(-64, -2.5, 80, 5);
      c.fillStyle = light; c.beginPath(); c.moveTo(70, -2); c.quadraticCurveTo(48, -34, 8, -22); c.lineTo(14, -6); c.quadraticCurveTo(40, -8, 70, -2); c.fill();
      c.fillStyle = gold; c.fillRect(6, -6, 8, 12);
  }
  c.restore();
}

function openGate(g: Game, e: Enemy) {
  const k = g.p.k;
  (k.gates as Gate[]).push({ x: clamp(e.x, 40, W - 40), y: clamp(e.y - 230, 60, H - 300), t: 0, life: 0.6, tgt: e, fired: false });
}
function gateShot(g: Game, gt: Gate) {
  const k = g.p.k;
  const lv = g.lvl("weapon");
  const dm = g.dmgMult();
  const e = gt.tgt && !gt.tgt.dead ? gt.tgt : g.nearest(gt.x, gt.y, 99999, (q) => q.y > gt.y);
  const ang = e ? Math.atan2(e.y - gt.y, e.x - gt.x) : Math.PI / 2;
  k.shot++;
  const god = lv >= 7 && k.shot % 4 === 0;
  g.spawnPB(gt.x, gt.y, ang, 980, god ? B + 20 : WEAPONS[k.shot % 4], 36 * dm * (god ? 2.2 : 1), { pierce: 3 + (lv >= 5 ? 2 : 0), explode: lv >= 5 ? 44 : 0, scale: god ? 1.5 : 1, life: 1.4, r: god ? 16 : 11 });
  g.fx.ring(gt.x, gt.y, 10, 54, 0.3, "#fff4c8", 3);
  g.fx.burst(gt.x, gt.y, 8, "#ffd24a", 80, 260, 0.35);
  g.sfx("shoot1", 0.05);
}

export const gateKit012: ShipKit = {
  init(g) {
    g.p.k = { n: 0, shot: 0, gates: [] as Gate[], vault: 0, vaultT: 0, vf: 0, cor: null as null | { t: number; dur: number; ax: number; ay: number; bs: { x: number; y: number }[]; re: number } };
  },
  update(g, dt) {
    const p = g.p;
    const k = p.k;
    const dm = g.dmgMult();
    const gs: Gate[] = k.gates;
    for (let i = gs.length - 1; i >= 0; i--) {
      const q = gs[i];
      q.t += dt;
      if (!q.fired && q.t >= 0.32) { q.fired = true; gateShot(g, q); }
      if (q.t > q.life) gs.splice(i, 1);
    }
    // 寶庫：兩側天門連射
    if (k.vaultT > 0) {
      k.vaultT -= dt;
      k.vf -= dt;
      if (k.vf <= 0) {
        k.vf = 0.25;
        const list = enemyList(g);
        for (const sx of [70, W - 70]) {
          const gy = Math.max(120, p.y - 200);
          const tg = list.length ? list[Math.floor(Math.random() * list.length)] : null;
          const a = tg ? Math.atan2(tg.y - gy, tg.x - sx) : Math.PI / 2;
          g.spawnPB(sx, gy, a, 920, B + 24, 14 * dm, { pierce: 2, life: 1.4, src: 2 });
          g.fx.ring(sx, gy, 6, 34, 0.25, "#ffd24a", 2);
          if (g.lvl("passive") >= 7 && Math.random() < 0.5) g.dropPickup(sx, gy, PK.COIN, 2);
        }
        g.sfx("shoot1", 0.08);
      }
    }
    // 回廊：入口收集我方子彈與敵彈，從出口轉射
    const cor = k.cor;
    if (cor) {
      cor.t += dt;
      const lv = g.lvl("active");
      cor.ax += (p.x - cor.ax) * Math.min(1, dt * 8);
      cor.ay += (p.y - 76 - cor.ay) * Math.min(1, dt * 8);
      cor.re -= dt;
      if (cor.re <= 0) {
        cor.re = 1.0;
        const t1 = strongest(g);
        const list = enemyList(g).sort(() => Math.random() - 0.5);
        const ts = [t1, list[0]].filter(Boolean) as Enemy[];
        cor.bs = (lv >= 7 ? ts : ts.slice(0, 1)).map((e) => ({ x: clamp(e.x, 50, W - 50), y: clamp(e.y - 160, 70, H - 300) }));
        if (!cor.bs.length) cor.bs = [{ x: W / 2, y: 200 }];
      }
      const mult = 2 + (lv >= 3 ? 0.6 : 0);
      let bi = 0;
      for (const b of g.pb.items) {
        if (b.src === 2 || b.kind >= 20 || b.life <= 0) continue;
        if (Math.hypot(b.x - cor.ax, b.y - cor.ay) < 46 && b.vy < 0) {
          const bp = cor.bs[bi++ % cor.bs.length];
          const tg = g.nearest(bp.x, bp.y, 99999, (e) => e.y > bp.y - 30);
          const a = tg ? Math.atan2(tg.y - bp.y, tg.x - bp.x) : Math.PI / 2;
          g.spawnPB(bp.x, bp.y, a, 940, b.style, b.dmg * mult, { pierce: b.pierce + 1, src: 2, life: 1.3, scale: b.scale, r: b.r });
          b.life = 0;
          g.fx.glowP(cor.ax, cor.ay, 0, 0, 0.2, 10, "#fff4c8", 28, 0, 0.8);
          g.fx.glowP(bp.x, bp.y, 0, 0, 0.2, 10, "#fff4c8", 28, 0, 0.8);
        }
      }
      for (let i = g.eb.items.length - 1; i >= 0; i--) {
        const b = g.eb.items[i];
        if (Math.hypot(b.x - cor.ax, b.y - cor.ay) < 54) {
          const bp = cor.bs[bi++ % cor.bs.length];
          const tg = g.nearest(bp.x, bp.y, 99999, (e) => e.y > bp.y - 30);
          const a = tg ? Math.atan2(tg.y - bp.y, tg.x - bp.x) : Math.PI / 2;
          g.spawnPB(bp.x, bp.y, a, 760, B + 24, 22 * dm, { pierce: 1, src: 2, life: 1.6, homing: 3 });
          g.eb.kill(i);
          g.addUlt(0.25);
        }
      }
      if (cor.t >= cor.dur) k.cor = null;
    }
  },
  fire(g) {
    const p = g.p;
    const k = p.k;
    const lv = g.lvl("weapon");
    const dm = g.dmgMult();
    p.fireT += fireGap(g, 0.2);
    k.n++;
    // 多重射擊卡：長矛兩側向外對稱加矛
    const ex = Math.min(3, g.p.s.extra | 0);
    for (const s of [-1, 1]) {
      g.spawnPB(p.x + s * 9, p.y - 30, -Math.PI / 2, 900, B + 19, 10 * dm, { pierce: 2, life: 1.4, r: 8 });
      for (let i = 0; i < ex; i++) g.spawnPB(p.x + s * (9 + 9 * (i + 1)), p.y - 30, -Math.PI / 2, 900, B + 19, 10 * dm, { pierce: 2, life: 1.4, r: 8 });
    }
    if (k.n % 4 === 0) {
      const tg = strongest(g);
      if (tg) {
        openGate(g, tg);
        if (lv >= 3) {
          const others = enemyList(g).filter((e) => e !== tg);
          if (others.length) openGate(g, others[Math.floor(Math.random() * others.length)]);
        }
      }
    }
    g.sfx("shoot1", 0.07);
  },
  onKill(g) {
    const k = g.p.k;
    const lv = g.lvl("passive");
    k.vault++;
    const need = 16 - (lv >= 3 ? 4 : 0);
    if (k.vault >= need) {
      k.vault = 0;
      k.vaultT = 5 + (lv >= 5 ? 2 : 0);
      k.vf = 0;
      g.toast("寶庫開啟！", "#ffd24a");
      g.sfx("gear");
    }
  },
  castActive(g) {
    const p = g.p;
    const lv = g.lvl("active");
    p.k.cor = { t: 0, dur: 7 + (lv >= 5 ? 3 : 0), ax: p.x, ay: p.y - 76, bs: [{ x: W / 2, y: 200 }], re: 0 };
    g.toast("門之回廊", "#ffd24a");
    g.sfx("morph");
    g.fx.ring(p.x, p.y - 76, 8, 70, 0.5, "#ffd24a", 5);
  },
  castUlt(g) {
    const lv = g.lvl("ult");
    const dur = 7;
    ultBegin(g, dur);
    const dm = g.dmgMult() * (lv >= 5 ? 1.3 : 1);
    const nG = lv >= 3 ? 16 : 12;
    const gates = Array.from({ length: nG }, (_, i) => ({ x: 36 + ((i + 0.5) * (W - 72)) / nG, y: 70 + (i % 2) * 54 + Math.sin(i * 1.7) * 12, t0: i * 0.06, nx: 1.15 + i * 0.045 }));
    interface Wp { x0: number; y0: number; tx: number; ty: number; t0: number; kind: number; done: boolean }
    const wps: Wp[] = [];
    const swordT = 5.0;
    const impT = 5.55;
    let impacted = false;
    let sndT = 0;
    const iy = H * 0.62;
    ultScene(g, {
      dur,
      dark: 0.62,
      update(dt, t) {
        sndT -= dt;
        gates.forEach((q, i) => {
          while (t >= q.nx && q.nx < 4.5) {
            q.nx += 0.36;
            const list = enemyList(g);
            const tg: Enemy | undefined = g.boss && i % 4 === 0 ? g.boss : list[Math.floor(Math.random() * list.length)];
            wps.push({ x0: q.x, y0: q.y + 20, tx: tg ? tg.x + rand(-26, 26) : rand(40, W - 40), ty: tg ? tg.y + rand(-20, 20) : rand(200, 760), t0: t, kind: Math.floor(Math.random() * 5), done: false });
            if (sndT <= 0) { sndT = 0.07; g.sfx("shoot1", 0.05); }
          }
        });
        for (const w of wps) {
          if (!w.done && t >= w.t0 + 0.3) {
            w.done = true;
            g.aoe(w.tx, w.ty, 58, 56 * dm, { pal: "gold", small: true, noText: true });
            g.fx.ring(w.tx, w.ty, 6, 56, 0.3, "#ffd24a", 3);
            if (Math.random() < 0.4) g.shake(3, 0.1);
          }
        }
        for (let i = wps.length - 1; i >= 0; i--) if (wps[i].done && t > wps[i].t0 + 0.7) wps.splice(i, 1);
        if (!impacted && t >= impT) {
          impacted = true;
          g.flash("#ffffff", 0.95);
          g.shake(20, 1);
          for (const e of enemyList(g)) g.damageEnemy(e, (Math.abs(e.x - W / 2) < 140 + e.r ? 540 : 170) * dm);
          g.clearBullets(W / 2, H / 2, 99999, "coin");
          g.fx.ring(W / 2, iy, 20, 800, 0.9, "#ffd24a", 10);
          g.fx.ring(W / 2, iy, 20, 560, 0.8, "#ffffff", 6);
          g.sfx("boom_l");
          if (lv >= 7) {
            for (const e of enemyList(g)) if (Math.abs(e.y - iy) < 70 + e.r) g.damageEnemy(e, 320 * dm);
            g.later(0.15, () => g.fx.ring(W / 2, iy, 20, 900, 0.8, "#2ff2d2", 7));
          }
          for (let i = 0; i < 16; i++) g.dropPickup(rand(60, W - 60), rand(100, 500), PK.COIN, 3);
        }
      },
      draw(c, t, k) {
        // 奧義金色氛圍光：全屏 lighter 呼吸光（大爺：「奧義時要更有光芒」）
        //   ⚠ 用 lighter 疊在暗幕上，只加亮不遮畫面；alpha 壓在 0.05~0.09 呼吸。
        c.save();
        c.globalCompositeOperation = "lighter";
        const br = (0.05 + 0.025 * Math.sin(t * 3.2)) * k;
        const bgg = c.createRadialGradient(W / 2, 120, 40, W / 2, 120, H * 0.9);
        bgg.addColorStop(0, `rgba(255,236,170,${br * 2.2})`);
        bgg.addColorStop(0.45, `rgba(255,210,74,${br})`);
        bgg.addColorStop(1, "rgba(255,210,74,0)");
        c.fillStyle = bgg;
        c.fillRect(0, 0, W, H);
        c.restore();
        // 放射光芒
        lightRays(c, W / 2, 90, 14, 1100, t * 0.2, "#ffe9a8", 0.1 * Math.min(1, t) * k);
        gates.forEach((q) => {
          const o = clamp((t - q.t0) / 0.5, 0, 1);
          if (o > 0) ring(c, q.x, q.y, 24 * o, t + q.x * 0.01, 0.95 * k);
        });
        // 飛行武具
        for (const w of wps) {
          const u = clamp((t - w.t0) / 0.3, 0, 1);
          if (u >= 1) continue;
          const x = w.x0 + (w.tx - w.x0) * u * u;
          const y = w.y0 + (w.ty - w.y0) * u * u;
          const ang = Math.atan2(w.ty - w.y0, w.tx - w.x0);
          c.save();
          c.globalCompositeOperation = "lighter";
          c.strokeStyle = "rgba(255,226,140,0.5)";
          c.lineWidth = 5;
          c.beginPath(); c.moveTo(x - Math.cos(ang) * 120 * u, y - Math.sin(ang) * 120 * u); c.lineTo(x, y); c.stroke();
          c.restore();
          // 武具本體金色光暈（越大越亮 → 奧義尾段更有光芒）
          c.save();
          c.globalCompositeOperation = "lighter";
          c.globalAlpha = 0.35 + 0.25 * u;
          const wr = 40 + u * 26;
          c.drawImage(glow("#ffd24a", 128), x - wr, y - wr, wr * 2, wr * 2);
          c.restore();
          weapon(c, w.kind, x, y, ang, 0.55 + u * 0.4);
        }
        // 神劍
        if (t >= 4.5) {
          const gq = clamp((t - 4.5) / 0.5, 0, 1);
          ring(c, W / 2, 150, 110 * gq, t * 1.4, k, "#fff4c8");
          if (t >= swordT) {
            const u = clamp((t - swordT) / (impT - swordT), 0, 1);
            const y = 120 + (iy - 120) * u * u;
            c.save();
            c.globalCompositeOperation = "lighter";
            const gr = c.createLinearGradient(W / 2 - 70, 0, W / 2 + 70, 0);
            gr.addColorStop(0, "rgba(255,210,74,0)"); gr.addColorStop(0.5, "rgba(255,250,220,0.65)"); gr.addColorStop(1, "rgba(255,210,74,0)");
            c.fillStyle = gr;
            c.fillRect(W / 2 - 70, 0, 140, y);
            c.restore();
            drawDivineSword(c, W / 2, y, 1.0 * (0.8 + u * 0.45), t);
          }
        }
        if (t < 1.2) bigText(c, "武 庫 解 放", W / 2, H * 0.45, 66, "#ffe08a", clamp(1.3 - t, 0, 1), "#2a1e08");
        if (impacted && t < impT + 1.3) bigText(c, "神 劍 ‧ 天 落", W / 2, H * 0.36, 58, "#ffffff", clamp(1 - (t - impT) / 1.3, 0, 1), "#7a5a10");
      },
    });
  },
  drawUnder(g, c) {
    const k = g.p.k;
    for (const q of k.gates as Gate[]) {
      const o = q.t < 0.2 ? q.t / 0.2 : q.t > q.life - 0.2 ? (q.life - q.t) / 0.2 : 1;
      ring(c, q.x, q.y, 26 * clamp(o, 0, 1), g.t * 2, 1);
      if (q.tgt && !q.tgt.dead && !q.fired) {
        c.save();
        c.strokeStyle = "rgba(255,226,140,0.4)";
        c.lineWidth = 1.5;
        c.setLineDash([6, 6]);
        c.beginPath(); c.moveTo(q.x, q.y); c.lineTo(q.tgt.x, q.tgt.y); c.stroke();
        c.restore();
      }
    }
    if (k.vaultT > 0) {
      const gy = Math.max(120, g.p.y - 200);
      const a = clamp(k.vaultT, 0, 1);
      ring(c, 70, gy, 30, g.t, a);
      ring(c, W - 70, gy, 30, -g.t, a);
    }
    const cor = k.cor;
    if (cor) {
      const a = clamp(cor.dur - cor.t, 0, 1);
      ring(c, cor.ax, cor.ay, 34, g.t * 1.5, a, "#2ff2d2");
      for (const b of cor.bs as { x: number; y: number }[]) {
        ring(c, b.x, b.y, 34, -g.t * 1.5, a);
        c.save();
        c.globalCompositeOperation = "lighter";
        c.strokeStyle = `rgba(255,226,140,${0.18 * a})`;
        c.lineWidth = 2;
        c.setLineDash([10, 10]);
        c.beginPath(); c.moveTo(cor.ax, cor.ay); c.lineTo(b.x, b.y); c.stroke();
        c.restore();
      }
    }
    if (k.vault > 0) {
      c.save();
      c.font = '900 11px "Noto Sans TC"';
      c.fillStyle = "#ffd24a";
      c.textAlign = "center";
      c.fillText(`寶庫 ${k.vault}/${16 - (g.lvl("passive") >= 3 ? 4 : 0)}`, g.p.x, g.p.y + 54);
      c.restore();
    }
  },
  preview: (t) => ({ mode: 0, aux: Math.sin(t) }),
};
