/**
 * 焰 — 一句話機制：雙股螺旋烈焰越燒越快（熱度），連續擊墜疊怒焰，主動技化為無敵突進三連斬，終極化身炎皇揮舞巨劍。
 * 拓撲：螺旋｜命中：殘留灼燒｜節奏：變頻（熱度）｜被動：連擊｜主動：突進斬｜奧義：變身
 */
const HELIX = EXT_BASE + 20;
const FIREBALL = EXT_BASE + 26;
// 主動技＝三絕斬（與 kagari 交換）；子彈色系跟隨烈焰
const BLADE_FIRE = EXT_BASE + 39;    // 斬擊刃（對齊飛行方向）
const BLADE_FIRE_DS = EXT_BASE + 40; // 歸位刃雨（對齊＋自轉）
interface Floor { x: number; y: number; t: number; dur: number; r: number; tick: number }
interface Ult { t: number; dur: number; ang: number; len: number; cd: Map<number, number>; tipT: number; slam: boolean }
interface Dash {
  i: number;
  n: number;
  phase: "go" | "hold" | "back";
  t: number;
  fx: number;
  fy: number;
  tx: number;
  ty: number;
  hx: number;
  hy: number;
  used: number[];
  r: number;
  /** 施放當下的 `k.formStamp`。dash 結束要還原 form／formT 時，只有戳記還相同才還原
   *  （戳記一變＝期間又發生了新的變身，例如「主動技後立刻接奧義」→ 不可以把奧義的變身蓋掉）。 */
  stamp: number;
}
interface Ghost { x: number; y: number; a: number }

function segDist(px: number, py: number, x0: number, y0: number, x1: number, y1: number) {
  const dx = x1 - x0, dy = y1 - y0, l2 = dx * dx + dy * dy || 1;
  const t = clamp(((px - x0) * dx + (py - y0) * dy) / l2, 0, 1);
  return Math.hypot(px - (x0 + dx * t), py - (y0 + dy * t));
}

// ── 三絕斬（主動技，原 kagari）輔助：鎖定強敵 → 三段突進 → 歸位 ──
function pickTarget(g: Game, used: number[]): { x: number; y: number; uid: number } | null {
  let best: Enemy | null = null;
  let bs = -1;
  for (const e of g.en.items) {
    if (e.dead || !e.alive || e.y < 30 || e.y > g.p.y + 20 || used.includes(e.uid)) continue;
    const s = (e.boss ? 1e6 : 0) + (e.def.mid ? 5e4 : 0) + e.maxHp;
    if (s > bs) {
      bs = s;
      best = e;
    }
  }
  if (!best) best = g.nearest(g.p.x, g.p.y - 100);
  return best ? { x: best.x, y: Math.min(best.y + 34, g.p.y + 10), uid: best.uid } : null;
}

// 十字斬視覺（色系跟隨烈焰）
function crossSlash(g: Game, x: number, y: number, r: number) {
  let t = 0;
  const a0 = rand(-0.5, 0.5);
  mkFx(
    g,
    2,
    (dt) => {
      t += dt;
      return t < 0.3;
    },
    (ctx) => {
      const u = t / 0.3;
      const len = r * 1.5 * easeOut(Math.min(1, u * 4));
      ctx.save();
      ctx.translate(x, y);
      ctx.globalCompositeOperation = "lighter";
      ctx.lineCap = "round";
      for (const a of [a0 - 0.7, a0 + 0.7]) {
        ctx.save();
        ctx.rotate(a);
        ctx.globalAlpha = 1 - u;
        ctx.strokeStyle = "#ff5a2e";
        ctx.lineWidth = 16 * (1 - u) + 3;
        ctx.beginPath();
        ctx.moveTo(-len, 0);
        ctx.lineTo(len, 0);
        ctx.stroke();
        ctx.strokeStyle = "#fff0c0";
        ctx.lineWidth = 5 * (1 - u) + 1;
        ctx.stroke();
        ctx.restore();
      }
      ctx.restore();
    }
  );
  g.fx.burst(x, y, 16, "#ffb06a", 120, 380, 0.5, 2);
  g.fx.ring(x, y, 6, r, 0.35, "#fff0c0", 3);
}

function slashStrike(g: Game, x: number, y: number, r: number, L: number) {
  g.aoe(x, y, r, 40, { pal: "fire" });
  const t = g.nearest(x, y, r);
  if (t) g.damageEnemy(t, 75 * g.dmgMult());
  crossSlash(g, x, y, r);
  audio.sfx("boom_m", 0.02);
  g.shake(7, 0.15);
  if (L >= 5) {
    for (const s of [-1, 1]) g.spawnPB(x, y, s > 0 ? 0 : Math.PI, 720, BLADE_FIRE, 16, { pierce: 6, life: 0.9, src: 2, kind: 0, scale: 0.8, spin: 0 });
  }
}

export const homuraKit: ShipKit = {
  init(g) {
    g.p.k = { heat: 0, phase: 0, fury: 0, furyT: 0, ult: null as Ult | null, floors: [] as Floor[], dash: null as Dash | null, ghosts: [] as Ghost[], form0: 0, formT0: 0, formStamp: 0 };
  },

  update(g, dt) {
    const p = g.p, k = p.k, Lp = g.lvl("passive"), dm = g.dmgMult();
    // 變身計時：先前缺這段，導致焚天劍皇的 formT 設了 15 之後永不歸零
    //（＝永久劍皇外觀 ＋ 永久 +30% 傷害）。三絕斬期間由下方 dash 段維持，不受影響。
    if (p.formT > 0) p.formT = Math.max(0, p.formT - dt);
    // ── 三絕斬（主動技，原 kagari）：殘影淡出 ＋ 突進狀態機 ──
    {
      const Ld = g.lvl("active");
      for (let i = k.ghosts.length - 1; i >= 0; i--) {
        k.ghosts[i].a -= dt * 3;
        if (k.ghosts[i].a <= 0) k.ghosts.splice(i, 1);
      }
      const d: Dash | null = k.dash;
      if (d) {
        d.t += dt;
        p.inv = Math.max(p.inv, 0.5);
        p.formT = Math.max(p.formT, 0.5);
        if (d.phase === "go") {
          const u = Math.min(1, d.t / 0.17);
          const e = easeOut(u);
          const nx = lerp(d.fx, d.tx, e);
          const ny = lerp(d.fy, d.ty, e);
          p.bank = clamp((nx - p.x) * 0.2, -1, 1);
          p.x = nx;
          p.y = ny;
          p.thrust = 1;
          k.ghosts.push({ x: p.x, y: p.y, a: 0.9 });
          if (u >= 1) {
            slashStrike(g, p.x, p.y - 30, d.r, Ld);
            d.phase = "hold";
            d.t = 0;
          }
        } else if (d.phase === "hold") {
          if (d.t >= 0.15) {
            d.i++;
            if (d.i >= d.n) {
              d.phase = "back";
              d.fx = p.x;
              d.fy = p.y;
              d.t = 0;
              audio.sfx("whoosh");
            } else {
              const tg = pickTarget(g, d.used);
              d.fx = p.x;
              d.fy = p.y;
              if (tg) {
                d.tx = tg.x;
                d.ty = tg.y;
                d.used.push(tg.uid);
              } else {
                d.tx = clamp(p.x + rand(-120, 120), 40, W - 40);
                d.ty = clamp(p.y - 120, 120, H - 100);
              }
              d.phase = "go";
              d.t = 0;
              audio.sfx("whoosh");
            }
          }
        } else {
          const u = Math.min(1, d.t / 0.28);
          const e = easeInOut(u);
          p.x = lerp(d.fx, d.hx, e);
          p.y = lerp(d.fy, d.hy, e);
          p.bank = 0;
          k.ghosts.push({ x: p.x, y: p.y, a: 0.5 });
          if (u >= 1) {
            k.dash = null;
            // 還原使用前的變身狀態：原本是劍皇 → 保持劍皇；原本是原形 → 變回原形。
            // ⚠ 只有在「期間沒有發生新的變身」時才還原：主動技後立刻接奧義時，
            //   castUlt 會把 formStamp 推進 → 這裡若照還原就會把奧義的 formT=15 蓋成 0
            //   → 劍皇外觀與 +30% 傷害整場消失（2026-10-11 大爺回報的「面板顯示 +0%」）。
            if (d.stamp === k.formStamp) {
              p.form = k.form0;
              p.formT = k.formT0;
            }
            p.inv = Math.max(p.inv, 0.7);
            if (Ld >= 7) {
              // 歸位時降下刃雨
              for (let i = 0; i < 16; i++) {
                g.later(i * 0.04, () => {
                  const x = rand(40, W - 40);
                  g.spawnPB(x, -20, Math.PI / 2 + rand(-0.15, 0.15), 900, BLADE_FIRE_DS, 22, { pierce: 4, life: 1.4, src: 2, kind: 0, scale: 0.8, spin: 14, rot: 0 });
                });
              }
              g.fx.ring(p.x, p.y, 10, 200, 0.5, "#ff5a2e", 4);
            }
          }
        }
      }
    }
    k.heat = Math.min(1, k.heat + dt / 4);
    const cap = Lp >= 7 ? 15 : 10, per = Lp >= 7 ? 0.05 : Lp >= 3 ? 0.04 : 0.03;
    if (k.furyT > 0) k.furyT -= dt; else if (k.fury > 0) k.fury = Math.max(0, k.fury - dt * 1.5);
    k.fury = Math.min(cap, k.fury);
    p.dmgMul = (1 + k.fury * per) * (p.formT > 0 ? 1.3 : 1);
    for (let i = k.floors.length - 1; i >= 0; i--) {
      const f: Floor = k.floors[i];
      f.t += dt; f.tick -= dt;
      if (f.tick <= 0) { f.tick = 0.3; g.aoe(f.x, f.y, f.r, 28 * dm, { noFx: true, noText: true, burn: true }); }
      if (f.t > f.dur) k.floors.splice(i, 1);
    }
    const u: Ult | null = k.ult;
    if (u) {
      u.t += dt;
      const Lu = g.lvl("ult");
      const Ls = 640 * (Lu >= 3 ? 1.2 : 1) * clamp(u.t / 0.8, 0, 1);
      const omega = Lu >= 5 ? 3.4 : 2.5;
      const raise = u.t > u.dur - 1.5 ? clamp((u.t - (u.dur - 1.5)) / 0.9, 0, 1) : 0;
      u.ang = lerp(-Math.PI / 2 + Math.sin(u.t * omega) * 1.2, -Math.PI / 2, raise);
      u.len = Ls * (1 + raise * 0.15);
      const bx = p.x, by = p.y - 12, tx = bx + Math.cos(u.ang) * u.len, ty = by + Math.sin(u.ang) * u.len;
      for (const [id, c] of u.cd) if (c > 0) u.cd.set(id, c - dt);
      for (const e of g.en.items) {
        if (!e.alive || e.dead) continue;
        if (segDist(e.x, e.y, bx, by, tx, ty) < e.r + 44) {
          if ((u.cd.get(e.uid) ?? 0) <= 0) { u.cd.set(e.uid, 0.12); g.damageEnemy(e, 32 * dm, { noText: true }); e.burn = Math.max(e.burn, 3); }
        }
      }
      for (let s = 0; s <= u.len; s += 48) g.clearBullets(bx + Math.cos(u.ang) * s, by + Math.sin(u.ang) * s, 54, "none");
      if (Lu >= 5) {
        u.tipT -= dt;
        if (u.tipT <= 0 && u.len > 200) {
          u.tipT = 0.18;
          g.spawnPB(tx, ty, u.ang + rand(-0.15, 0.15), 720, FIREBALL, 30 * dm, { homing: 3, life: 2, r: 12, burn: 3, src: 2, pierce: 1 });
        }
      }
      if (Math.random() < 0.9) { const s = rand(u.len); g.fx.glowP(bx + Math.cos(u.ang) * s + rand(-20, 20), by + Math.sin(u.ang) * s, rand(-30, 30), rand(-120, -30), 0.5, rand(14, 34), Math.random() < 0.5 ? "#ff7a2e" : "#ffc24a", 4, 0.5, 0.8); }
      if (!u.slam && u.t >= u.dur - 0.55) {
        u.slam = true;
        const dmg = 420 * dm * (Lu >= 7 ? 2 : 1) * (1 + 0.08 * (Lu - 1));
        for (const e of g.en.items) if (e.alive && !e.dead) { g.damageEnemy(e, dmg, { crit: true }); g.fx.explosion(e.x, e.y, 34, PAL.fire); }
        g.clearBullets(0, 0, 0, "score", 0, W, 0, H);
        g.flash("#fff0c0", 1); g.shake(30, 1);
        for (let i = 0; i < 3; i++) g.fx.ring(p.x, p.y - 200, 20, 600 + i * 120, 0.8 + i * 0.15, i % 2 ? "#ffc24a" : "#ff5a2e", 9);
        g.fx.burst(p.x, p.y - 300, 80, "#ffc24a", 200, 900, 1, 3);
        audio.sfx("boom_l", 0.05);
        if (Lu >= 7) for (let i = 0; i < 7; i++) k.floors.push({ x: W * (0.07 + i * 0.143), y: H * 0.42 + (i % 2) * 90, t: 0, dur: 4, r: 85, tick: 0.1 } as Floor);
      }
      if (u.t >= u.dur) k.ult = null;
    }
    if (p.form && p.formT <= 0) { p.form = 0; p.formT = 0; k.ult = null; }
  },

  fire: wrapFire((g) => {
    const p = g.p, k = p.k, L = g.lvl("weapon"), dm = g.dmgMult();
    const full = k.heat >= 0.99;
    const n = full && L >= 7 ? 4 : 2;
    for (let i = 0; i < n; i++) {
      g.spawnPB(p.x, p.y - 18, -Math.PI / 2, 860, HELIX, 4.6 * dm * (full && L >= 7 ? 1.2 : 1), {
        kind: 20, r: 8, life: 1.1, burn: 2.5, a: k.phase + (i / n) * TAU, b: p.x, c: 20 + k.heat * 10, src: 0, selfMove: 1,
      });
    }
    if (full && L >= 3) g.spawnPB(p.x, p.y - 24, -Math.PI / 2, 1000, FIREBALL, 7 * dm, { r: 10, life: 1.1, burn: 3, pierce: 1, scale: 0.7, src: 0 });
    k.phase += 0.85;
    // 射速：÷ 升級倍率（= p.s.rate / base.rate），讓「射速強化」卡片生效；×base.rate 抵銷以維持原基準射速
    p.fireT += (lerp(0.115, 0.05, k.heat) * g.ship.base.rate) / p.s.rate;
    audio.sfx("shoot2", 0.06);
  }),

  updateBullet(g, b, dt) {
    if (b.style !== HELIX) return;
    b.y += b.vy * dt;
    b.x = b.b + Math.sin(b.t * 13 + b.a) * b.c;
    if (Math.random() < 0.18) g.fx.glowP(b.x, b.y + 6, 0, 30, 0.25, 12, "#ff7a2e", 3, 0, 0.5);
  },

  onKill(g, e) {
    const p = g.p, k = p.k, Lp = g.lvl("passive"), dm = g.dmgMult();
    const cap = Lp >= 7 ? 15 : 10;
    k.fury = Math.min(cap, k.fury + 1); k.furyT = 3;
    if (e.burn > 0 && g.lvl("weapon") >= 5) g.aoe(e.x, e.y, 75, 32 * dm, { pal: "fire", small: true });
    if (Lp >= 5 && k.fury >= cap) g.aoe(e.x, e.y, 95, 46 * dm, { pal: "fire" });
  },

  onPlayerHit(g) {
    g.p.k.heat *= 0.5;
  },

  castActive(g) {
    // 緋走・三絕斬（原 kagari 主動技）：無敵突進，鎖定強敵三連斬後歸位
    const p = g.p, k = p.k;
    const L = g.lvl("active");
    const n = L >= 5 ? 4 : 3;
    const used: number[] = [];
    const first = pickTarget(g, used);
    k.formStamp++; // 新的變身事件：讓進行中的 dash 還原失效（見 Dash.stamp）
    const d: Dash = {
      i: 0, n, phase: "go", t: 0, fx: p.x, fy: p.y,
      tx: first ? first.x : clamp(p.x, 40, W - 40), ty: first ? first.y : clamp(p.y - 260, 120, H - 100),
      hx: p.x, hy: p.y, used, r: 110 * (L >= 3 ? 1.25 : 1),
      stamp: k.formStamp,
    };
    if (first) used.push(first.uid);
    k.dash = d;
    // 記錄使用前的變身狀態，結束時還原（避免三絕斬把焚天劍皇的變身吃掉）
    k.form0 = p.form;
    k.formT0 = p.formT;
    p.form = 2;
    p.formT = Math.max(p.formT, 3);
    p.inv = Math.max(p.inv, 3);
    g.fx.ring(p.x, p.y, 6, 70, 0.3, "#ff5a2e", 4);
    audio.sfx("whoosh");
  },

  castUlt(g) {
    const p = g.p, k = p.k;
    const DUR = 9;       // 奧義演出（巨劍橫掃 → 收招爆炸），維持原本長度
    const FORM_DUR = 15; // 劍皇變身持續（外觀 ＋ 1.3x 傷害），比演出長、獨立計時
    k.formStamp++;       // 新的變身事件：進行中的三絕斬 dash 結束時不可再還原（否則會蓋掉這次變身）
    p.form = 2; p.formT = FORM_DUR; p.inv = Math.max(p.inv, DUR + 1); p.ultBusy = DUR;
    k.ult = { t: 0, dur: DUR, ang: -Math.PI / 2, len: 0, cd: new Map<number, number>(), tipT: 0, slam: false } as Ult;
    ultBackdrop(g, { dur: DUR, color: "#ff5a2e", dark: 0.7, title: "焚天劍皇 ・ 炎皇降臨", fadeOut: 0.9 });
    ultTimerBar(g, DUR, "#ff7a2e");
    g.tint("#ff3a10", 0.2, DUR);
    // 地面火海與升騰餘燼（場景元素）
    let t = 0;
    mkFx(g, 0, (dt) => {
      t += dt;
      if (Math.random() < 0.9) g.fx.glowP(rand(W), H + 10, rand(-20, 20), rand(-260, -120), rand(0.8, 1.4), rand(10, 26), Math.random() < 0.5 ? "#ff7a2e" : "#ffc24a", 3, 0.2, 0.7);
      return t < DUR;
    }, (ctx) => {
      const k = clamp(Math.min(t / 0.8, (DUR - t) / 0.8), 0, 1);
      ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.7 * k;
      const gr = ctx.createLinearGradient(0, H * 0.7, 0, H);
      gr.addColorStop(0, "rgba(255,90,30,0)");
      gr.addColorStop(1, "rgba(255,160,60,.9)");
      ctx.fillStyle = gr; ctx.fillRect(0, H * 0.7, W, H * 0.3);
      for (let i = 0; i < 9; i++) {
        const x = (i / 8) * W, h = 90 + Math.sin(t * 6 + i * 1.7) * 40;
        ctx.fillStyle = "rgba(255,120,40,.5)";
        ctx.beginPath(); ctx.moveTo(x - 40, H); ctx.quadraticCurveTo(x, H - h * 1.6, x + 40, H); ctx.fill();
      }
      ctx.restore();
    });
  },

  drawUnder(g, ctx) {
    // === 火海 (Floors) ===
    for (const f of g.p.k.floors as Floor[]) {
      const a = clamp(Math.min(f.t / 0.3, (f.dur - f.t) / 0.5), 0, 1);
      if (a <= 0) continue; // 效能優化：完全透明時不渲染

      ctx.save();
      ctx.translate(f.x, f.y); // 將座標系移至火海中心，簡化後續數學運算
      ctx.globalCompositeOperation = "lighter";
      ctx.globalAlpha = a;

      // 熔岩基底：以多頻擾動半徑繪製有機池緣，避免單純圓形
      const pulse = 1 + Math.sin(g.time * 6 + f.x * 0.05) * 0.1;
      ctx.globalAlpha = 0.75 * a;
      ctx.fillStyle = "rgba(190,35,0,.6)";
      ctx.beginPath();
      const baseR = f.r * 1.2 * pulse;
      for (let i = 0; i <= 28; i++) {
        const ang = (i / 28) * Math.PI * 2;
        const wob = 1 + Math.sin(ang * 5 + g.time * 2.5) * 0.09 + Math.cos(ang * 3 - g.time * 1.8) * 0.07;
        const rr = baseR * wob;
        const px = Math.cos(ang) * rr;
        const py = Math.sin(ang) * rr * 0.32; // 透視壓扁，製造地面感
        if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
      }
      ctx.closePath();
      ctx.fill();

      // 底部岩漿光暈（雙層暖色核心）
      ctx.globalAlpha = 0.55 * a;
      const r = f.r * 1.55 * pulse;
      ctx.drawImage(glow("#ff2200", 64), -r, -r * 0.75, r * 2, r * 1.5);
      ctx.drawImage(glow("#ffaa22", 64), -r * 0.6, -r * 0.45, r * 1.2, r * 0.9);

      // 主火焰柱：分三層（外焰、內焰、焰心），以相位差創造自然搖曳
      const FL = 6;
      for (let i = 0; i < FL; i++) {
        const phase = g.time * (8.5 + i * 0.7) + i * 1.8 + f.x * 0.04;
        const fx = ((i - (FL - 1) / 2) / ((FL - 1) / 2)) * f.r * 0.6;
        const sway = Math.sin(phase) * (12 + i * 2.5);
        const sway2 = Math.sin(phase * 1.3 + 1) * 6;
        const fh = 60 + Math.sin(phase * 1.1) * 32 + i * 7;
        const fw = 18 + Math.cos(phase * 0.8) * 5 + i * 2;
        const bx = fx + sway;
        const by = 10;

        // 外焰（暗紅橘，較寬）
        ctx.globalAlpha = 0.5 * a;
        ctx.fillStyle = "rgba(255,60,15,.9)";
        ctx.beginPath();
        ctx.moveTo(bx - fw, by);
        ctx.bezierCurveTo(bx - fw * 1.45 + sway2, by - fh * 0.4, bx - fw * 0.65 + sway, by - fh * 0.85, bx + sway * 1.2, by - fh);
        ctx.bezierCurveTo(bx + fw * 0.65 + sway, by - fh * 0.85, bx + fw * 1.45 + sway2, by - fh * 0.4, bx + fw, by);
        ctx.closePath();
        ctx.fill();

        // 內焰（亮橘，較窄）
        ctx.globalAlpha = 0.8 * a;
        ctx.fillStyle = "rgba(255,135,35,.95)";
        ctx.beginPath();
        ctx.moveTo(bx - fw * 0.62, by);
        ctx.bezierCurveTo(bx - fw * 0.95 + sway2 * 0.5, by - fh * 0.35, bx - fw * 0.38 + sway * 0.6, by - fh * 0.78, bx + sway * 0.8, by - fh * 0.82);
        ctx.bezierCurveTo(bx + fw * 0.38 + sway * 0.6, by - fh * 0.78, bx + fw * 0.95 + sway2 * 0.5, by - fh * 0.35, bx + fw * 0.62, by);
        ctx.closePath();
        ctx.fill();

        // 焰心（黃白，細長跳動）
        ctx.globalAlpha = 0.9 * a;
        ctx.fillStyle = "rgba(255,230,150,1)";
        ctx.beginPath();
        ctx.moveTo(bx - fw * 0.24, by);
        ctx.quadraticCurveTo(bx + sway * 0.45, by - fh * 0.55, bx + sway * 0.5, by - fh * 0.45);
        ctx.quadraticCurveTo(bx + sway * 0.4, by - fh * 0.5, bx + fw * 0.24, by);
        ctx.closePath();
        ctx.fill();
      }
      ctx.restore();
    }

    // === 絕斬突進殘影 ===
    const gh = g.p.k.ghosts as Ghost[];
    if (gh.length) {
      const sum = g.lvl("weapon") + g.lvl("passive") + g.lvl("active") + g.lvl("ult");
      const tier = sum >= 22 ? 3 : sum >= 16 ? 2 : sum >= 10 ? 1 : 0;
      for (const q of gh) {
        ctx.save();
        ctx.translate(q.x, q.y);
        const o = makeOpts("homura", g.t, tier);
        ctx.globalAlpha = Math.max(0, q.a * 0.5);
        o.thrust = 1;
        ctx.globalCompositeOperation = "lighter";
        drawShip(ctx, "homura", o);
        ctx.restore();
      }
    }
  },

  drawOver(g, ctx) {
    const p = g.p, k = p.k;
    const u: Ult | null = k.ult;

    // === 巨大火劍 (Ult) ===
    if (u && u.len > 4) {
      const L = u.len;
      ctx.save();
      // 效能與代碼極簡化：先平移並旋轉，之後所有的繪製只需考慮 X(長度) 與 Y(寬度)
      ctx.translate(p.x, p.y - 12);
      ctx.rotate(u.ang); 
      
      ctx.globalCompositeOperation = "lighter";

      // 刀刃動態脈動
      const timePulse = Math.sin(g.time * 25);
      
      // 刀刃多層次堆疊：[半寬度, 尖端寬度, 顏色, 長度比例]
      const layers: [number, number, string, number][] = [
        [35, 12, "rgba(255, 50, 0, 0.25)", 1.05],     // 最外層熱空氣扭曲/暗紅
        [22, 6, "rgba(255, 120, 30, 0.6)", 1.0],      // 主體火炎橘
        [10, 2, "rgba(255, 230, 100, 0.9)", 0.95],    // 高溫焰心黃
        [3 + timePulse, 0, "rgba(255, 255, 255, 1)", 0.9] // 白熱化核心雷射
      ];

      for (const [w, ew, col, ls] of layers) {
        const l = L * ls;
        const taperPoint = l * 0.85; // 開始收縮為劍尖的位置
        
        ctx.fillStyle = col;
        ctx.beginPath();
        ctx.moveTo(0, w);                  // 底部上緣
        ctx.lineTo(taperPoint, ew);        // 劍身上緣
        ctx.lineTo(l, 0);                  // 劍尖端
        ctx.lineTo(taperPoint, -ew);       // 劍身下緣
        ctx.lineTo(0, -w);                 // 底部下緣
        ctx.closePath();
        ctx.fill();
      }

      // 劍尖高能爆發光暈
      ctx.globalAlpha = 0.8;
      ctx.drawImage(glow("#ff9a2e", 64), L - 60, -60, 120, 120);
      ctx.drawImage(glow("#ffffff", 64), L - 25, -25, 50, 50);

      // 十字護手與能量噴射口 (取代原本單調的直線)
      ctx.globalAlpha = 0.9;
      // 噴射口光暈
      ctx.fillStyle = "rgba(255, 180, 50, 0.8)";
      ctx.beginPath();
      ctx.ellipse(0, 0, 12, 35 + timePulse * 5, 0, 0, Math.PI * 2);
      ctx.fill();

      // 實體/高亮護手曲線
      ctx.strokeStyle = "#fff0c0"; 
      ctx.lineWidth = 3; 
      ctx.beginPath(); 
      ctx.moveTo(-10, -40); 
      ctx.quadraticCurveTo(8, 0, -10, 40); // 帶有流線型的護手
      ctx.stroke();

      ctx.restore();
    }

    // === 狀態條 (保持原樣) ===
    drawBars(ctx, p.x, p.y + 40, 40, 4, k.heat, k.heat >= 0.99 ? "#ffffff" : "#ff7a2e");
    if (k.fury > 0) drawBars(ctx, p.x, p.y + 47, 40, 3, k.furyT > 0 ? Math.min(1, k.fury / (g.lvl("passive") >= 7 ? 15 : 10)) : 0.1, "#ffd23f");
  },
};
