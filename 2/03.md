### 1.1 色票工具 `src/game/render/gearPalette.ts`（**✅ 已建立，本節供你理解 API**）

**純函式、零 import**（避免循環依賴，也能被任何家族安全引用）。**已經在專案裡了，你不用建立它**，
直接 `import` 使用即可（API 速查在 §4.1）。以下是它的完整內容，供你確認行為：

```ts
// src/game/render/gearPalette.ts
// 裝備外觀色票工具：從機體色票推導「同色系金屬」。
// 目的：解決「各家族 drawGear 把金屬件寫死成鋼鐵灰／固定稀有度色 → 對非銀灰色系機體像外掛零件」。
// ⚠ 純函式、零 import。只產生顏色字串，不得呼叫任何 ctx 繪製 API（那會動到圖元索引）。

export interface RGB { r: number; g: number; b: number }

export function hex2rgb(h: string): RGB {
  let s = h.replace("#", "");
  if (s.length === 3) s = s[0] + s[0] + s[1] + s[1] + s[2] + s[2];
  const n = parseInt(s.slice(0, 6), 16) || 0;
  return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 };
}

export function rgb2hex(r: number, g: number, b: number): string {
  const c = (v: number) => Math.max(0, Math.min(255, Math.round(v))).toString(16).padStart(2, "0");
  return "#" + c(r) + c(g) + c(b);
}

export function hsl2hex(h: number, s: number, l: number): string {
  h = ((h % 360) + 360) % 360;
  const c = (1 - Math.abs(2 * l - 1)) * s;
  const x = c * (1 - Math.abs(((h / 60) % 2) - 1));
  const m = l - c / 2;
  let r = 0, g = 0, b = 0;
  if (h < 60) { r = c; g = x; } else if (h < 120) { r = x; g = c; } else if (h < 180) { g = c; b = x; }
  else if (h < 240) { g = x; b = c; } else if (h < 300) { r = x; b = c; } else { r = c; b = x; }
  return rgb2hex((r + m) * 255, (g + m) * 255, (b + m) * 255);
}

/** hex → HSL（h 0..360, s/l 0..1） */
export function rgb2hsl(h: string): { h: number; s: number; l: number } {
  const { r, g, b } = hex2rgb(h);
  const R = r / 255, G = g / 255, B = b / 255;
  const max = Math.max(R, G, B), min = Math.min(R, G, B), d = max - min;
  const l = (max + min) / 2;
  const s = d === 0 ? 0 : d / (1 - Math.abs(2 * l - 1));
  let hh = 0;
  if (d !== 0) {
    if (max === R) hh = ((G - B) / d) % 6;
    else if (max === G) hh = (B - R) / d + 2;
    else hh = (R - G) / d + 4;
  }
  return { h: (hh * 60 + 360) % 360, s, l };
}

const _cache = new Map<string, string>();

/**
 * 取 base 的色相 → 低飽和、指定明度的金屬色（近中性色則直接用純灰）。
 *
 * ⚠ 中性色判定必須用「色度 chroma = (max−min)/max」，**不可用 HSL 飽和度**：
 *   HSL 的 s = d/(1−|2l−1|)，在接近白／黑時分母趨近 0 → 極小的 d 也會算出很大的 s。
 *   例：`#f7f3ee`（墨染主色，實質是白）HSL s = 0.35 → 會被誤判成「暖橘」，
 *   推出橘褐色金屬，正是大爺回報的「不合」。用 chroma 判 → 0.036 → 正確走純灰。
 *
 * @param l 明度 0..1  @param s 飽和 0..1
 */
export function metal(base: string | undefined, l: number, s = 0.16): string {
  if (!base) return hsl2hex(0, 0, l);
  const key = base + "|" + l + "|" + s;
  const hit = _cache.get(key);
  if (hit) return hit;
  const { r, g, b } = hex2rgb(base);
  const max = Math.max(r, g, b), min = Math.min(r, g, b), d = max - min;
  const chroma = max > 0 ? d / max : 0;
  let out: string;
  if (chroma < 0.1) {
    out = hsl2hex(0, 0, l); // 白／黑／近灰 → 純灰金屬
  } else {
    let h = 0;
    if (max === r) h = ((g - b) / d) % 6;
    else if (max === g) h = (b - r) / d + 2;
    else h = (r - g) / d + 4;
    out = hsl2hex((h * 60 + 360) % 360, s, l);
  }
  _cache.set(key, out);
  return out;
}

/** 稀有度明度階梯（沿用 2026-10-12 在 shipUtil 用過的那組） */
export const LADDER: Record<number, { l: number; s: number }> = {
  1: { l: 0.34, s: 0.28 }, // R
  2: { l: 0.48, s: 0.26 }, // SR
  3: { l: 0.62, s: 0.24 }, // SSR
  4: { l: 0.76, s: 0.20 }, // UR
};

/** 單一稀有度 → 金屬色（tier 0 → 空字串，代表「未裝備」） */
export function gearColor(base: string | undefined, tier: number): string {
  if (!tier) return "";
  const st = LADDER[tier] ?? LADDER[4];
  return metal(base, st.l, st.s);
}

/**
 * 產生 5 格色表：index 0 = 空字串（未裝備），1..4 = R/SR/SSR/UR。
 * 直接取代各家族寫死的 `const GC = ["", "#9fb3c8", ...]`。
 */
export function gearColors(c1: string | undefined, _c2?: string): string[] {
  return ["", gearColor(c1, 1), gearColor(c1, 2), gearColor(c1, 3), gearColor(c1, 4)];
}

/** 同色系明暗調整：dl 明度增減、ds 飽和增減 */
export function shade(base: string, dl: number, ds = 0): string {
  const { h, s, l } = rgb2hsl(base);
  return hsl2hex(h, Math.max(0, Math.min(1, s + ds)), Math.max(0, Math.min(1, l + dl)));
}

/** 兩色線性混合（k=0 取 a、k=1 取 b） */
export function mix(a: string, b: string, k: number): string {
  const A = hex2rgb(a), B = hex2rgb(b);
  return rgb2hex(A.r + (B.r - A.r) * k, A.g + (B.g - A.g) * k, A.b + (B.b - A.b) * k);
}

/** hex → rgba() 字串（canvas 不接受 #rrggbbaa 在所有舊瀏覽器） */
export function rgba(base: string, a: number): string {
  const { r, g, b } = hex2rgb(base);
  return `rgba(${r},${g},${b},${Math.max(0, Math.min(1, a))})`;
}

/** 直接指定 HSL */
export function hsl(h: number, s: number, l: number): string { return hsl2hex(h, s, l); }

/**
 * 從任何家族／任何呼叫點取得機體色票。
 * ⚠ `o.id` 在大部分呼叫點**不存在**（`shipDraw.ts` 的 `makeOpts()` 不產生 id）
 *   → 一律用這個函式讀色，不要依賴 id。
 */
export function shipColors(o: unknown): [string, string] {
  const x = o as { color?: string; color2?: string; c1?: string; c2?: string };
  return [x.color ?? x.c1 ?? "#8b95a8", x.color2 ?? x.c2 ?? "#c9d2e3"];
}
```

## 2. 範本 B — Step 0（**✅ 已於 2026-10-12 完成，本節只留結論與最終程式碼長相**）

> **對外部模型來說，這一節你不需要做任何事。** 它已經在專案裡了。
> 你只需要知道三件事：
> 1. **`drawGear` 的圖元不佔覆蓋索引**（記 `GEAR_OP = -1`）→ 你可以自由增刪 gear 圖元，不會弄壞既有覆蓋。
> 2. 家族的 `drawGear` 現在是 **wrapper**，繪製內文在 **`drawGearImpl`**（做 L1 時插在後者，見 §1.2）。
> 3. `drawGear` 進出時會呼叫 recorder 的 `gearBegin()`／`gearEnd()`，**任何走 `drawGear` 路徑的圖元都自動被歸類為裝備外觀**。

**最終程式碼長相（供你理解上下文，不要重寫它）：**

```ts
// shipOverrides.ts / overrideRuntime.ts

class ShapeRecorder {
  private inGear = false;
  gearBegin(): void { this.inGear = true; }
  gearEnd(): void   { this.inGear = false; }

  private settle(kind: "fill" | "stroke"): void {
    if (this.inGear) {
      const sh = this.touch();
      sh.ops.push(GEAR_OP); sh.gear = true;      // 不消耗 opCount、不查覆蓋表
      /* …填色／線寬／alpha／comp 照抄既有邏輯，最後 this.real[kind]() 直接畫… */
      return;
    }
    const op = this.opCount++;
    /* …原本的邏輯… */
  }
  // fillRect() 內也有同一組 inGear 分支
}

// 17 支家族 drawGear 的 wrapper（以 cf5ShipUtil.ts 為例，其餘同型）

export function drawGear(ctx: CanvasRenderingContext2D, o: ShipDrawOpts, front: boolean) {
  const r = ctx as unknown as { gearBegin?: () => void; gearEnd?: () => void };
  // ⚠ 兩支都要用 typeof 檢查：只檢查其中一支，tsc 會對另一支報 TS2722（possibly undefined）。
  // ⚠ 也**不可以**先解構存成常數再呼叫 —— 方法脫離 this 會在 runtime 炸。
  if (typeof r.gearBegin !== "function" || typeof r.gearEnd !== "function") return drawGearImpl(ctx, o, front);
  r.gearBegin();
  try { drawGearImpl(ctx, o, front); } finally { r.gearEnd(); }
}
```

```ts
// src/game/render/gearThemeDraw.ts
import { TAU } from "../constants";
import { glow } from "./sprites";
import { gearColors, metal, rgba, shade } from "./gearPalette";
import type { GearKind, GearTheme } from "./gearTheme";

/** drawGear 只讀這幾個欄位（見 01 §2.1）。
 *  ⚠ 刻意**不列 gw/gl**：那兩個欄位只有 cfUUtil 有，靠它們會讓其他 13 家族拿不到尺寸。
 *  尺寸一律走 GearTheme 的 w/h。 */
interface Opts {
  t: number; thrust: number; gear: number[];
  color?: string; color2?: string; c1?: string; c2?: string;
}

const DEF = { t: 0, thrust: 1, gear: [0, 0, 0, 0, 0, 0] };

/** 機體半翼展／半機長：**由主題帶**（`02` §1 的「本地範圍」抄過來）。 */
function size(th: GearTheme): { gw: number; gl: number } {
  return { gw: th.w, gl: th.h };
}

/** 主題繪製入口 */
export function drawThemedGear(
  ctx: CanvasRenderingContext2D,
  o: Opts,
  front: boolean,
  th: GearTheme,
): void {
  const g = o.gear ?? DEF.gear;
  const t = o.t ?? 0;
  const { gw, gl } = size(th);
  const C = gearColors(th.c1, th.c2);      // index 0..4，1=R … 4=UR
  const P = th.params ?? {};
  ctx.save();

  if (!front) {
    // ── 徽章（back）：最外圈
    if (th.slots.emblem && g[5] > 0) drawEmblem(ctx, th.slots.emblem, C[g[5]], g[5], t, gl, P);
    // ── 裝甲（back）：側板
    if (th.slots.armor && g[1] > 0) drawArmor(ctx, th.slots.armor, C[g[1]], g[1], t, gw, gl, th.c2, P);
    // ── 引擎（back）：噴口
    if (th.slots.engine && g[2] > 0) drawEngine(ctx, th.slots.engine, C[g[2]], g[2], t, gw, gl, o.thrust ?? 1, P);
  } else {
    if (th.slots.weapon && g[0] > 0) drawWeapon(ctx, th.slots.weapon, C[g[0]], g[0], t, gw, gl, P);
    if (th.slots.core && g[3] > 0) drawCore(ctx, th.slots.core, C[g[3]], g[3], t, gl, P);
    if (th.slots.chip && g[4] > 0) drawChip(ctx, th.slots.chip, C[g[4]], g[4], t, gw, gl, P);
  }
  ctx.restore();
}

// ── 分流器（**這是 `gearThemeDraw.ts` 目前的實際狀態**，不是骨架） ────────
// 每個槽的 default 就是「未實作 kind」會拿到的通用零件。
```

```ts
// ── 通用零件（各家族原本就有的款式，抽出來共用） ─────────────────────
function dashRing(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gl: number, P: Record<string, number>) {
  const r = gl * (P.ringR ?? 1.35);
  ctx.save(); ctx.globalAlpha = 0.28 + 0.1 * lv; ctx.strokeStyle = c; ctx.lineWidth = 1.6 + lv * 0.5;
  ctx.setLineDash([10, 6]); ctx.lineDashOffset = -t * 24;
  ctx.beginPath(); ctx.arc(0, 4, r, 0, TAU); ctx.stroke(); ctx.setLineDash([]);
  if (lv >= 3) { ctx.globalAlpha = 0.5; ctx.lineWidth = 1; ctx.beginPath(); ctx.arc(0, 4, r * 1.12, 0, TAU); ctx.stroke(); }
  ctx.restore();
}
function plate(ctx: CanvasRenderingContext2D, c: string, lv: number, gw: number, gl: number, P: Record<string, number>) {
  ctx.fillStyle = "rgba(20,20,30,.9)"; ctx.strokeStyle = c; ctx.lineWidth = 1.2;
  for (const s of [-1, 1]) {
    ctx.beginPath();
    ctx.moveTo(s * gw * 0.77, -gl * 0.1); ctx.lineTo(s * gw * 1.3, gl * 0.1);
    ctx.lineTo(s * gw * 1.23, gl * 0.7); ctx.lineTo(s * gw * 0.69, gl * 0.6);
    ctx.closePath(); ctx.fill(); ctx.stroke();
  }
}
function nozzle(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, thrust: number, P: Record<string, number>) {
  ctx.globalCompositeOperation = "lighter";
  ctx.globalAlpha = 0.35 + 0.1 * lv + Math.sin(t * 12) * 0.05;
  ctx.drawImage(glow(c, 64), -26, 18, 52, 40);
  ctx.strokeStyle = c; ctx.lineWidth = 1.5;
  ctx.beginPath(); ctx.ellipse(0, gl * 0.79, 14 + lv * 2, 4, 0, 0, TAU); ctx.stroke();
  ctx.globalCompositeOperation = "source-over"; ctx.globalAlpha = 1;
}
function pod(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, P: Record<string, number>) {
  const pairs = Math.ceil(lv / 2);
  for (let i = 0; i < pairs; i++) for (const s of [-1, 1]) {
    const x = s * gw * (0.65 + i * 0.3), y = -gl * 0.05 + i * gl * 0.1;
    ctx.fillStyle = "#1a1c28"; ctx.strokeStyle = c; ctx.lineWidth = 1;
    ctx.beginPath(); ctx.roundRect(x - 2.2, y - 9, 4.4, 14, 2); ctx.fill(); ctx.stroke();
    ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.7 + Math.sin(t * 10 + i) * 0.2;
    ctx.drawImage(glow(c, 32), x - 4, y - 13, 8, 8);
    ctx.globalCompositeOperation = "source-over"; ctx.globalAlpha = 1;
  }
}
function gem(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gl: number, P: Record<string, number>) {
  const y = gl * 0.16;
  ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.55 + Math.sin(t * 5) * 0.2 + lv * 0.06;
  ctx.drawImage(glow(c, 64), -9 - lv, y - 9 - lv, 18 + lv * 2, 18 + lv * 2);

function satellite(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, P: Record<string, number>) {
  ctx.strokeStyle = c; ctx.lineWidth = 1.2;

/** 花瓣路徑（跟 cfUDrawPetal.ts 的 leaf() 同一套曲線） */
function leafPath(ctx: CanvasRenderingContext2D, ang: number, len: number, w: number) {
  ctx.beginPath(); ctx.moveTo(0, 0);
  ctx.quadraticCurveTo(len * 0.5, -w, len, 0);
  ctx.quadraticCurveTo(len * 0.5, w * 0.8, 0, 0);
  ctx.closePath();
  ctx.save(); ctx.rotate(ang); /* 路徑已在本地座標，這裡只為對稱 */ ctx.restore();
}

function petalRing(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gl: number, P: Record<string, number>) {
  const n = P.petals ?? 8, r = gl * (P.ringR ?? 1.25);
  ctx.save(); ctx.globalAlpha = 0.5 + 0.12 * lv; ctx.translate(0, 4); ctx.rotate(t * 0.5);
  for (let i = 0; i < n; i++) {
    ctx.save(); ctx.rotate((i * TAU) / n);
    ctx.translate(0, -r); ctx.rotate(Math.PI / 2);
    leafPath(ctx, 0, gl * 0.26, gl * 0.1);
    ctx.fillStyle = rgba(c, 0.35 + 0.1 * lv); ctx.fill();
    ctx.strokeStyle = c; ctx.lineWidth = 1; ctx.stroke();
    ctx.restore();
  }
  ctx.restore();
}

function petalPlate(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, c2: string, P: Record<string, number>) {
  const curl = P.curl ?? 0.35;
  for (const s of [-1, 1]) {
    ctx.save(); ctx.scale(s, 1);
    ctx.beginPath();
    ctx.moveTo(gw * 0.7, -gl * 0.12);
    ctx.quadraticCurveTo(gw * (1.05 + curl), -gl * 0.05, gw * 1.25, gl * 0.35);
    ctx.quadraticCurveTo(gw * 0.95, gl * 0.55, gw * 0.66, gl * 0.3);
    ctx.closePath();
    ctx.fillStyle = rgba(c2, 0.5); ctx.fill();
    ctx.strokeStyle = c; ctx.lineWidth = 1.3; ctx.stroke();
    if (lv >= 3) {
      ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.5 + 0.3 * Math.sin(t * 4);
      ctx.strokeStyle = c; ctx.lineWidth = 2;
      ctx.beginPath(); ctx.moveTo(gw * 0.75, -gl * 0.05); ctx.quadraticCurveTo(gw * 1.0, 0, gw * 1.15, gl * 0.3); ctx.stroke();
    }
    ctx.restore();
  }
}

function stamen(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, thrust: number, P: Record<string, number>) {
  const spokes = P.spokes ?? 6;

    ctx.restore();

  ctx.drawImage(glow(c, 48), -18, -6, 36, 36);
  ctx.globalCompositeOperation = "source-over"; ctx.globalAlpha = 1;

function branchPod(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, P: Record<string, number>) {

function sakuraCore(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gl: number, P: Record<string, number>) {

function petalFall(ctx: CanvasRenderingContext2D, c: string, lv: number, t: number, gw: number, gl: number, P: Record<string, number>) {
  const n = 3 + lv * 2, sway = P.sway ?? 2.4;
  for (let i = 0; i < n; i++) {
    const ph = i * 1.7;
    const fall = ((t * 0.5 + i * 0.37) % 1);            // 0→1 循環飄落
```

### 3.5 實作注意

1. **所有零件函式的簽名都要帶 `t: number`**（動畫的唯一輸入）。不要用模組層可變狀態傳時間——那違反 `01` R5（殘影／縮圖會抖）。
2. `glow()` 來自 `render/sprites.ts`，回傳的是 `HTMLCanvasElement`，**用 `ctx.drawImage` 畫 → 不消耗圖元索引**，是安全且高效的做法。
3. `ctx.roundRect` 在本專案已有使用（`cf5ShipUtil.ts:152`），可用。
4. **不要用 `ctx.filter` / `ctx.shadowBlur`**（本專案其他家族都沒用，行動裝置會掉幀）。
5. 半透明一律用 `rgba()` 字串，**不要寫 `#rrggbbaa`**（舊 Safari 不支援 8 位 hex）。
6. 若同一台要「六格都換」，`params` 請保持扁平 `Record<string, number>`，不要嵌套。

### 4.1 色票工具（`src/game/render/gearPalette.ts`）

**明度階梯**（`LADDER`）：R `0.34/0.28`、SR `0.48/0.26`、SSR `0.62/0.24`、UR `0.76/0.20`（`明度/飽和`）。

### 4.2 🔴 鐵則：L2 的零件繪製器**只能用原生 `ctx` API**

**不要呼叫各家族自己的繪圖 helper。** 原因：那 17 個家族是從 17 個不同來源搬過來的，
**同名 helper 的簽名幾乎都不一樣**，而且**有些同名函式根本不是同一件事**。實測對照：

| 名字 | 你以為的 | 實際上有幾種簽名 | 陷阱 |
|---|---|---|---|
| `mirror` | 左右鏡像繪製 | **2 種**：`mirror(ctx, (s)=>{…})`（callback 版，9 個家族）／`mirror(pts) → pts`（**點陣版**，回傳新陣列：`ak5`／`cft`／`a`／`ar`） | 點陣版傳 ctx 進去會直接爆 |
| `flame` | 引擎尾焰 | **7 種**參數順序 | `(x,y,len,wid,…)` vs `(x,y,w,len,…)`；`cf5` 多一個 `thrust`；`pl` 把 `thrust` 插在第 5 位；`cfL` 多一個 `phase` |
| `rad` | 徑向漸層 | **2 種** | 原生／`cf5`／`cft`／`a`／`ar`／`pl`／`sg`／`ak5` 是**雙半徑** `(x,y,r0,r1,stops)`；`sv`／`cfU` 是**單半徑** `(x,y,r,stops)` |
| `poly` | 多邊形 | **2 種**且**不同家族** | `cfU`／`cfL` 的 `poly(ctx, pts: number[])` 是**純路徑**（還要自己 `fill()`）；`spCommon` 的 `poly(ctx, pts, fill, stroke?, lw?)` **自帶填色** |
| `bolt` | 閃電 | **8 種以上** | `(seed,segs,amp,color,w)`／`(segs,jitter,color,w)`／`(color,w,seed)`／`(seg,amp,color,lw,seed)`… 幾乎沒有兩個家族相同 |
| `ring` | 圓／橢圓環 | **7 種** | 原生是橢圓 `(x,y,rx,ry,…)`；`ar`／`sg`／`sv` 多了 `dash`／`from,to`；`pl`／`cft` 只有 `(x,y,r,color,w)` |
| `metal` | 金屬色推導 | **3 種完全不同的東西** | `gearPalette.metal(base,l,s)` = 顏色字串；`spCommon.metal(ctx,x0,y0,x1,y1,a,b,mid?)` = **漸層物件**；`cfU.metal(ctx,x0,x1,a,b,c)` = **漸層物件** |
| `glow` | 光暈 | **2 種** | `render/sprites.ts` 的 `glow(color,size) → HTMLCanvasElement`（**要用 `drawImage` 畫**）；`spCommon` 的 `drawGlow(ctx,x,y,r,color,a)`（**直接畫**） |
| `canopy` | 座艙罩 | **2 種** | `sv`：`(ctx,y,w,h,c,t)`；`spCommon`：`(ctx,x,y,w,h,col,t=0)` |
| `both` | 對稱繪製 | 只有 `ak5` 有 | 其他家族沒有；原生要改用 `mirror` |

**所以 L2 零件的寫法只有一套：**

```ts
import { TAU } from "../constants";            // 2π（不要寫 Math.PI * 2 之外的常數名）
import { glow } from "./sprites";              // 光暈貼圖（drawImage 用，不消耗索引）
import { gearColors, metal, rgba, shade, mix } from "./gearPalette";

// ✅ 只用 ctx 原生 API
ctx.save(); ctx.translate(x, y); ctx.rotate(a); ctx.scale(s, 1);
ctx.beginPath(); ctx.moveTo(…); ctx.lineTo(…);
ctx.quadraticCurveTo(cx, cy, ex, ey); ctx.arc(0, 0, r, 0, TAU);
ctx.ellipse(0, 0, rx, ry, rot, 0, TAU); ctx.roundRect(x, y, w, h, r);
ctx.closePath(); ctx.fillStyle = c; ctx.fill(); ctx.strokeStyle = c; ctx.lineWidth = 1.2; ctx.stroke();
ctx.setLineDash([6, 5]); ctx.lineDashOffset = -t * 24;
ctx.createLinearGradient(x0, y0, x1, y1).addColorStop(0, c);
ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.6;
ctx.drawImage(glow(c, 64), x - 32, y - 32, 64, 64);
ctx.restore();

// ❌ 不要用（各家簽名不同，貼過來就會壞）
// mirror / flame / rad / poly / bolt / ring / metal / canopy / exhaust / bladePath
```

### 4.3 哪些 ctx 呼叫會「消耗圖元索引」

外觀編輯器的圖元索引＝**`fill()`／`stroke()`／`fillRect()` 的遞增序**（`01` §3 R1）。
Step 0 之後 `drawGear` 內的圖元不佔索引，所以**你可以自由增刪**——但這個知識在兩個地方仍然重要：

| 呼叫 | 消耗索引？ | 備註 |
|---|---|---|
| `ctx.fill()`／`ctx.stroke()` | ✅ 是 | 一個呼叫＝一個圖元 |
| `ctx.fillRect(x,y,w,h)` | ✅ 是 | 捷徑，**也會**算一個 |
| `ctx.strokeRect()`／`ctx.fillText()`／`ctx.drawImage()` | ❌ 否 | 不進 recorder 的計數器 |
| `beginPath`／`moveTo`／`lineTo`／`arc`／`ellipse`／`quadraticCurveTo`／`bezierCurveTo`／`roundRect`／`closePath`／`clip` | ❌ 否 | 只是建路徑 |
| `save`／`restore`／`translate`／`rotate`／`scale`／`globalAlpha`／`globalCompositeOperation`／`setLineDash`／`lineDashOffset`／`fillStyle`／`strokeStyle`／`lineWidth` | ❌ 否 | 狀態設定 |

**實際影響**：`drawGear` 之外（也就是機體本體）的圖元數**不能任意增刪**，
但你在 L2 只會動到 `gearThemeDraw.ts` 與 `THEME` 表 → **不會碰到機體本體**。

### 4.4 效能與相容性（本專案既有家族都遵守）

- **不要用 `ctx.filter`／`ctx.shadowBlur`**（行動裝置掉幀，其他家族都沒用）。
- 半透明一律 `rgba(r,g,b,a)` 字串，**不要寫 `#rrggbbaa`**（舊 Safari 不支援 8 位 hex）。
- 需要「發光」用 `drawImage(glow(c, size), …)` ＋ `globalCompositeOperation = "lighter"`。
- 幾何形狀**不可**由 `Math.random()` 決定（殘影／縮圖會抖）；動畫只能用傳入的 `t`。
