import type { Game } from "../engine/Game";
import type { Enemy, Fx, PBullet } from "../entities";
import { bolt } from "../render/shipUtil";
import { W, H } from "../constants";

export interface ShipKit {
  init(g: Game): void;
  fire(g: Game, dt: number): void;
  update(g: Game, dt: number): void;
  castActive(g: Game): void;
  castUlt(g: Game): void;
  onKill?(g: Game, e: Enemy): void;
  onHit?(g: Game, b: PBullet, e: Enemy, dmg: number): void;
  onGraze?(g: Game): void;
  onPlayerHit?(g: Game): void;
  updateBullet?(g: Game, b: PBullet, dt: number): void;
  drawUnder?(g: Game, ctx: CanvasRenderingContext2D): void;
  drawOver?(g: Game, ctx: CanvasRenderingContext2D): void;
  /** 展示用：給定時間回傳外觀範例值（供機庫／選機畫面預覽，移植自 arena 的 preview） */
  preview?(t: number): { mode: number; aux: number };
}

/** 建立並註冊技能特效：update 回傳 false 代表結束 */
export function mkFx(g: Game, layer: 0 | 1 | 2, update: (dt: number) => boolean | void, draw: (ctx: CanvasRenderingContext2D) => void): Fx {
  const fx: Fx = {
    alive: true,
    layer,
    update: (dt) => {
      if (update(dt) === false) fx.alive = false;
    },
    draw,
  };
  g.addFx(fx);
  return fx;
}

/** 短暫閃電線 */
export function boltFx(g: Game, x1: number, y1: number, x2: number, y2: number, color: string, life = 0.16, w = 2.6) {
  let t = 0;
  const seed = Math.random() * 100;
  mkFx(
    g,
    1,
    (dt) => {
      t += dt;
      return t &lt; life;
    },
    (ctx) => {
      ctx.globalCompositeOperation = "lighter";
      ctx.globalAlpha = Math.max(0, 1 - t / life);
      bolt(ctx, x1, y1, x2, y2, seed + Math.floor(t * 40), 7, 9, color, w, "#ffffff");
      ctx.globalAlpha = 1;
      ctx.globalCompositeOperation = "source-over";
    }
  );
}

/** 螢幕黑邊（電影感） */
export function drawBars(ctx: CanvasRenderingContext2D, a: number, W: number, H: number) {
  const h = 70 * a;
  ctx.fillStyle = "#000";
  ctx.fillRect(0, 0, W, h);
  ctx.fillRect(0, H - h, W, h);
}

/** 點到線段距離（點 (px,py) 到線段 (x0,y0)-(x1,y1)） */
export function segDist(px: number, py: number, x0: number, y0: number, x1: number, y1: number): number {
  const dx = x1 - x0, dy = y1 - y0;
  const l2 = dx * dx + dy * dy;
  let t = l2 > 0 ? ((px - x0) * dx + (py - y0) * dy) / l2 : 0;
  t = Math.max(0, Math.min(1, t));
  return Math.hypot(px - (x0 + dx * t), py - (y0 + dy * t));
}

/** 特效貼圖（來源 fx 圖集 API 的保留殼，dst 無 fx 圖集）
 *  idx===3：來源 來源機體D 奧義魔法陣貼圖（霓虹紫魔法陣）→ dst 用 canvas 程式重繪
 *  其他 idx：no-op（dst 無對應貼圖）
 *  繪製參數跟來源 drawCell 語意一致：以 (cx,cy) 中心、size 邊長、rot 旋轉、alpha 透明度、mode 混合模式。
 *  來源機體D 兩層反向旋轉：外層 size×1、內層 size×0.62，rot 相反號，重疊出旋渦感。 */
export function fxTex(ctx: CanvasRenderingContext2D, idx: number, cx: number, cy: number, size: number, rot = 0, alpha = 1, mode: GlobalCompositeOperation = "screen") {
  if (idx !== 3) return; // 來源 fx 圖集只有第 3 格（魔法陣）被 來源機體D 用到，其他格不搬
  if (alpha &lt;= 0.01) return;
  const r = size / 2;
  ctx.save();
  ctx.globalAlpha = Math.min(1, alpha);
  ctx.globalCompositeOperation = mode;
  ctx.translate(cx, cy);
  ctx.rotate(rot);
  ctx.scale(r / 100, r / 100); // 把 size 歸一化到 100 單位，內部幾何用 0~100 座標
  // ── 外環符號帶（主紫 #c76bff）──
  ctx.strokeStyle = "rgba(199,107,255,0.6)";
  ctx.lineWidth = 1.6;
  ctx.beginPath();
  ctx.arc(0, 0, 92, 0, Math.PI * 2);
  ctx.stroke();
  ctx.strokeStyle = "rgba(255,95,168,0.45)";
  ctx.lineWidth = 1.2;
  ctx.beginPath();
  ctx.arc(0, 0, 80, 0, Math.PI * 2);
  ctx.stroke();
  // 外環符號（16 個，每 22.5° 一個，主紫 rune）
  ctx.strokeStyle = "rgba(199,107,255,0.85)";
  ctx.lineWidth = 1;
  for (let i = 0; i &lt; 16; i++) {
    const a = (i / 16) * Math.PI * 2;
    const x = Math.cos(a) * 86, y = Math.sin(a) * 86;
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(a + Math.PI / 2);
    ctx.beginPath();
    ctx.moveTo(0, -4);
    ctx.lineTo(0, 4);
    ctx.moveTo(0, -4);
    ctx.lineTo(3, -1);
    ctx.stroke();
    ctx.restore();
  }
  // ── 六角星（兩個反向三角，粉紅 #ff5fa8）──
  ctx.strokeStyle = "rgba(255,95,168,0.8)";
  ctx.lineWidth = 1.6;
  for (const flip of [0, 1]) {
    ctx.beginPath();
    for (let i = 0; i &lt; 3; i++) {
      const a = -Math.PI / 2 + flip * Math.PI + (i / 3) * Math.PI * 2;
      const x = Math.cos(a) * 58, y = Math.sin(a) * 58;
      if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    }
    ctx.closePath();
    ctx.stroke();
  }
  // ── 八角星（兩個反向正方形，主紫 #c76bff）──
  ctx.strokeStyle = "rgba(199,107,255,0.9)";
  ctx.lineWidth = 1.2;
  for (const rot2 of [Math.PI / 8, 0]) {
    ctx.beginPath();
    for (let i = 0; i &lt; 4; i++) {
      const a = -Math.PI / 2 + rot2 + (i / 4) * Math.PI * 2;
      const x = Math.cos(a) * 34, y = Math.sin(a) * 34;
      if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    }
    ctx.closePath();
    ctx.stroke();
  }
  // ── 放射線（12 條，主紫，時鐘刻度感）──
  ctx.strokeStyle = "rgba(199,107,255,0.55)";
  ctx.lineWidth = 0.8;
  for (let i = 0; i &lt; 12; i++) {
    const a = (i / 12) * Math.PI * 2;
    ctx.beginPath();
    ctx.moveTo(Math.cos(a) * 42, Math.sin(a) * 42);
    ctx.lineTo(Math.cos(a) * 78, Math.sin(a) * 78);
    ctx.stroke();
  }
  // ── 中心光點（粉紅 #ff5fa8，符機體主色、不純白）──
  ctx.strokeStyle = "rgba(255,95,168,0.85)";
  ctx.lineWidth = 1.4;
  ctx.beginPath();
  ctx.arc(0, 0, 10, 0, Math.PI * 2);
  ctx.stroke();
  ctx.fillStyle = "rgba(255,95,168,0.6)";
  ctx.beginPath();
  ctx.arc(0, 0, 4, 0, Math.PI * 2);
  ctx.fill();
  ctx.restore();
}

/** 奧義暗場（全螢幕，來源有、dst 無自動版 → 保留實作） */
export function ultDark(ctx: CanvasRenderingContext2D, a: number, color = "#04020a") {
  if (a &lt;= 0.005) return;
  ctx.save();
  ctx.globalAlpha = Math.min(1, a);
  ctx.fillStyle = color;
  ctx.fillRect(-20, -20, W + 40, H + 40);
  ctx.restore();
}

/** 生成某區間內的非死亡敵機清單（含 Boss） */
export function liveEnemies(g: Game): Enemy[] {
  const out: Enemy[] = [];
  for (const e of g.enemies) if (!e.dead && e.y > -30) out.push(e);
  return out;
}

export interface SpawnOpt {
  elite?: boolean;
  dir?: number;
  hpMul?: number;
  a?: number;
  b?: number;
}
export type TEvent =
  | { t: number; k: "spawn"; e: string; x: number; y: number; o?: SpawnOpt }
  | { t: number; k: "event"; name: string; dur: number }
  | { t: number; k: "mid" }
  | { t: number; k: "boss" }
  | { t: number; k: "banner"; text: string; sub?: string }
  | { t: number; k: "flyby"; kind: string }
  | { t: number; k: "waitMid" };

class Builder {
  ev: TEvent[] = [];
  constructor(public R: () => number, public diff: number) {}
  sp(t: number, e: string, x: number, y = -40, o?: SpawnOpt) {
    this.ev.push({ t, k: "spawn", e, x, y, o });
  }
  rr(a: number, b: number) {
    return a + this.R() * (b - a);
  }
  ri(a: number, b: number) {
    return Math.floor(this.rr(a, b + 1));
  }
  n(base: number, i: number) {
    return Math.max(1, Math.round(base * (1 + i * 0.6) * (1 + this.diff * 0.12)));
  }

  // ---- 波次樣板：回傳「這一段持續幾秒」 ----
  // 下面兩個只是「怎麼用 this.sp 排一串敵人」的示範。**波次設計請自行發明**：
  // 進場角度、隊形、時間差、分批交錯、掩護、假退真進……形狀沒有限制，
  // 只要回傳持續秒數，其餘交給 buildTimeline 排。
  line(t: number, i: number, e: string) {
    const n = this.n(5, i);
    for (let k = 0; k &lt; n; k++) this.sp(t + k * 0.3, e, 0.12 + (0.76 * k) / Math.max(1, n - 1));
    return 4;
  }
  vee(t: number, i: number, e: string) {
    const n = this.n(5, i);
    const cx = this.rr(0.3, 0.7);
    for (let k = 0; k &lt; n; k++) {
      const off = k - (n - 1) / 2;
      this.sp(t + Math.abs(off) * 0.3, e, cx + off * 0.09);
    }
    return 4.2;
  }
}

/** 依關卡組出完整時間軸 —— 節奏由你設計，這裡只示範「怎麼組」 */
export function buildTimeline(stage: StageDef, diff: number): TEvent[] {
  const R = makeRng(stage.id * 7919 + 17);
  const b = new Builder(R, diff);
  const len = stage.len;
  const midT = len * 0.5;
  const pool = stage.pool.length ? stage.pool : ["scout"]; // pool 不可為空
  const pick = () => pool[Math.floor(R() * pool.length)];
  b.ev.push({ t: 0.6, k: "banner", text: `STAGE ${stage.id}`, sub: stage.name });
  // 前半段：自己決定排哪些波次、多密、什麼時候插入環境事件
  let t = 1.6;
  for (let guard = 0; t &lt; midT - 7 && guard &lt; 80; guard++) t += (R() &lt; 0.5 ? b.line(t, 0, pick()) : b.vee(t, 0, pick())) * 0.9;
  b.ev.push({ t: midT * 0.55, k: "event", name: stage.events[0], dur: 12 });
  b.ev.push({ t: midT - 3, k: "banner", text: "WARNING", sub: "中型敵艦接近" });
  b.ev.push({ t: midT, k: "mid" });
  b.ev.push({ t: midT + 0.01, k: "waitMid" });
  // 後半段
  const t2 = midT + 3;
  b.ev.push({ t: t2 + 0.5, k: "flyby", kind: stage.id % 3 === 0 ? "whale" : stage.id % 2 ? "ship" : "arc" });
  t = t2 + 1;
  for (let guard = 0; t &lt; len - 8 && guard &lt; 80; guard++) t += (R() &lt; 0.5 ? b.line(t, 0.8, pick()) : b.vee(t, 0.8, pick())) * 0.85;
  b.ev.push({ t: t2 + (len - 8 - t2) * 0.4, k: "event", name: stage.events[stage.events.length - 1], dur: 12 });
  b.ev.push({ t: len - 4.5, k: "banner", text: "WARNING", sub: "巨大反應接近" });
  b.ev.push({ t: len, k: "boss" });
  b.ev.sort((a, c) => a.t - c.t);
  return b.ev;
}

export const ENDLESS_POOL = ["scout", "dart", "wing", "sentry", "kami", "turret", "spinner", "minelayer", "sniper", "ghost", "aegis", "orbiter"];
export const ENDLESS_EVENTS = ["meteor", "lightning", "gale", "laserGrid", "torpedo"];

export const ENDLESS_KINDS: BgKind[] = ["ocean", "city", "ice", "volcano", "cyber", "space"];

export function genEndlessChunk(chunk: number, startT: number, diff: number): TEvent[] {
  const R = makeRng(chunk * 104729 + 31);
  const b = new Builder(R, diff);
  const lvl = Math.min(1.6, chunk * 0.09);
  const i = Math.min(1, lvl);
  const ev = b.ev;
  const bossChunk = chunk > 0 && chunk % 9 === 8;
  const midChunk = chunk > 0 && chunk % 6 === 3;
  const pool = ENDLESS_POOL;
  let t = startT;
  if (chunk === 0) ev.push({ t: startT + 0.5, k: "banner", text: "ENDLESS", sub: "無限模式・撐到最後" });
  for (let k = 0; k &lt; 4 && t &lt; startT + 15; k++) t += (R() &lt; 0.5 ? b.line(t, i, pool[k % pool.length]) : b.vee(t, i, pool[k % pool.length])) * 0.85;
  if (chunk % 4 === 2) ev.push({ t: startT + 4, k: "event", name: ENDLESS_EVENTS[Math.floor(R() * ENDLESS_EVENTS.length)], dur: 12 });
  if (midChunk) {
    ev.push({ t: startT + 12, k: "banner", text: "WARNING", sub: "中型敵艦接近" });
    ev.push({ t: startT + 15, k: "mid" });
    ev.push({ t: startT + 15.01, k: "waitMid" });
  }
  if (bossChunk) {
    ev.push({ t: startT + 12, k: "banner", text: "WARNING", sub: "巨大反應接近" });
    ev.push({ t: startT + 15, k: "boss" });
  }
  if (chunk % 5 === 4) ev.push({ t: startT + 6, k: "flyby", kind: ["ship", "whale", "arc"][chunk % 3] });
  ev.sort((a, c) => a.t - c.t);
  return ev;
}
