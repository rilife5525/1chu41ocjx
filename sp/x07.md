// 奧義 Cut-in 演出（canvas 繪製）：暗化 → 放射線 → 斜切色帶掃過 → 立繪肖像滑入 →
// 中文標題逐字落下 + 駕駛名牌；進場 0.12s 極短白閃，無全白閃屏。
// 移植自 06advanced-TT 的 canvas cutin，改吃 T 版角色資料（SHIPS.cutin / portrait / cut）。
import { loadPortrait } from "../assets";
import { H, W, clamp, easeInOut, easeOut } from "./constants";
import { SHIPS } from "./data/ships";
import type { ShipDef } from "./data/ships";
import type { ShipId } from "./types";
import { drawShip, makeOpts } from "./render/shipDraw";

export const CUTIN_DUR = 1.45; // 與 Game.tryUlt 的 cutT 倒數一致
/** 六台移植機體自帶特殊奧義演出：cutin 快播不續暗 */
export const CUTIN_DUR_FAST = 0.9;
export const FAST_CUTIN_SHIPS: ShipId[] = [];

// ------------------------------------------------------------
// 立繪肖像裁切參數（整體觀感調這裡）
// ------------------------------------------------------------
const CROP_ZOOM = 1.75; // 立繪放大倍率（越大臉越大、裁到胸部以上）
const FACE_POS = 0.42; // 臉部在裁切視窗中由上而下的位置（0.42≈脖子為中心）
const PLOT_W = 0.66; // 立繪視窗寬（佔 W 比例）
const PLOT_H = 0.52; // 立繪視窗高（佔 H 比例）
const PLOT_X = 0.1; // 視窗中心偏右（佔 W 比例，加上 W/2）

/** 十六色 + alpha → rgba 字串 */
function rgba(hex: string, a: number): string {
  const h = hex.replace("#", "");
  const r = parseInt(h.slice(0, 2), 16);
  const g = parseInt(h.slice(2, 4), 16);
  const b = parseInt(h.slice(4, 6), 16);
  return `rgba(${r},${g},${b},${a})`;
}

let dotPattern: CanvasPattern | null = null;
function getDots(ctx: CanvasRenderingContext2D): CanvasPattern | null {
  if (dotPattern) return dotPattern;
  const c = document.createElement("canvas");
  c.width = 12;
  c.height = 12;
  const g = c.getContext("2d")!;
  g.fillStyle = "rgba(0,0,0,0.32)";
  g.beginPath();
  g.arc(6, 6, 2.6, 0, Math.PI * 2);
  g.fill();
  dotPattern = ctx.createPattern(c, "repeat");
  return dotPattern;
}

/**
 * cutin 專用立繪裁切：以角色資料的臉部位置（portrait.focus／focusX）為錨點，
 * 裁出「胸部以上、脖子為中心」的肖像，cover 填滿目標矩形。
 * 來源為 1080×1920 單張立繪（非圖集），水平以 focusX（預設 0.5 圖中心）對準。
 */
function drawPilotBust(
  ctx: CanvasRenderingContext2D,
  ship: ShipDef,
  dx: number,
  dy: number,
  dw: number,
  dh: number
) {
  const img = loadPortrait(ship.id);
  if (!img || !img.complete || !img.naturalWidth) return;
  const iw = img.naturalWidth; // 1080
  const ih = img.naturalHeight; // 1920
  // 裁切高度：zoom 決定肖像緊繃程度（越大越接近「頭像」）
  const sh = clamp(ih / CROP_ZOOM, 200, ih);
  const faceY = (ship.portrait.focus ?? 0.3) * ih;
  const sy = clamp(faceY - sh * FACE_POS, 0, ih - sh);
  // 水平：依視窗長寬比裁，角色以 focusX 水平對準（預設 0.5 圖中心）
  const fx = ship.portrait.focusX ?? 0.5;
  const target = dw / dh;
  let sw = sh * target;
  let sx = fx * iw - sw / 2;
  if (sw > iw) {
    sw = iw;
    sx = 0;
  }
  sx = clamp(sx, 0, iw - sw);
  ctx.drawImage(img, sx, sy, sw, sh, dx, dy, dw, dh);
}

function bandPath(ctx: CanvasRenderingContext2D, cx: number, cy: number, ang: number, x0: number, x1: number, h: number) {
  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(ang);
  ctx.beginPath();
  ctx.rect(x0, -h / 2, x1 - x0, h);
  ctx.restore();
}

/** 逐字落下（黑描邊 + 主色偏移陰影 + 白字） */
function drawTitle(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, fs: number, c1: string, T: number) {
  ctx.font = `900 ${fs}px 'Noto Sans TC','Microsoft JhengHei',sans-serif`;
  ctx.lineJoin = "round";
  let cx = x;
  for (let i = 0; i &lt; text.length; i++) {
    const ch = text[i];
    const a = clamp((T - 0.28 - i * 0.05) / 0.16, 0, 1);
    const w = ctx.measureText(ch).width;
    if (a > 0) {
      ctx.save();
      ctx.translate(cx, y - (1 - a) * 30);
      ctx.globalAlpha = a;
      ctx.lineWidth = Math.max(5, fs * 0.18);
      ctx.strokeStyle = "#000";
      ctx.strokeText(ch, 0, 0);
      ctx.fillStyle = c1;
      ctx.fillText(ch, 5, 6);
      ctx.fillStyle = "#fff";
      ctx.fillText(ch, 0, 0);
      ctx.restore();
    }
    cx += w;
  }
}

/** 繪製奧義 cutin；k 為進度 0~1（= elapsed / CUTIN_DUR）；tier/gear 供機體疾馳沿用玩家機體外觀 */
export function drawCutin(ctx: CanvasRenderingContext2D, shipId: ShipId, k: number, tier = 2, gear: number[] = [0, 0, 0, 0, 0, 0], durOverride?: number) {
  const s = SHIPS[shipId];
  const cut = s.cutin;
  // 六台移植機體：用較短的 dur（機器疾馳更快播完、不續暗）
  const dur = durOverride ?? CUTIN_DUR;
  const T = k * dur;
  const cx = W / 2;
  const cy = H * 0.46;

  // 1. 暗幕（六台移植機體暗度降低，讓自帶 cutinDeco 可見）
  const fast = FAST_CUTIN_SHIPS.includes(shipId);
  const dimAlpha = fast ? 0.55 : 1;
  ctx.fillStyle = rgba("#040210", Math.min(1, T / 0.12) * 0.82 * dimAlpha);
  ctx.fillRect(0, 0, W, H);

  // 1.5 機體自帶 cutinDeco 掛勾：新增機體若要在暗幕上自繪奧義裝飾，在此表登記。
  if (fast) {
    const DECO: Record&lt;string, (ctx: CanvasRenderingContext2D, t: number, W: number, H: number) => void> = {};
    DECO[shipId]?.(ctx, T, W, H);
  }

  // 2. 放射速度線
  ctx.save();
  ctx.globalAlpha = 0.4 * Math.min(1, T / 0.3);
  ctx.strokeStyle = "#ffffff";
  for (let i = 0; i &lt; 34; i++) {
    const a = (i / 34) * Math.PI * 2 + 0.3;
    const r0 = 150 + ((i * 53) % 120) + T * 120;
    const r1 = r0 + 260 + ((i * 97) % 300);
    ctx.lineWidth = 1 + (i % 3);
    ctx.beginPath();
    ctx.moveTo(cx + Math.cos(a) * r0, cy + Math.sin(a) * r0);
    ctx.lineTo(cx + Math.cos(a) * r1, cy + Math.sin(a) * r1);
    ctx.stroke();
  }
  ctx.restore();

  // 3. 主色帶（由左向右展開）
  const sweep = easeOut(clamp(T / 0.24, 0, 1));
  const bandH = H * 0.44;
  const x1 = -W * 1.1 + W * 2.2 * sweep;
  ctx.save();
  bandPath(ctx, cx, cy + 16, cut.angle, -W * 1.1, x1, bandH + 14);
  ctx.fillStyle = "#000";
  ctx.fill();
  ctx.restore();
  ctx.save();
  bandPath(ctx, cx, cy, cut.angle, -W * 1.1, x1, bandH);
  ctx.clip();
  const gr = ctx.createLinearGradient(0, cy - bandH / 2, W, cy + bandH / 2);
  gr.addColorStop(0, cut.c2);
  gr.addColorStop(0.55, cut.c1);
  gr.addColorStop(1, cut.c3);
  ctx.fillStyle = gr;
  ctx.fillRect(-W, 0, W * 3, H);
  const dots = getDots(ctx);
  if (dots) {
    ctx.fillStyle = dots;
    ctx.fillRect(-W, 0, W * 3, H);
  }
  ctx.globalAlpha = 0.22;
  ctx.strokeStyle = "#fff";
  ctx.lineWidth = 2;
  for (let i = 0; i &lt; 4; i++) {
    const yy = cy - bandH / 2 + ((i + 0.5) * bandH) / 4 + (T * 200) % (bandH / 4);
    ctx.beginPath();
    ctx.moveTo(-W, yy);
    ctx.lineTo(W, yy - bandH * 0.3);
    ctx.stroke();
  }
  ctx.restore();
  // 色帶上下白線
  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(cut.angle);
  ctx.strokeStyle = "#ffffff";
  ctx.lineWidth = 4;
  ctx.beginPath();
  ctx.moveTo(-W * 1.1, -bandH / 2);
  ctx.lineTo(x1, -bandH / 2);
  ctx.moveTo(-W * 1.1, bandH / 2);
  ctx.lineTo(x1, bandH / 2);
  ctx.stroke();
  ctx.restore();

  // 4. 立繪肖像視窗（由右側滑入，斜切四邊形，胸部以上特寫）
  const pIn = clamp((T - 0.1) / 0.36, 0, 1);
  const slide = 1 - easeOut(pIn);
  const pw = W * PLOT_W;
  const ph = H * PLOT_H;
  const inset = Math.round(ph * 0.09); // 斜切內縮
  ctx.save();
  ctx.globalAlpha = clamp((T - 0.1) / 0.2, 0, 1);
  ctx.translate(cx + W * PLOT_X + slide * W * 0.55, cy - H * 0.05);
  ctx.rotate(cut.angle * 0.5);
  // 陰影
  ctx.fillStyle = "rgba(0,0,0,0.75)";
  ctx.beginPath();
  ctx.moveTo(-pw / 2 + inset + 10, -ph / 2 + 10);
  ctx.lineTo(pw / 2 + 10, -ph / 2 + 10);
  ctx.lineTo(pw / 2 - inset + 10, ph / 2 + 10);
  ctx.lineTo(-pw / 2 + 10, ph / 2 + 10);
  ctx.closePath();
  ctx.fill();
  // 白邊
  ctx.fillStyle = "#ffffff";
  ctx.beginPath();
  ctx.moveTo(-pw / 2 + inset - 7, -ph / 2 - 7);
  ctx.lineTo(pw / 2 + 7, -ph / 2 - 7);
  ctx.lineTo(pw / 2 - inset + 7, ph / 2 + 7);
  ctx.lineTo(-pw / 2 - 7, ph / 2 + 7);
  ctx.closePath();
  ctx.fill();
  // 肖像（斜切裁切）
  ctx.save();
  ctx.beginPath();
  ctx.moveTo(-pw / 2 + inset, -ph / 2);
  ctx.lineTo(pw / 2, -ph / 2);
  ctx.lineTo(pw / 2 - inset, ph / 2);
  ctx.lineTo(-pw / 2, ph / 2);
  ctx.closePath();
  ctx.clip();
  ctx.fillStyle = cut.c2;
  ctx.fillRect(-pw / 2 - 8, -ph / 2 - 8, pw + 16, ph + 16);
  drawPilotBust(ctx, s, -pw / 2 - 8, -ph / 2 - 8, pw + 16, ph + 16);
  // 底部主色漸層（提亮胸口、压住裁切下緣）
  const pg = ctx.createLinearGradient(0, -ph / 2, 0, ph / 2);
  pg.addColorStop(0, rgba(cut.c1, 0));
  pg.addColorStop(1, rgba(cut.c1, 0.4));
  ctx.fillStyle = pg;
  ctx.fillRect(-pw / 2, -ph / 2, pw, ph);
  // 進場閃光掃過
  const sh = clamp((T - 0.32) / 0.25, 0, 1);
  if (sh > 0 && sh &lt; 1) {
    ctx.globalCompositeOperation = "lighter";
    ctx.fillStyle = `rgba(255,255,255,${0.5 * (1 - sh)})`;
    const sx = -pw / 2 + sh * (pw + 120) - 60;
    ctx.beginPath();
    ctx.moveTo(sx, -ph / 2);
    ctx.lineTo(sx + 50, -ph / 2);
    ctx.lineTo(sx - 30, ph / 2);
    ctx.lineTo(sx - 80, ph / 2);
    ctx.closePath();
    ctx.fill();
  }
  ctx.restore();
  ctx.restore();

  // 5. 標題（左下：中文奧義名逐字落下 + 英文名）
  const name = s.cut.line2; // 中文奧義名
  const fs = Math.min(84, (W * 0.62) / Math.max(4, name.length));
  ctx.save();
  ctx.translate(W * 0.055, H * 0.72);
  ctx.rotate(cut.angle * 0.7);
  // ULT 標籤
  const tagIn = easeInOut(clamp((T - 0.18) / 0.16, 0, 1));
  ctx.save();
  ctx.translate(-64 * (1 - tagIn), -fs - 42);
  ctx.globalAlpha = tagIn;
  ctx.fillStyle = "#e5102e";
  ctx.beginPath();
  ctx.moveTo(8, 0);
  ctx.lineTo(200, 0);
  ctx.lineTo(190, 30);
  ctx.lineTo(0, 30);
  ctx.closePath();
  ctx.fill();
  ctx.fillStyle = "#fff";
  ctx.font = "900 18px 'Noto Sans TC','Microsoft JhengHei',sans-serif";
  ctx.textBaseline = "middle";
  ctx.fillText("ULTIMATE · 奧義", 20, 15);
  ctx.restore();
  // 中文逐字
  ctx.textBaseline = "alphabetic";
  drawTitle(ctx, name, 0, 0, fs, cut.c1, T);
  // 英文名（下方小字）
  const enA = clamp((T - 0.55) / 0.2, 0, 1);
  ctx.globalAlpha = enA;
  ctx.font = `900 italic ${Math.max(18, fs * 0.32)}px 'Bebas Neue','Impact',sans-serif`;
  ctx.lineWidth = 4;
  ctx.strokeStyle = "#000";
  ctx.strokeText(`${s.cut.line1} · ${s.skills.ult.en}`, 4, fs * 0.42);
  ctx.fillStyle = cut.c3;
  ctx.fillText(`${s.cut.line1} · ${s.skills.ult.en}`, 4, fs * 0.42);
  ctx.restore();

  // 6. 駕駛名牌（右上滑入：駕駛名／機體名）
  const nIn = easeInOut(clamp((T - 0.42) / 0.2, 0, 1));
  ctx.save();
  ctx.translate(W - 18 - 236 * (1 - nIn), H * 0.185);
  ctx.globalAlpha = nIn;
  ctx.fillStyle = "#f6f1e7";
  ctx.beginPath();
  ctx.moveTo(-218, 0);
  ctx.lineTo(10, 0);
  ctx.lineTo(0, 34);
  ctx.lineTo(-228, 34);
  ctx.closePath();
  ctx.fill();
  ctx.fillStyle = "#0b0b10";
  ctx.font = "900 20px 'Noto Sans TC','Microsoft JhengHei',sans-serif";
  ctx.textBaseline = "middle";
  ctx.textAlign = "right";
  ctx.fillText(`${s.pilot}／${s.name}`, -14, 18);
  ctx.restore();

  // 6.5 機體疾馳（玩家機體由右往左飛過，帶三層拖影；沿用玩家 tier/gear 外觀）
  const shipK = clamp((T - dur * 0.2) / (dur * 0.5), 0, 1);
  if (shipK > 0 && shipK &lt; 1) {
    const sxp = W + 120 - shipK * (W + 320);
    const sopts = makeOpts(shipId, T, tier, gear);
    ctx.save();
    ctx.translate(sxp, cy + 120 - shipK * 70);
    ctx.rotate(-Math.PI / 2 - 0.15);
    ctx.scale(2.6, 2.6);
    ctx.globalAlpha = 0.95;
    // 拖影（沿機尾方向殘影：本地 +y 經 rotate(-π/2) 後 = 螢幕右 = 飛行後方）
    for (let i = 3; i >= 1; i--) {
      ctx.save();
      ctx.globalAlpha = 0.1;
      ctx.translate(0, i * 16);
      drawShip(ctx, shipId, sopts);
      ctx.restore();
    }
    drawShip(ctx, shipId, sopts);
    ctx.restore();
  }

  // 7. 進場極短白閃
  if (T &lt; 0.12) {
    ctx.fillStyle = `rgba(255,255,255,${0.6 * (1 - T / 0.12)})`;
    ctx.fillRect(0, 0, W, H);
  }
}

// 粒子與特效系統：爆炸、火花、環形衝擊波、漂浮數字。全部走物件池
import { Pool, makeParticle, makeText } from "../entities";
import type { Particle, FloatText } from "../entities";
import { TAU, rand } from "../constants";
import { glow } from "./sprites";

export interface Palette {
  core: string;
  mid: string;
  edge: string;
  smoke: string;
}
export const PAL: Record&lt;string, Palette> = {
  fire: { core: "#fff6c8", mid: "#ffb347", edge: "#ff3a1a", smoke: "#2b1a18" },
  ice: { core: "#f0fdff", mid: "#7fe4ff", edge: "#2f8dff", smoke: "#16283a" },
  elec: { core: "#fffbd0", mid: "#c58bff", edge: "#6a3dff", smoke: "#1a1230" },
  void: { core: "#f2d8ff", mid: "#a45cff", edge: "#3a1a80", smoke: "#0d0618" },
  jade: { core: "#f0fff8", mid: "#3dffb0", edge: "#11a06a", smoke: "#0a2a20" },
  gold: { core: "#ffffff", mid: "#ffd36b", edge: "#ff8a1a", smoke: "#2a2010" },
  cyan: { core: "#ffffff", mid: "#7cf3ff", edge: "#2a9cff", smoke: "#0c2430" },
  red: { core: "#ffffff", mid: "#ff5a6e", edge: "#b0081f", smoke: "#2a0a10" },
  white: { core: "#ffffff", mid: "#e8f0ff", edge: "#9cb4ff", smoke: "#202838" },
};

export class FxSys {
  ps = new Pool&lt;Particle>(makeParticle);
  texts = new Pool&lt;FloatText>(makeText);
  max = 900;
  pMul = 1;
  showText = true;

  private get(): Particle | null {
    if (this.ps.items.length >= this.max) return null;
    const p = this.ps.spawn();
    p.spr = null;
    p.a0 = 1;
    p.drag = 0;
    p.grav = 0;
    p.rot = 0;
    p.vr = 0;
    p.size2 = 0;
    p.kind = 0;
    p.w = 2;
    return p;
  }

  /** 光暈粒子 */
  glowP(x: number, y: number, vx: number, vy: number, life: number, size: number, color: string, size2 = 0, drag = 0, a0 = 1) {
    const p = this.get();
    if (!p) return null;
    p.x = x; p.y = y; p.vx = vx; p.vy = vy; p.life = p.max = life; p.size = size; p.size2 = size2; p.spr = glow(color); p.color = color; p.drag = drag; p.a0 = a0; p.kind = 0;
    return p;
  }

  /** 火花線粒子 */
  spark(x: number, y: number, vx: number, vy: number, life: number, w: number, color: string, drag = 0.5) {
    const p = this.get();
    if (!p) return null;
    p.x = x; p.y = y; p.vx = vx; p.vy = vy; p.life = p.max = life; p.w = w; p.color = color; p.drag = drag; p.kind = 1; p.size = 1;
    return p;
  }

  /** 擴散環 */
  ring(x: number, y: number, r0: number, r1: number, life: number, color: string, width = 4) {
    const p = this.get();
    if (!p) return null;
    p.x = x; p.y = y; p.vx = 0; p.vy = 0; p.life = p.max = life; p.size = r0; p.size2 = r1; p.color = color; p.w = width; p.kind = 2;
    return p;
  }

  shard(x: number, y: number, vx: number, vy: number, life: number, size: number, color: string, grav = 0) {
    const p = this.get();
    if (!p) return null;
    p.x = x; p.y = y; p.vx = vx; p.vy = vy; p.life = p.max = life; p.size = size; p.color = color; p.kind = 3; p.rot = rand(TAU); p.vr = rand(-10, 10); p.grav = grav; p.drag = 0.4;
    return p;
  }

  smoke(x: number, y: number, vx: number, vy: number, life: number, size: number, size2: number, color = "#2b1a18", a0 = 0.5) {
    const p = this.get();
    if (!p) return null;
    p.x = x; p.y = y; p.vx = vx; p.vy = vy; p.life = p.max = life; p.size = size; p.size2 = size2; p.spr = glow(color); p.color = color; p.kind = 4; p.a0 = a0; p.drag = 1.2;
    return p;
  }

  /** 放射狀火花爆發 */
  burst(x: number, y: number, n: number, color: string, s0: number, s1: number, life: number, w = 2.4) {
    n = Math.round(n * this.pMul);
    for (let i = 0; i &lt; n; i++) {
      const a = rand(TAU);
      const s = rand(s0, s1);
      this.spark(x, y, Math.cos(a) * s, Math.sin(a) * s, rand(life * 0.6, life), w, color);
    }
  }

  /** 定向錐形火花 */
  cone(x: number, y: number, ang: number, spread: number, n: number, color: string, s0: number, s1: number, life: number, w = 2.2) {
    n = Math.round(n * this.pMul);
    for (let i = 0; i &lt; n; i++) {
      const a = ang + rand(-spread, spread);
      const s = rand(s0, s1);
      this.spark(x, y, Math.cos(a) * s, Math.sin(a) * s, rand(life * 0.6, life), w, color);
    }
  }

  /** 綜合爆炸：核心閃光 + 火花 + 碎片 + 衝擊環 + 煙 */
  explosion(x: number, y: number, size: number, pal: Palette = PAL.fire) {
    const m = this.pMul;
    this.glowP(x, y, 0, 0, 0.18 + size * 0.006, size * 1.7, pal.core, size * 2.4, 0, 1);
    this.glowP(x, y, 0, 0, 0.32 + size * 0.008, size * 2.1, pal.mid, size * 3, 0, 0.75);
    this.ring(x, y, size * 0.3, size * 2.2, 0.3 + size * 0.005, pal.mid, Math.max(2, size * 0.12));
    const n = Math.round((6 + size * 0.5) * m);
    for (let i = 0; i &lt; n; i++) {
      const a = rand(TAU);
      const s = rand(60, 120 + size * 6);
      this.spark(x, y, Math.cos(a) * s, Math.sin(a) * s, rand(0.25, 0.6), rand(1.6, 3.2), Math.random() &lt; 0.5 ? pal.mid : pal.core);
    }
    const ns = Math.round((2 + size * 0.15) * m);
    for (let i = 0; i &lt; ns; i++) {
      const a = rand(TAU);
      const s = rand(40, 100 + size * 3);
      this.shard(x, y, Math.cos(a) * s, Math.sin(a) * s, rand(0.4, 0.9), rand(2, 4 + size * 0.1), Math.random() &lt; 0.5 ? pal.edge : pal.mid, 120);
    }
    if (size > 14) {
      const nsm = Math.round((2 + size * 0.06) * m);
      for (let i = 0; i &lt; nsm; i++) this.smoke(x + rand(-size, size) * 0.5, y + rand(-size, size) * 0.5, rand(-20, 20), rand(-20, 20), rand(0.6, 1.1), size * 0.7, size * 1.8, pal.smoke, 0.5);
    }
  }

  hit(x: number, y: number, color: string) {
    this.glowP(x, y, 0, 0, 0.09, 12, color, 22, 0, 0.9);
    const n = Math.round(3 * this.pMul);
    for (let i = 0; i &lt; n; i++) {
      const a = rand(-Math.PI, 0) + rand(-0.4, 0.4);
      const s = rand(80, 200);
      this.spark(x, y, Math.cos(a) * s, Math.sin(a) * s, rand(0.1, 0.25), 1.6, color);
    }
  }

  text(x: number, y: number, str: string, color = "#fff", size = 14, life = 0.7) {
    if (!this.showText || this.texts.items.length > 40) return;
    const t = this.texts.spawn();
    t.x = x; t.y = y; t.text = str; t.color = color; t.size = size; t.life = t.max = life; t.vy = -60;
  }

  update(dt: number) {
    const a = this.ps.items;
    for (let i = a.length - 1; i >= 0; i--) {
      const p = a[i];
      p.life -= dt;
      if (p.life &lt;= 0 || !p.alive) {
        this.ps.kill(i);
        continue;
      }
      if (p.drag) {
        const d = Math.max(0, 1 - p.drag * dt);
        p.vx *= d;
        p.vy *= d;
      }
      p.vy += p.grav * dt;
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      p.rot += p.vr * dt;
    }
    const t = this.texts.items;
    for (let i = t.length - 1; i >= 0; i--) {
      const f = t[i];
      f.life -= dt;
      if (f.life &lt;= 0) {
        this.texts.kill(i);
        continue;
      }
      f.y += f.vy * dt;
      f.vy *= 1 - 3 * dt;
    }
  }

  drawSmoke(ctx: CanvasRenderingContext2D) {
    const a = this.ps.items;
    ctx.globalCompositeOperation = "source-over";
    for (let i = 0; i &lt; a.length; i++) {
      const p = a[i];
      if (p.kind !== 4) continue;
      const r = p.life / p.max;
      const s = p.size + (p.size2 - p.size) * (1 - r);
      ctx.globalAlpha = p.a0 * r;
      ctx.drawImage(p.spr!, p.x - s, p.y - s, s * 2, s * 2);
    }
    ctx.globalAlpha = 1;
  }

  draw(ctx: CanvasRenderingContext2D) {
    const a = this.ps.items;
    ctx.globalCompositeOperation = "lighter";
    for (let i = 0; i &lt; a.length; i++) {
      const p = a[i];
      const r = p.life / p.max;
      switch (p.kind) {
        case 0: {
          const s = p.size2 ? p.size + (p.size2 - p.size) * (1 - r) : p.size * (0.4 + r * 0.6);
          ctx.globalAlpha = Math.min(1, p.a0 * r * 1.4);
          ctx.drawImage(p.spr!, p.x - s, p.y - s, s * 2, s * 2);
          break;
        }
        case 1: {
          ctx.globalAlpha = Math.min(1, r * 1.6);
          ctx.strokeStyle = p.color;
          ctx.lineWidth = p.w * (0.4 + r * 0.6);
          ctx.beginPath();
          ctx.moveTo(p.x, p.y);
          ctx.lineTo(p.x - p.vx * 0.045, p.y - p.vy * 0.045);
          ctx.stroke();
          break;
        }
        case 2: {
          const rad = p.size + (p.size2 - p.size) * (1 - r * r);
          ctx.globalAlpha = Math.min(1, r * 1.5);
          ctx.strokeStyle = p.color;
          ctx.lineWidth = Math.max(0.5, p.w * r);
          ctx.beginPath();
          ctx.arc(p.x, p.y, rad, 0, TAU);
          ctx.stroke();
          break;
        }
        case 3: {
          ctx.globalAlpha = Math.min(1, r * 1.8);
          ctx.fillStyle = p.color;
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate(p.rot);
          const s = p.size * (0.5 + r * 0.5);
          ctx.beginPath();
          ctx.moveTo(s, 0);
          ctx.lineTo(-s * 0.6, s * 0.7);
          ctx.lineTo(-s * 0.4, -s * 0.6);
          ctx.closePath();
          ctx.fill();
          ctx.restore();
          break;
        }
      }
    }
    ctx.globalAlpha = 1;
    ctx.globalCompositeOperation = "source-over";
  }

  drawTexts(ctx: CanvasRenderingContext2D) {
    const t = this.texts.items;
    if (!t.length) return;
    ctx.textAlign = "center";
    ctx.lineJoin = "round";
    for (let i = 0; i &lt; t.length; i++) {
      const f = t[i];
      const r = f.life / f.max;
      ctx.globalAlpha = Math.min(1, r * 2);
      ctx.font = `900 ${f.size * (r > 0.85 ? 1 + (r - 0.85) * 3 : 1)}px "Bebas Neue", Impact, "Noto Sans TC", sans-serif`;
      ctx.lineWidth = 3;
      ctx.strokeStyle = "rgba(0,0,0,0.85)";
      ctx.strokeText(f.text, f.x, f.y);
      ctx.fillStyle = f.color;
      ctx.fillText(f.text, f.x, f.y);
    }
    ctx.globalAlpha = 1;
  }

  clear() {
    this.ps.clear();
    this.texts.clear();
  }
}
