function drawUlt(g: Game, ctx: CanvasRenderingContext2D, k: K) {
  const u = k.u, t = g.time;
  ctx.save();
  if (u &lt; 3.1) {
    const panels = [
      { y: 70, h: 230, sk: 30, at: 0 },
      { y: 320, h: 280, sk: -30, at: 0.8 },
      { y: 620, h: 250, sk: 30, at: 1.6 },
    ];
    panels.forEach((pn, i) => {
      const f = easeOut(clamp((u - pn.at) / 0.3, 0, 1));
      if (f &lt;= 0) return;
      ctx.save();
      ctx.translate((1 - f) * (i % 2 ? -W : W), 0);
      ctx.beginPath(); ctx.moveTo(16, pn.y + Math.max(0, pn.sk)); ctx.lineTo(W - 16, pn.y + Math.max(0, -pn.sk)); ctx.lineTo(W - 16, pn.y + pn.h - Math.max(0, pn.sk)); ctx.lineTo(16, pn.y + pn.h - Math.max(0, -pn.sk)); ctx.closePath();
      ctx.fillStyle = "#ffffff"; ctx.fill();
      ctx.save(); ctx.clip();
      if (i === 0) { drawPilot(ctx, { ...g.ship.portrait, focus: g.ship.portrait.focus * 0.85 }, 16, pn.y, W - 32, pn.h, 1.8); ctx.font = `900 44px "Noto Sans TC"`; ctx.fillStyle = K; ctx.textAlign = "right"; ctx.fillText("……！", W - 40, pn.y + pn.h - 30); }
      else if (i === 1) {
        ctx.fillStyle = Y; ctx.fillRect(0, pn.y, W, pn.h);
        ctx.strokeStyle = K; ctx.lineWidth = 2;
        for (let j = 0; j &lt; 40; j++) { const a = (j / 40) * TAU; ctx.beginPath(); ctx.moveTo(W / 2 + Math.cos(a) * 80, pn.y + pn.h / 2 + Math.sin(a) * 80); ctx.lineTo(W / 2 + Math.cos(a) * 500, pn.y + pn.h / 2 + Math.sin(a) * 500); ctx.stroke(); }
        ctx.save(); ctx.translate(W / 2, pn.y + pn.h / 2 + 10); ctx.scale(2.2, 2.2); const o = makeOpts("comic", t, 3, g.gear); o.thrust = 1; drawShip(ctx, "comic", o); ctx.restore();
      } else {
        ctx.fillStyle = "#1a0a2a"; ctx.fillRect(0, pn.y, W, pn.h);
        ctx.font = `900 64px "Noto Sans TC"`; ctx.textAlign = "center";
        for (let j = 0; j &lt; 3; j++) { ctx.save(); ctx.translate(110 + j * 160 + Math.sin(t * 20 + j) * 3, pn.y + pn.h / 2 + 20 + j * 8); ctx.rotate(-0.15 + j * 0.1); ctx.lineWidth = 8; ctx.strokeStyle = "#ffffff"; ctx.strokeText("轟", 0, 0); ctx.fillStyle = "#b07cff"; ctx.fillText("轟", 0, 0); ctx.restore(); }
      }
      ctx.restore();
      ctx.lineWidth = 6; ctx.strokeStyle = K; ctx.stroke();
      ctx.restore();
    });
  } else if (u &lt; 7.6) {
    halftone(ctx, M, 0.18, t);
    halftone(ctx, Y, 0.12, t + 1);
  }
  // 「爆」字
  ctx.textAlign = "center"; ctx.textBaseline = "middle"; ctx.lineJoin = "round";
  for (const s of k.slams) {
    const f = s.t;
    const sc = f &lt; 0.12 ? 3 - (f / 0.12) * 2 : 1 + (f - 0.12) * 0.3;
    const a = f &lt; 0.6 ? 1 : 1 - (f - 0.6) / 0.3;
    ctx.save(); ctx.globalAlpha = Math.max(0, a); ctx.translate(s.x, s.y); ctx.rotate(-0.15); ctx.scale(sc, sc);
    ctx.font = `900 110px "Noto Sans TC"`;
    ctx.lineWidth = 16; ctx.strokeStyle = K; ctx.strokeText("爆", 0, 0);
    ctx.fillStyle = Y; ctx.fillText("爆", 0, 0);
    ctx.fillStyle = M; ctx.globalAlpha *= 0.5; ctx.fillText("爆", 4, 4);
    ctx.restore();
  }
  // 「完」字大印
  if (u >= 7.5) {
    const f = clamp((u - 7.5) / 0.7, 0, 1);
    const sc = 3 - easeOut(f) * 2;
    const a = u > 9 ? 1 - (u - 9) / 0.5 : 1;
    ctx.save(); ctx.globalAlpha = Math.max(0, a); ctx.translate(W / 2, H * 0.42); ctx.rotate(-0.08); ctx.scale(sc, sc);
    ctx.fillStyle = "#e8102e"; ctx.fillRect(-90, -90, 180, 180);
    ctx.strokeStyle = "#ffffff"; ctx.lineWidth = 6; ctx.strokeRect(-78, -78, 156, 156);
    ctx.font = `900 130px "Noto Sans TC"`; ctx.fillStyle = "#ffffff"; ctx.fillText("完", 0, 6);
    ctx.restore();
  }
  ctx.restore();
  void glowAt; void hexA; void bigText;
}

// 雙子 GEMINI：鏡像雙機交叉火網，兩機之間的連結光束切割敵機與敵彈

import type { Game } from "../engine/Game";
import { type ShipKit, mkFx } from "./common";
import { W, H, TAU, rand, clamp, easeInOut } from "../constants";
import { EXT_BASE_ZR } from "../render/pbStylesExt";
import { drawShip, makeOpts } from "../render/shipDraw";
import { iv, lineHit, lineClear, startUlt, tickUlt, ultBackdrop, bigText, glowAt, hexA } from "./zr_util";

const SP = EXT_BASE_ZR + 8, SB = EXT_BASE_ZR + 9;
const P = "#ff7ad9", B = "#5ad8ff";

interface K { alt: number; tx: number; ty: number; rapid: number; over: number; linkT: number; u: number; tick: number; fused: boolean; novaT: number; final: boolean; swapFx: number }
const K = (g: Game) => g.p.k as K;

function twinTarget(g: Game) { return { x: W - g.p.x, y: g.p.y }; }

export const geminiKit: ShipKit = {
  init(g) { g.p.k = { alt: 0, tx: W - g.p.x, ty: g.p.y, rapid: 0, over: 0, linkT: 0, u: -1, tick: 0, fused: false, novaT: 0, final: false, swapFx: 0 } as K; },
  update(g, dt) {
    const k = K(g), p = g.p, pl = g.lvl("passive"), dm = g.dmgMult();
    if (k.rapid > 0) k.rapid -= dt;
    if (k.over > 0) k.over -= dt;
    if (k.swapFx > 0) k.swapFx -= dt;
    // 鏡像機追隨
    const tt = twinTarget(g);
    if (!k.fused) { k.tx += (tt.x - k.tx) * Math.min(1, dt * 14); k.ty += (tt.y - k.ty) * Math.min(1, dt * 14); }
    // 同步連結光束
    if (!k.fused) {
      k.linkT -= dt;
      if (k.linkT &lt;= 0) {
        k.linkT = 0.1;
        const d = Math.hypot(k.tx - p.x, k.ty - p.y);
        const close = clamp(1 - d / W, 0, 1);
        let mul = 1 + close * 2;
        if (pl >= 7 && d &lt; 80) mul *= 3;
        if (k.over > 0) mul *= 4;
        const w = k.over > 0 ? 40 : pl >= 3 ? 22 : 14;
        lineHit(g, p.x, p.y, k.tx, k.ty, w, 4 * dm * mul);
        const n = lineClear(g, p.x, p.y, k.tx, k.ty, w);
        if (n && pl >= 5) g.addUlt(n * 0.3);
      }
    }
    if (p.formT > 0) { k.u += dt; ultStep(g, k, dt); }
    if (tickUlt(g, dt)) { k.u = -1; k.fused = false; k.final = false; }
  },
  fire(g) {
    const p = g.p, k = K(g), lv = g.lvl("weapon");
    const dm = g.dmgMult();
    const n = lv >= 5 ? 3 : lv >= 2 ? 2 : 1;
    const shoot = (x: number, y: number, style: number) => {
      const inward = x &lt; W / 2 ? 1 : -1;
      for (let i = 0; i &lt; n; i++) {
        const a = -Math.PI / 2 + inward * (0.1 + i * 0.07);
        g.spawnPB(x, y - 20, a, 820, style, 8 * dm, { pierce: lv >= 3 ? 1 : 0, r: 6, explode: lv >= 7 ? 24 : 0 });
      }
    };
    if (k.fused) { p.fireT += iv(g, 0.1); return; }
    if (k.alt++ % 2 === 0) shoot(p.x, p.y, SP);
    else { shoot(k.tx, k.ty, SB); if (k.rapid > 0) shoot(k.tx, k.ty - 10, SB); }
    p.fireT += iv(g, 0.075);
    g.sfx("shoot5", 0.07);
  },
  castActive(g) {
    const k = K(g), p = g.p, lv = g.lvl("active"), dm = g.dmgMult();
    const ox = p.x, oy = p.y;
    p.x = clamp(k.tx, 14, W - 14); p.y = k.ty;
    k.tx = ox; k.ty = oy;
    p.inv = Math.max(p.inv, 0.6);
    const R = lv >= 3 ? 150 : 110;
    for (const [x, y] of [[p.x, p.y], [ox, oy]]) {
      g.clearBullets(x, y, R, "score");
      g.aoe(x, y, R * 0.7, 40 * dm, { pal: "white", small: true });
      g.fx.ring(x, y, 10, R, 0.45, x === p.x ? P : B, 4);
      for (let i = 0; i &lt; 10; i++) { const a = rand(TAU); g.fx.glowP(x, y, Math.cos(a) * 260, Math.sin(a) * 260, 0.4, 10, i % 2 ? P : B, 2); }
    }
    k.rapid = lv >= 5 ? 5 : 3;
    if (lv >= 7) k.over = 1.5;
    k.swapFx = 0.3;
    g.sfx("whoosh");
  },
  castUlt(g) {
    const k = K(g);
    startUlt(g, 9);
    k.u = 0; k.tick = 0; k.fused = false; k.final = false; k.novaT = 1.5;
    ultBackdrop(g, 9, 0.55, 0.7);
    bigText(g, "雙星合璧", "BINARY STAR", P, 1.2);
    g.p.form = 0;
    mkFx(g, 1, () => g.p.formT > 0, (ctx) => drawUlt(g, ctx, k));
  },
  drawOver(g, ctx) {
    const k = K(g), p = g.p;
    if (k.fused) return;
    // 連結光束
    ctx.save();
    ctx.globalCompositeOperation = "lighter";
    const w = k.over > 0 ? 40 : g.lvl("passive") >= 3 ? 22 : 14;
    const gr = ctx.createLinearGradient(p.x, p.y, k.tx, k.ty);
    gr.addColorStop(0, hexA(P, 0.6)); gr.addColorStop(1, hexA(B, 0.6));
    ctx.strokeStyle = gr; ctx.lineWidth = w * (0.6 + Math.sin(g.time * 20) * 0.1); ctx.lineCap = "round";
    ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo(k.tx, k.ty); ctx.stroke();
    ctx.strokeStyle = "rgba(255,255,255,0.8)"; ctx.lineWidth = 2;
    ctx.beginPath();
    const n = 20;
    for (let i = 0; i &lt;= n; i++) { const f = i / n; const x = p.x + (k.tx - p.x) * f, y = p.y + (k.ty - p.y) * f + Math.sin(f * 20 + g.time * 30) * 3; i ? ctx.lineTo(x, y) : ctx.moveTo(x, y); }
    ctx.stroke();
    ctx.restore();
    // 鏡像機
    ctx.save();
    ctx.translate(k.tx, k.ty); ctx.scale(0.62, 0.62);
    const o = makeOpts("gemini", g.time + 0.5, g.tier, g.gear);
    o.mode = 1; o.bank = -p.bank; o.thrust = p.thrust;

    drawShip(ctx, "gemini", o);
    ctx.restore();
  },
