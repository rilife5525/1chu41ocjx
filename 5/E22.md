/** 建立並登記時序特效。update 回傳 false 即結束。layer 0 敵機下 / 1 敵機上 / 2 全螢幕最上 */
export function mkFx(g: Game, layer: 0 | 1 | 2, update: (dt: number) => boolean, draw: (ctx: CanvasRenderingContext2D) => void): Fx {
  const fx: Fx = {
    alive: true,
    layer,
    update(dt) {
      if (!update(dt)) fx.alive = false;
    },
    draw,
  };
  g.addFx(fx);
  return fx;
}

/** 閃電折線特效（兩點之間） */
export function boltFx(g: Game, x1: number, y1: number, x2: number, y2: number, color = "#ffffff", life = 0.18, w = 3): Fx {
  let t = 0;
  const pts: [number, number][] = [];
  const n = Math.max(4, Math.floor(Math.hypot(x2 - x1, y2 - y1) / 26));
  for (let i = 0; i &lt;= n; i++) {
    const k = i / n;
    const j = i === 0 || i === n ? 0 : (Math.random() - 0.5) * 22;
    pts.push([x1 + (x2 - x1) * k + j, y1 + (y2 - y1) * k + j * 0.6]);
  }
  return mkFx(g, 1, (dt) => ((t += dt), t &lt; life), (ctx) => {
    const a = 1 - t / life;
    ctx.save();
    ctx.globalCompositeOperation = "lighter";
    ctx.globalAlpha = a;
    ctx.lineJoin = "round";
    for (const [lw, c] of [[w * 3, color], [w, "#ffffff"]] as [number, string][]) {
      ctx.strokeStyle = c;
      ctx.lineWidth = lw;
      ctx.beginPath();
      pts.forEach((p, i) => (i ? ctx.lineTo(p[0], p[1]) : ctx.moveTo(p[0], p[1])));
      ctx.stroke();
    }
    ctx.restore();
  });
}

/** 場上有效敵機（未死、已進入畫面） */
export function enemyList(g: Game): Enemy[] {
  const out: Enemy[] = [];
  for (const e of g.en.items) if (!e.dead && e.y > -30 && e.y &lt; H + 30) out.push(e);
  return out;
}
/** 點到線段距離 */
export function segDist(px: number, py: number, x1: number, y1: number, x2: number, y2: number) {
  const dx = x2 - x1;
  const dy = y2 - y1;
  const l2 = dx * dx + dy * dy || 1;
  const t = clamp(((px - x1) * dx + (py - y1) * dy) / l2, 0, 1);
  return Math.hypot(px - (x1 + dx * t), py - (y1 + dy * t));
}
/** 取得敵群重心（無敵機時回傳螢幕上方中央） */
export function enemyCenter(g: Game): { x: number; y: number; n: number } {
  let x = 0;
  let y = 0;
  let n = 0;
  for (const e of g.en.items) if (!e.dead && e.y > 0 && e.y &lt; H * 0.8) { x += e.x; y += e.y; n++; }
  return n ? { x: x / n, y: y / n, n } : { x: W / 2, y: 260, n: 0 };
}
/** 最強敵（Boss > 精英 > 最高血量） */
export function strongest(g: Game, maxY = H): Enemy | null {
  if (g.boss && !g.boss.dead) return g.boss;
  let best: Enemy | null = null;
  let bs = -1;
  for (const e of g.en.items) {
    if (e.dead || e.y &lt; 10 || e.y > maxY || e.invuln) continue;
    const s = e.hp + (e.elite ? 500 : 0) + (e.def.mid ? 2000 : 0);
    if (s > bs) { bs = s; best = e; }
  }
  return best;
}

/** 奧義開場：鎖定形態時間與無敵 */
export function ultBegin(g: Game, dur: number) {
  g.p.form = 1;
  g.p.formT = dur;
  g.p.inv = Math.max(g.p.inv, dur + 0.6);
}

  const idx = ultIdx(g);
