export const echoKit: ShipKit = {
  init(g) { g.p.k = { hist: [] as Hist[], hAcc: 0, ghosts: [] as Ghost[], moveT: 0, ph: [] as Phantom[], lx: g.p.x, ly: g.p.y }; },

  update(g, dt) {
    const p = g.p, k = p.k, pl = g.lvl("passive"), al = g.lvl("active");
    // 軌跡紀錄（40Hz，保留 4.6 秒）
    k.hAcc += dt;
    if (k.hAcc >= 0.025) { k.hAcc = 0; k.hist.push({ x: p.x, y: p.y, t: g.t }); if (k.hist.length > 190) k.hist.shift(); }
    // 殘像駐留
    const sp = Math.hypot(p.x - k.lx, p.y - k.ly) / Math.max(dt, 0.001); k.lx = p.x; k.ly = p.y;
    if (sp > 120) k.moveT += dt;
    if (k.moveT >= 2.2) {
      k.moveT = 0;
      const ghosts: Ghost[] = k.ghosts, max = pl >= 3 ? 4 : 3;
      if (ghosts.length >= max) ghosts.shift();
      ghosts.push({ x: p.x, y: p.y, t: pl >= 3 ? 4.5 : 3.2, fire: 0 });
      g.fx.ring(p.x, p.y, 10, 60, 0.4, VI, 3); audio.sfx("lock", 0.1);
    }
    const ghosts: Ghost[] = k.ghosts;
    for (let i = ghosts.length - 1; i >= 0; i--) {
      const q = ghosts[i];
      q.t -= dt; q.fire -= dt;
      if (q.fire &lt;= 0) { q.fire = 0.22; needles(g, q.x, q.y, 3.6, S(10), pl >= 7 ? { homing: 0.8 } : {}); }
      if (pl >= 5) for (const eb of g.eb.items) if (eb.alive && Math.hypot(eb.x - q.x, eb.y - q.y) &lt; 26 + eb.r) { eb.alive = false; g.score += 6; g.fx.glowP(eb.x, eb.y, 0, 0, 0.2, 16, TE, 4); }
      if (q.t &lt;= 0) {
        if (pl >= 7) g.aoe(q.x, q.y, 90, 22 * g.dmgMult(), { pal: "void", small: true });
        else g.fx.explosion(q.x, q.y, 12, { core: "#fff", mid: TE, edge: VI, smoke: "#0b1030" });
        ghosts.splice(i, 1);
      }
    }
    // 幻影重播
    const phs: Phantom[] = k.ph;
    for (let i = phs.length - 1; i >= 0; i--) {
      const f = phs[i];
      if (f.wait > 0) { f.wait -= dt; continue; }
      f.u += dt;
      const dur = f.path.length ? f.path[f.path.length - 1].t - f.path[0].t : 0.5;
      const [x, y] = pose(f.path, Math.min(f.u, dur)); f.x = x; f.y = y;
      f.fire -= dt; f.big -= dt;
      if (f.fire &lt;= 0) { f.fire = 0.1; needles(g, x, y, 6, S(10)); }
      if (f.big &lt;= 0) { f.big = 0.5; g.spawnPB(x, y - 30, UP, 1000, S(13), 20, { pierce: 3, r: 10 }); }
      g.clearBullets(x, y, 34, "score");
      if (al >= 5) {
        f.trail.push(x, y); if (f.trail.length > 90) f.trail.splice(0, 2);
        if (Math.floor(f.u * 5) !== Math.floor((f.u - dt) * 5)) for (const e of g.en.items) { if (e.dead || !e.alive) continue; for (let j = 0; j &lt; f.trail.length; j += 4) if (Math.hypot(e.x - f.trail[j], e.y - f.trail[j + 1]) &lt; 20 + e.r * 0.5) { g.damageEnemy(e, 7 * g.dmgMult(), { noText: true, noHitFx: true }); break; } }
      }
      if (f.u >= dur) {
        f.done = true;
        g.aoe(x, y, 125, (38 + al * 4) * g.dmgMult(), { pal: "void" }); g.fx.ring(x, y, 10, 150, 0.5, TE, 5); audio.sfx("boom_m", 0.05);
        if (al >= 7) {
          for (const e of g.en.items) if (!e.dead && e.alive && Math.abs(e.x - x) &lt; 46 + e.r && e.y &lt; y) g.damageEnemy(e, 85 * g.dmgMult(), {});
          g.addFx(mkFx(g, 1, (() => { let tt = 0; return (d: number) => (tt += d) &lt; 0.4; })(), (ctx) => { ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.fillStyle = "rgba(61,245,208,.55)"; ctx.fillRect(x - 40, 0, 80, y); ctx.fillStyle = "#fff"; ctx.fillRect(x - 10, 0, 20, y); ctx.restore(); }));
        }
        phs.splice(i, 1);
      }
    }
  },

  fire(g) {
    const p = g.p, lv = g.lvl("weapon");
    needles(g, p.x, p.y, 7, S(10));
    const x = p.x, y = p.y;
    g.later(0.6, () => echoShot(g, x, y, 1));
    if (lv >= 3) g.later(1.2, () => echoShot(g, x, y, 2));
    p.fireT += 0.12 / p.s.rate;
  },

  castActive(g) {
    const k = g.p.k, al = g.lvl("active");
    const now = g.t, path: Hist[] = (k.hist as Hist[]).filter((h) => h.t >= now - 4.5);
    if (path.length &lt; 4) path.push({ x: g.p.x, y: g.p.y, t: now - 0.3 }, { x: g.p.x, y: g.p.y - 200, t: now });
    const offs = al >= 7 ? [0, 0.9, 1.8] : al >= 3 ? [0, 1.2] : [0];
    for (const w of offs) k.ph.push({ path: path.map((h) => ({ ...h })), u: 0, wait: w, x: path[0].x, y: path[0].y, trail: [], fire: 0, big: 0.2, done: false });
    g.fx.ring(g.p.x, g.p.y, 10, 120, 0.5, VI, 4); g.flash("#8a5cff", 0.2);
  },

  drawUnder(g, ctx) {
    const k = g.p.k;
    for (const q of k.ghosts as Ghost[]) {
      ctx.save(); ctx.translate(q.x, q.y); ctx.globalAlpha = clamp(q.t, 0, 1) * 0.45;
      drawShip(ctx, "echo", { ...makeOpts("echo", g.t, g.tier(), g.gearR), thrust: 0.3 }); ctx.restore();
      ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.strokeStyle = "rgba(138,92,255,.7)"; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(q.x, q.y, 34, -Math.PI / 2, -Math.PI / 2 + TAU * clamp(q.t / 4.5, 0, 1)); ctx.stroke(); ctx.restore();
    }
    for (const f of k.ph as Phantom[]) {
      if (f.wait > 0) continue;
      if (f.trail.length > 4) {
        ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.strokeStyle = "rgba(61,245,208,.5)"; ctx.lineWidth = 6; ctx.lineCap = "round"; ctx.beginPath(); ctx.moveTo(f.trail[0], f.trail[1]);
        for (let i = 2; i &lt; f.trail.length; i += 2) ctx.lineTo(f.trail[i], f.trail[i + 1]); ctx.stroke(); ctx.restore();
      }
      ctx.save(); ctx.translate(f.x, f.y); ctx.globalAlpha = 0.7;
      drawShip(ctx, "echo", { ...makeOpts("echo", g.t, g.tier(), g.gearR), thrust: 0.8 }); ctx.restore();
    }
  },

  castUlt(g) {
    const p = g.p, lv = g.lvl("ult"), DUR = 9.2;
    p.form = 1; p.formT = DUR; p.inv = Math.max(p.inv, DUR + 0.5);
    g.flash("#8a5cff", 0.8); g.shake(8, 0.5); g.tint("#0b1030", 0.4, DUR);
    const COLS = [TE, VI, "#ff3d7f", "#ffc83d", "#35f0ff", "#6dffb5"];
    let t = 0, beat = 0, shot = 0, ended = false;
    const pos = [0, 0, 0, 0, 0, 0].map(() => ({ x: p.x, y: p.y }));
    let beamX = -1, beamHit = 0;
    audio.sfx("charge");
    g.addFx(mkFx(g, 1, (dt) => {
      t += dt;
      for (let i = 0; i &lt; 6; i++) {
        const o = pos[i];
        if (t &lt; 1.4) { const a = (i / 6) * TAU + t * 1.2; o.x = p.x + Math.cos(a) * 120 * Math.min(1, t); o.y = p.y + Math.sin(a) * 120 * Math.min(1, t) - 40; }
        else if (t &lt; 6.4) { const tt = t - 1.4, ph = (i / 6) * TAU; const tx = W / 2 + 190 * Math.sin(tt * 0.9 + ph), ty = 230 + 330 * (0.5 + 0.5 * Math.sin(tt * 0.6 + ph * 1.7)); o.x += (tx - o.x) * Math.min(1, dt * 6); o.y += (ty - o.y) * Math.min(1, dt * 6); }
        else { const a = (i / 6) * TAU + t * 0.8, tx = W / 2 + Math.cos(a) * 120, ty = 380 + Math.sin(a) * 90; o.x += (tx - o.x) * Math.min(1, dt * 5); o.y += (ty - o.y) * Math.min(1, dt * 5); }
      }
      if (t > 1.4 && t &lt; 6.4) {
        shot -= dt;
        if (shot &lt;= 0) { shot = 0.1; for (let i = 0; i &lt; 6; i++) { const e = g.nearest(pos[i].x, pos[i].y, 9999, (q) => q.y > 0); const a = e ? Math.atan2(e.y - pos[i].y, e.x - pos[i].x) : UP; g.spawnPB(pos[i].x, pos[i].y, a + rand(-0.08, 0.08), 950, S(10), 6.5, { homing: 0.9, src: 2 }); } }
        beat -= dt;
        if (beat &lt;= 0) {
          beat = 0.5; audio.sfx("boom_s", 0.05);
          const dmg = (30 + lv * 4) * (lv >= 3 ? 1.3 : 1) * g.dmgMult();
          for (let i = 0; i &lt; 6; i++) { g.aoe(pos[i].x, pos[i].y, 135, dmg, { noFx: true, noText: true }); g.fx.ring(pos[i].x, pos[i].y, 10, 140, 0.4, COLS[i], 4); if (lv >= 5) for (let j = 0; j &lt; 3; j++) g.spawnPB(pos[i].x, pos[i].y, rand(0, TAU), 700, S(11), 6, { homing: 1.2, src: 2 }); }
          g.fx.ring(p.x, p.y, 20, 520, 0.5, "#ffffff", 3); g.shake(3, 0.1);
        }
      }
      // 終曲：合奏光束
      if (t >= 7.2) {
        const sweeps = lv >= 7 ? 2 : 1, k = (t - 7.2) / 1.6 * sweeps;
        if (k &lt; sweeps) { const ph = k % 1, dir = Math.floor(k) % 2 ? 1 - ph : ph; beamX = 40 + dir * (W - 80); }
        else beamX = -1;
        beamHit -= dt;
        if (beamX >= 0 && beamHit &lt;= 0) {
          beamHit = 0.06; const dmg = (44 + lv * 6) * g.dmgMult();
          for (const e of g.en.items) if (!e.dead && e.alive && Math.abs(e.x - beamX) &lt; 80 + e.r * 0.6) g.damageEnemy(e, dmg, { noText: Math.random() &lt; 0.6 });
          g.clearBullets(0, 0, 0, "coin", beamX, 170, 0, H);
          g.shake(4, 0.1); g.fx.glowP(beamX, rand(0, H), 0, 0, 0.3, 40, "#ffffff", 10, 0, 0.7);
        }
        if (!ended && t >= 8.8) { ended = true; g.flash("#ffffff", 0.9); g.fx.text(W / 2, H / 2, "ENCORE!", "#3df5d0", 54, 1.2); audio.sfx("boom_l"); }

      ctx.globalCompositeOperation = "lighter";
      for (let i = 0; i &lt; 4; i++) { const ph = ((t * 2 + i / 4) % 1); ctx.strokeStyle = `rgba(138,92,255,${(1 - ph) * 0.5})`; ctx.lineWidth = 3; ctx.beginPath(); ctx.arc(p.x, p.y, 40 + ph * 560, 0, TAU); ctx.stroke(); }
      ctx.globalCompositeOperation = "source-over";

        const o = pos[i], a = clamp((t - i * 0.18) * 2, 0, 1) * 0.85;
