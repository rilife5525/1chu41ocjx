// 映界 MIRRORGATE —— 一句話機制：兩面鏡門把穿過的光束倍增，並將觸碰的敵彈反射成我方碎光。
// 主武裝＝鏡門倍增；被動＝全反射（鏡能）；主動＝鏡界展開；奧義＝萬華鏡・終景
import { W, H, TAU, clamp, rand } from "../constants";
import { mkFx, segDist, type ShipKit } from "./common";
import { EXT_BASE_LX } from "../render/pbStylesExt";
import { audio } from "../audio";
import type { Game } from "../engine/Game";

const S = (n: number) => EXT_BASE_LX + n;
const UP = -Math.PI / 2;
const CY = "#35f0ff", WH = "#eafcff";
interface Pane { x: number; y: number; hw: number; outer: boolean; side: number }

function shard(g: Game, x: number, y: number, dmg: number, homing = 0.8) {
  const e = g.nearest(x, y, 900, (q) => q.y > 0);
  const ang = e ? Math.atan2(e.y - y, e.x - x) : UP + rand(-0.3, 0.3);
  return g.spawnPB(x, y, ang, 780, S(9), dmg, { homing, src: 2, pierce: 0 });
}

export const kagamiKit: ShipKit = {
  init(g) {
    g.p.k = { panes: [] as Pane[], charge: 0, hall: null as null | { x: number; t: number; hw: number; born: number }, shardAt: 0, webT: 0, shield: 0 };
  },

  update(g, dt) {
    const p = g.p, k = p.k, lv = g.lvl("weapon"), pl = g.lvl("passive");
    const panes: Pane[] = k.panes;
    const want = lv >= 3 ? 4 : 2;
    if (panes.length !== want) {
      panes.length = 0;
      for (const side of [-1, 1]) panes.push({ x: p.x + side * 42, y: p.y - 130, hw: 24, outer: false, side });
      if (want === 4) for (const side of [-1, 1]) panes.push({ x: p.x + side * 100, y: p.y - 168, hw: 28, outer: true, side });
    }
    for (const m of panes) {
      const tx = p.x + m.side * (m.outer ? 100 : 42), ty = p.y - (m.outer ? 168 : 130);
      m.x += (tx - m.x) * Math.min(1, dt * 9); m.y += (ty - m.y) * Math.min(1, dt * 9);
    }
    // 光束倍增
    for (const b of g.pb.items) {
      if (!b.alive || b.kind !== 0 || b.style !== S(8) || b.a >= (lv >= 3 ? 3 : 2)) continue;
      for (const m of panes) {
        const prev = b.y - b.vy * dt;
        if (Math.abs(b.x - m.x) &lt; m.hw && ((prev - m.y) * (b.y - m.y) &lt;= 0 || Math.abs(b.y - m.y) &lt; 6) && b.b !== (m.outer ? 2 : 1) + m.side * 0.1) {
          b.b = (m.outer ? 2 : 1) + m.side * 0.1;
          const raw = b.c * 0.85, sp = Math.hypot(b.vx, b.vy), ang = Math.atan2(b.vy, b.vx);
          b.vx = Math.cos(ang + 0.13) * sp; b.vy = Math.sin(ang + 0.13) * sp; b.style = S(9); b.dmg *= 0.85; b.a++;
          const c = g.spawnPB(b.x, b.y, ang - 0.13, sp, S(9), raw, { a: b.a, b: b.b, c: raw, homing: lv >= 5 ? 0.7 : g.p.s.homing, pierce: lv >= 5 ? 1 : g.p.s.pierce, src: 0 });
          if (lv >= 5) { b.homing = 0.7; b.pierce = Math.max(b.pierce, 1); }
          void c;
          g.fx.glowP(b.x, b.y, 0, 0, 0.25, 22, WH, 6, 0, 0.9);
          g.fx.spark(b.x, b.y, rand(-120, 120), rand(-120, 120), 0.2, 1.5, CY, 2);
          break;
        }
      }
    }
    // 吞彈與鏡能
    const need = pl >= 3 ? 16 : 20;
    let absorbed = 0;
    for (const m of panes) {
      for (const eb of g.eb.items) {
        if (!eb.alive) continue;
        const rr = (pl >= 5 ? 14 : 8);
        if (Math.abs(eb.x - m.x) &lt; m.hw + eb.r + rr && Math.abs(eb.y - m.y) &lt; 10 + eb.r + rr * 0.5) {
          eb.alive = false; absorbed++; k.charge++; g.score += 8;
          if (pl >= 5) g.addUlt(0.15);
          if (g.t >= k.shardAt) { k.shardAt = g.t + 0.06; shard(g, m.x, m.y, 8); }
          g.fx.glowP(eb.x, eb.y, 0, 0, 0.2, 16, WH, 4, 0, 0.9);
        }
      }
    }
    if (absorbed) audio.sfx("graze", 0.05);
    if (k.charge >= need) {
      k.charge = 0; g.toast("映返！", CY); audio.sfx("power"); g.flash("#bff8ff", 0.25);
      for (const m of panes) { g.fx.ring(m.x, m.y, 6, 80, 0.4, CY, 4); for (let i = 0; i &lt; 8; i++) shard(g, m.x, m.y, 9, 1.1); }
      if (pl >= 7) { k.shield = 2; p.inv = Math.max(p.inv, 2); }
    }
    if (k.shield > 0) k.shield -= dt;
    // 鏡牆
    const h = k.hall;
    if (h) {
      h.t -= dt;
      const al = g.lvl("active");
      if (al >= 5) h.x += (p.x - h.x) * Math.min(1, dt * 1.6);
      const xL = h.x - h.hw, xR = h.x + h.hw;
      for (const b of g.pb.items) {
        if (!b.alive || b.kind >= 20 || b.src === 2 && b.c > 6) continue;
        const px = b.x - b.vx * dt;
        for (const wx of [xL, xR]) {
          if ((px - wx) * (b.x - wx) &lt; 0 && b.c &lt; 6) {
            b.vx = -b.vx; b.x = wx + Math.sign(b.vx) * 2; b.c++; b.dmg *= 1.25;
            g.fx.glowP(wx, b.y, 0, 0, 0.2, 18, WH, 5, 0, 0.9);
          }
        }
        if (al >= 7 && (px - h.x) * (b.x - h.x) &lt; 0 && b.a &lt; 3 && b.style === S(9)) {
          b.a++; const sp = Math.hypot(b.vx, b.vy), a0 = Math.atan2(b.vy, b.vx);
          for (const d of [-0.15, 0.15]) g.spawnPB(b.x, b.y, a0 + d, sp, S(9), (b.dmg / Math.max(0.01, g.dmgMult())) * 0.7, { a: b.a, c: b.c });
        }
      }
      for (const eb of g.eb.items) {
        if (!eb.alive) continue;
        const px = eb.x - eb.vx * dt * eb.slow;
        for (const wx of [xL, xR]) if ((px - wx) * (eb.x - wx) &lt; 0) { eb.alive = false; g.score += 8; k.charge++; if (g.t >= k.shardAt) { k.shardAt = g.t + 0.05; shard(g, wx, eb.y, 10); } g.fx.glowP(wx, eb.y, 0, 0, 0.2, 18, CY, 5, 0, 0.9); }
      }
      if (h.t &lt;= 0) { k.hall = null; g.fx.ring(h.x, H / 2, 20, 200, 0.5, CY, 4); }
    }
    // Lv7 稜光線：兩面內側鏡門之間的切割
    if (lv >= 7 && panes.length >= 2) {
      k.webT -= dt;
      if (k.webT &lt;= 0) {
        k.webT = 0.15;
        const a = panes[0], c = panes[1];
        for (const e of g.en.items) if (!e.dead && e.alive && segDist(e.x, e.y, a.x, a.y, c.x, c.y) &lt; 12 + e.r * 0.6) g.damageEnemy(e, 8 * g.dmgMult(), { noText: true, noHitFx: true });
      }
    }
  },

  fire(g) {
    const p = g.p;
    for (const a of [-0.4, -0.2, 0, 0.2, 0.4]) {
      const raw = 6;
      g.spawnPB(p.x, p.y - 32, UP + a, 950, S(8), raw, { b: 0, c: raw, a: 0, r: 6 });

      ctx.fillStyle = gr; ctx.fillRect(wx - 16, 0, 32, H);

      ctx.globalCompositeOperation = "lighter";
      for (let i = 0; i &lt; 5; i++) { const y = ((g.t * 260 + i * 210) % (H + 120)) - 60; const lg = ctx.createLinearGradient(0, y - 40, 0, y + 40); lg.addColorStop(0, "rgba(255,255,255,0)"); lg.addColorStop(0.5, "rgba(255,255,255,.7)"); lg.addColorStop(1, "rgba(255,255,255,0)"); ctx.fillStyle = lg; ctx.fillRect(wx - 6, y - 40, 12, 80); }

    if (g.lvl("active") >= 7) { ctx.strokeStyle = "rgba(255,255,255,.6)"; ctx.setLineDash([10, 8]); ctx.beginPath(); ctx.moveTo(h.x, 0); ctx.lineTo(h.x, H); ctx.stroke(); }
    ctx.restore();
  },
