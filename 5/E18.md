function detonate(g: Game, e: Enemy) {
  const lvp = g.lvl("passive");
  e.mark = 0;
  const R = 90 * (lvp >= 3 ? 1.3 : 1);
  g.aoe(e.x, e.y, R, (30 + lvp * 5) * g.dmgMult(), { pal: "white", noText: false });
  comicWord(g, e.x, e.y, WORDS[Math.floor(Math.random() * WORDS.length)], 80);
  g.shake(3, 0.12); audio.sfx("boom_m", 0.05);
  for (const o of g.en.items) {
    if (o === e || o.dead || !o.alive) continue;
    const d = Math.hypot(o.x - e.x, o.y - e.y);
    if (lvp >= 5 && d &lt; R * 1.5) o.mark += 1;
    if (lvp >= 7 && d &lt; R) o.freeze = Math.max(o.freeze, 1.0);
  }
}

interface Frame { dx: number; w: number; h: number; cx: number; cy: number; snapped: boolean }

export const panelKit: ShipKit = {
  init(g) { g.p.k = { volley: 0, frames: [] as Frame[], ft: 0, fPhase: 0 }; },

  update(g, dt) {
    const k = g.p.k, p = g.p, fr: Frame[] = k.frames;
    if (!fr.length) return;
    k.ft += dt;
    const AIM = 0.9, HOLD = 3.2;
    if (k.fPhase === 0) {
      for (const f of fr) { f.cx = clamp(p.x + f.dx, f.w / 2 + 6, W - f.w / 2 - 6); f.cy = clamp(p.y - 250, f.h / 2 + 70, H - f.h / 2 - 20); }
      if (k.ft >= AIM) {
        k.fPhase = 1; k.ft = 0; g.flash("#ffffff", 0.6); g.shake(5, 0.2); audio.sfx("lock");
        for (const f of fr) {
          f.snapped = true;
          for (const e of g.en.items) if (!e.dead && e.alive && Math.abs(e.x - f.cx) &lt; f.w / 2 + e.r * 0.4 && Math.abs(e.y - f.cy) &lt; f.h / 2 + e.r * 0.4) g.damageEnemy(e, 40 * g.dmgMult(), { noHitFx: false });
          comicWord(g, f.cx, f.cy, "定", 110);
        }
      }
    } else {
      for (const f of fr) {
        for (const e of g.en.items) if (!e.dead && e.alive && Math.abs(e.x - f.cx) &lt; f.w / 2 && Math.abs(e.y - f.cy) &lt; f.h / 2) { if (!e.boss || k.ft &lt; 1.2) e.freeze = Math.max(e.freeze, 0.25); }
        for (const b of g.eb.items) if (b.alive && Math.abs(b.x - f.cx) &lt; f.w / 2 && Math.abs(b.y - f.cy) &lt; f.h / 2) b.slow = 0;
      }
      if (k.ft >= HOLD) {
        const lv = g.lvl("active");
        for (const f of fr) {
          for (const e of g.en.items) if (!e.dead && e.alive && Math.abs(e.x - f.cx) &lt; f.w / 2 + e.r * 0.4 && Math.abs(e.y - f.cy) &lt; f.h / 2 + e.r * 0.4) g.damageEnemy(e, 150 * g.dmgMult());
          g.clearBullets(0, 0, 0, "coin", f.cx, f.w, f.cy - f.h / 2, f.h);
          comicWord(g, f.cx, f.cy, "轟", 130);
          g.fx.explosion(f.cx, f.cy, 50, { core: "#fff", mid: "#f7f2ff", edge: "#ff2f4f", smoke: "#08060f" });
        }
        g.shake(10, 0.4); g.flash("#ffffff", 0.7); audio.sfx("boom_l");
        if (lv >= 7) { g.clearBullets(0, 0, 9999, "coin"); g.addUlt(10); }
        k.frames = []; k.fPhase = 0;
      }
    }
  },

  fire(g) {
    const p = g.p, k = p.k, lv = g.lvl("weapon");
    k.volley++;
    if (k.volley % 2 === 1) for (const dx of [-13, 0, 13]) g.spawnPB(p.x + dx, p.y - 28, UP, 900, S(3), 6, { r: 7 });
    else for (let i = -2; i &lt;= 2; i++) g.spawnPB(p.x, p.y - 26, UP + i * 0.13, 780, S(3), 5, { r: 7 });
    if (k.volley % (lv >= 3 ? 3 : 4) === 0) {
      g.spawnPB(p.x, p.y - 36, UP, 720, S(2), 28, { r: 20, pierce: 3, a: 0 });
      audio.sfx("shoot4", 0.05);
    }
    p.fireT += 0.16 / p.s.rate;
  },

  onHit(g, b, e) {
    if (b.style !== S(3) && b.style !== S(2) && b.style !== S(14)) return;
    const need = g.lvl("weapon") >= 5 ? 2 : 3;
    e.mark += b.style === S(2) ? 2 : 1;
    if (g.lvl("weapon") >= 7 && b.style === S(2) && b.a === 0) {
      b.a = 1;
      for (let i = -1; i &lt;= 1; i++) g.spawnPB(b.x, b.y, UP + i * 0.55, 780, S(14), 14, { pierce: 1, r: 14 });
    }
    if (e.mark >= need) detonate(g, e);
  },

  onKill(g, e) {
    if (e.mark &lt;= 0) return;
    const t = g.nearest(e.x, e.y, 180, (q) => q !== e);
    if (t) { t.mark += e.mark; g.fx.burst(t.x, t.y, 5, "#ff2f4f", 80, 200, 0.3); }
  },

  castActive(g) {
    const k = g.p.k, lv = g.lvl("active"), s = lv >= 3 ? 1.25 : 1;
    k.frames = lv >= 5
      ? [{ dx: -140, w: 200 * s, h: 320 * s, cx: 0, cy: 0, snapped: false }, { dx: 140, w: 200 * s, h: 320 * s, cx: 0, cy: 0, snapped: false }]
      : [{ dx: 0, w: 270 * s, h: 340 * s, cx: 0, cy: 0, snapped: false }];
    k.ft = 0; k.fPhase = 0;
    g.fx.ring(g.p.x, g.p.y, 10, 90, 0.3, "#ff2f4f", 3);
  },

  drawUnder(g, ctx) {
    const k = g.p.k, fr: Frame[] = k.frames;
    if (!fr.length) return;
    ctx.save();
    for (const f of fr) {
      const x = f.cx - f.w / 2, y = f.cy - f.h / 2;
      if (k.fPhase === 0) {
        const near = k.ft / 0.9;
        ctx.fillStyle = `rgba(247,242,255,${0.06 + near * 0.12})`; ctx.fillRect(x, y, f.w, f.h);
        ctx.strokeStyle = "#f7f2ff"; ctx.lineWidth = 3; ctx.setLineDash([14, 8]); ctx.strokeRect(x, y, f.w, f.h); ctx.setLineDash([]);
        ctx.strokeStyle = "#ff2f4f"; ctx.lineWidth = 5;
        for (const [cx, cy, sx, sy] of [[x, y, 1, 1], [x + f.w, y, -1, 1], [x, y + f.h, 1, -1], [x + f.w, y + f.h, -1, -1]]) { ctx.beginPath(); ctx.moveTo(cx + sx * 26, cy); ctx.lineTo(cx, cy); ctx.lineTo(cx, cy + sy * 26); ctx.stroke(); }
        ctx.strokeStyle = "rgba(255,47,79,.7)"; ctx.lineWidth = 1.5;
        ctx.beginPath(); ctx.moveTo(f.cx - 16, f.cy); ctx.lineTo(f.cx + 16, f.cy); ctx.moveTo(f.cx, f.cy - 16); ctx.lineTo(f.cx, f.cy + 16); ctx.stroke();
        stamp(ctx, String(Math.max(1, Math.ceil(0.9 - k.ft + 0.1))), f.cx, f.cy, 70 - near * 20, 0, "rgba(255,255,255,.6)", "#ff2f4f");
      } else {
        const left = 1 - k.ft / 3.2, pat = halftone(ctx);
        ctx.fillStyle = "rgba(247,242,255,.18)"; ctx.fillRect(x, y, f.w, f.h);
        if (pat) { ctx.fillStyle = pat; ctx.globalAlpha = 0.5; ctx.fillRect(x, y, f.w, f.h); ctx.globalAlpha = 1; }
        ctx.save(); ctx.beginPath(); ctx.rect(x, y, f.w, f.h); ctx.clip();
        speedLines(ctx, f.cx, f.cy, 40, Math.max(f.w, f.h), 26, Math.floor(g.t * 12), "rgba(8,6,15,.35)"); ctx.restore();
        ctx.strokeStyle = "#08060f"; ctx.lineWidth = 9; ctx.strokeRect(x, y, f.w, f.h);
        ctx.strokeStyle = left &lt; 0.25 && Math.floor(g.t * 12) % 2 ? "#ff2f4f" : "#f7f2ff"; ctx.lineWidth = 4; ctx.strokeRect(x, y, f.w, f.h);
        ctx.fillStyle = "#ff2f4f"; ctx.fillRect(x, y + f.h + 6, f.w * left, 4);
      }
    }
    ctx.restore();
  },

  drawOver(g, ctx) {
    // 標記顯示：敵人頭上的墨點
    ctx.save();
    for (const e of g.en.items) {
      if (e.dead || !e.alive || e.mark &lt;= 0) continue;
      const need = g.lvl("weapon") >= 5 ? 2 : 3;
      for (let i = 0; i &lt; need; i++) {
        const x = e.x + (i - (need - 1) / 2) * 11, y = e.y - e.r - 12;
        ctx.fillStyle = i &lt; e.mark ? "#ff2f4f" : "rgba(8,6,15,.5)"; ctx.strokeStyle = "#fff"; ctx.lineWidth = 1.4;
        ctx.beginPath(); ctx.arc(x, y, 4, 0, TAU); ctx.fill(); ctx.stroke();
      }
    }
    ctx.restore();
  },

  castUlt(g) {
    const p = g.p, lv = g.lvl("ult"), DUR = 8.8, tx = (t0: number) => t0;
    p.form = 1; p.formT = DUR; p.inv = Math.max(p.inv, DUR + 0.5);
    g.flash("#ffffff", 0.9); g.shake(10, 0.5);
    // 斜切四格
    const xv = (y: number) => W / 2 + 40 - (80 * y) / H, yh = (x: number) => H / 2 - 30 + (60 * x) / W;
    let cx = W / 2, cy = H / 2;
    for (let i = 0; i &lt; 6; i++) { cy = yh(cx); cx = xv(cy); }
    const polys: [number, number][][] = [
      [[0, 0], [xv(0), 0], [cx, cy], [0, yh(0)]], [[xv(0), 0], [W, 0], [W, yh(W)], [cx, cy]],
      [[0, yh(0)], [cx, cy], [xv(H), H], [0, H]], [[cx, cy], [W, yh(W)], [W, H], [xv(H), H]],
    ];
    const quad = (x: number, y: number) => (x &lt; xv(y) ? 0 : 1) + (y &lt; yh(x) ? 0 : 2);
    const centers = polys.map((pl) => [pl.reduce((a, q) => a + q[0], 0) / 4, pl.reduce((a, q) => a + q[1], 0) / 4]);
    const WORD = ["斬", "撃", "裂", "終"], T0 = [1.0, 2.4, 3.8, 5.2], WIN = 1.4;
    const started = [false, false, false, false];
    let t = 0, spread = false, nextTick = 0;
    g.addFx(mkFx(g, 2, (dt) => {
      t += dt; nextTick -= dt;
      for (let i = 0; i &lt; 4; i++) {
        if (!started[i] && t >= T0[i]) {
          started[i] = true; audio.sfx(i === 3 ? "boom_l" : "whoosh"); g.shake(8 + i * 2, 0.3); g.flash("#ffffff", 0.5);
          const dmg = (80 + lv * 14) * (lv >= 3 ? 1.3 : 1) * g.dmgMult();
          for (const e of g.en.items) if (!e.dead && e.alive && e.y > -20 && quad(e.x, e.y) === i) { g.damageEnemy(e, dmg, { noHitFx: false }); g.fx.explosion(e.x, e.y, 24, { core: "#fff", mid: "#f7f2ff", edge: "#ff2f4f", smoke: "#08060f" }); }
          if (lv >= 5 || i === 3) for (const b of g.eb.items) if (b.alive && quad(b.x, b.y) === i) { b.alive = false; g.dropPickup(b.x, b.y, 1, 1); }
          comicWord(g, centers[i][0], centers[i][1], WORD[i], 120);
        }
      }
      if (nextTick &lt;= 0) {
        nextTick = 0.22;
        for (let i = 0; i &lt; 4; i++) if (t >= T0[i] && t &lt; T0[i] + WIN) for (const e of g.en.items) if (!e.dead && e.alive && e.y > -20 && quad(e.x, e.y) === i) g.damageEnemy(e, 18 * g.dmgMult(), { noText: true });
      }
      if (!spread && t >= 6.9) {
