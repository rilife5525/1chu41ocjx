// 律音：豎琴弧翼——修長白色機身、金色琴臂與粉色翼膜，翼端是發光的音符頭
import { drawGear, flame, fillPoly, lin, mirror, rad, type ShipDrawOpts } from "./shipUtil";
import { halo } from "./shapes";

export function drawCadence(c: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t;
  const tier = o.tier;
  drawGear(c, o, false);
  // 譜線光帶（Tier1+）：自翼端拖曳的五線譜
  if (tier >= 1) {
    c.save();
    c.globalCompositeOperation = "lighter";
    for (const s of [-1, 1]) for (let i = 0; i &lt; 5; i++) {
      c.strokeStyle = `rgba(255,201,74,${0.22 - i * 0.02})`;
      c.lineWidth = 0.9;
      c.beginPath();
      c.moveTo(s * 38, -30 + i * 1.8);
      c.bezierCurveTo(s * (46 + i), -4, s * (40 + i * 2), 34, s * (30 + i * 2), 60 + Math.sin(t * 5 + i) * 3);
      c.stroke();
    }
    c.restore();
  }
  flame(c, -4.5, 34, 20 * (0.5 + o.thrust * 0.8), 3.2, t, "#ffffff", "#ff7ab8");
  flame(c, 4.5, 34, 20 * (0.5 + o.thrust * 0.8), 3.2, t, "#ffffff", "#ffc94a");
  // 琴臂與翼膜
  mirror(c, () => {
    c.beginPath();
    c.moveTo(6, 12);
    c.quadraticCurveTo(32, 6, 38, -32);
    c.quadraticCurveTo(26, -8, 8, -12);
    c.closePath();
    c.fillStyle = lin(c, 6, 12, 38, -32, [[0, "rgba(255,240,248,0.95)"], [1, "rgba(255,122,184,0.8)"]]);
    c.fill();
    c.strokeStyle = "#ffc94a";
    c.lineWidth = 3.2;
    c.lineCap = "round";
    c.beginPath();
    c.moveTo(7, 17);
    c.quadraticCurveTo(36, 15, 39, -34);
    c.stroke();
    c.strokeStyle = "#ff7ab8";
    c.lineWidth = 1.3;
    c.stroke();
    // 琴弦
    c.strokeStyle = "rgba(255,230,160,0.85)";
    c.lineWidth = 0.8;
    for (let i = 0; i &lt; 3; i++) {
      c.beginPath();
      c.moveTo(10 + i * 5, 12 - i * 3);
      c.lineTo(16 + i * 7, -8 - i * 8);
      c.stroke();
    }
    // 翼端音符頭
    c.save();
    c.globalCompositeOperation = "lighter";
    c.translate(39, -34);
    halo(c, "#ff7ab8", 10 + Math.sin(t * 6) * 1.5, 10, 0.9);
    c.restore();
    c.fillStyle = "#fff6e0";
    c.beginPath();
    c.ellipse(39, -34, 3.2, 2.3, -0.4, 0, 6.3);
    c.fill();
  });
  // 機身
  fillPoly(c, [[0, -42], [6, -24], [8, 6], [5, 30], [0, 38], [-5, 30], [-8, 6], [-6, -24]], lin(c, 0, -42, 0, 38, [[0, "#ffffff"], [0.5, "#ffe0ee"], [1, "#e89ac0"]]), "#ffc94a", 1.4);
  c.strokeStyle = "rgba(255,122,184,0.7)";
  c.lineWidth = 1;
  c.beginPath();
  c.moveTo(0, -30);
  c.lineTo(0, 30);
  c.stroke();
  // 座艙
  c.beginPath();
  c.ellipse(0, -10, 4.4, 11, 0, 0, 6.3);
  c.fillStyle = rad(c, -1, -14, 1, 12, [[0, "#ffffff"], [0.5, "#ffb0d8"], [1, "#b03a78"]]);
  c.fill();
  // Tier2：頭冠
  if (tier >= 2) {
    c.fillStyle = "#ffc94a";
    for (const [x, h] of [[-4, 7], [0, 10], [4, 7]]) {
      c.beginPath();
      c.moveTo(x - 2, -38);
      c.lineTo(x, -38 - h);
      c.lineTo(x + 2, -38);
      c.fill();
    }
  }
  // Tier3：音符光環
  if (tier >= 3) {
    c.save();
    c.globalCompositeOperation = "lighter";
    c.strokeStyle = "rgba(255,201,74,0.5)";
    c.lineWidth = 1.2;
    for (let i = 0; i &lt; 3; i++) {
      c.beginPath();
      c.ellipse(0, 4, 50 + i * 5, 16 + i * 2, 0, 0, 6.3);
      c.stroke();
    }
    for (let i = 0; i &lt; 4; i++) {
      const a = t * 1.8 + (i * Math.PI) / 2;
      c.fillStyle = "#fff0c8";
      c.beginPath();
      c.arc(Math.cos(a) * 52, 4 + Math.sin(a) * 17, 2.2, 0, 6.3);
      c.fill();
    }
    c.restore();
  }
  drawGear(c, o, true);
}

// 詰將：將棋駒五角機身——藏青底、金邊、緋紅點綴，肩甲是「桂馬」造型的斜翼
import { drawGear, fillPoly, flame, lin, mirror, type ShipDrawOpts } from "./shipUtil";
import { halo } from "./shapes";

export function drawGambit(c: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t;
  const tier = o.tier;
  drawGear(c, o, false);
  // Tier3：棋盤光格
  if (tier >= 3) {
    c.save();
    c.globalCompositeOperation = "lighter";
    c.strokeStyle = "rgba(255,201,74,0.28)";
    c.lineWidth = 0.8;
    for (let i = -3; i &lt;= 3; i++) {
      c.beginPath(); c.moveTo(i * 14, -52); c.lineTo(i * 14, 52); c.stroke();
      c.beginPath(); c.moveTo(-48, i * 14); c.lineTo(48, i * 14); c.stroke();
    }
    c.restore();
  }
  flame(c, -9, 30, 18 * (0.5 + o.thrust * 0.8), 3.6, t, "#fff", "#ff2d5a");
  flame(c, 9, 30, 18 * (0.5 + o.thrust * 0.8), 3.6, t, "#fff", "#ff2d5a");
  // 桂馬斜翼
  mirror(c, () => {
    fillPoly(c, [[20, -12], [46, 6], [42, 26], [26, 22], [22, 8]], lin(c, 20, -12, 46, 26, [[0, "#3a3068"], [1, "#14102c"]]), "#ffc94a", 1.4);
    fillPoly(c, [[26, 2], [40, 10], [37, 20], [27, 16]], "#ff2d5a");
  });
  // 棋駒本體（五角）
  fillPoly(c, [[0, -44], [22, -18], [29, 32], [-29, 32], [-22, -18]], lin(c, 0, -44, 0, 32, [[0, "#4a3d8a"], [0.5, "#241c4c"], [1, "#100c28"]]), "#ffc94a", 2);
  fillPoly(c, [[0, -36], [16, -15], [21, 24], [-21, 24], [-16, -15]], "rgba(255,201,74,0.1)", "rgba(255,201,74,0.55)", 1);
  // 「王」字
  c.save();
  c.font = '900 24px "Noto Serif TC", serif';
  c.textAlign = "center";
  c.textBaseline = "middle";
  c.shadowColor = "#ffc94a";
  c.shadowBlur = 8;
  c.fillStyle = "#ffe08a";
  c.fillText(tier >= 2 ? "王" : "歩", 0, 6);
  c.restore();
  // 緋紅寶石
  c.save();
  c.globalCompositeOperation = "lighter";
  c.translate(0, -26);
  halo(c, "#ff2d5a", 8, 8, 0.9);
  c.restore();
  c.fillStyle = "#ffd0d8";
  c.beginPath(); c.arc(0, -26, 2.2, 0, 6.3); c.fill();
  // Tier1：金冠尖角
  if (tier >= 1) {
    c.fillStyle = "#ffc94a";
    for (const x of [-8, 0, 8]) { c.beginPath(); c.moveTo(x - 2.5, -40 + Math.abs(x) * 0.3); c.lineTo(x, -50 + Math.abs(x) * 0.5); c.lineTo(x + 2.5, -40 + Math.abs(x) * 0.3); c.fill(); }
  }
  // Tier2：環繞小棋駒
  if (tier >= 2) {
    for (let i = 0; i &lt; 3; i++) {
      const a = t * 1.6 + (i * Math.PI * 2) / 3;
      const x = Math.cos(a) * 40;
      const y = 4 + Math.sin(a) * 14;
      c.save(); c.translate(x, y); c.scale(Math.cos(a) > 0 ? 1 : 0.8, 1);
      fillPoly(c, [[0, -7], [5, -2], [6, 6], [-6, 6], [-5, -2]], "#2a2250", "#ffc94a", 1);
      c.restore();
    }
  }
  drawGear(c, o, true);
}

// 天門：白金華麗機身，雙肩是門環飾翼，背後懸著一圈旋轉的金色天門光輪
import { drawGear, fillPoly, flame, lin, mirror, rad, type ShipDrawOpts } from "./shipUtil";
import { halo } from "./shapes";

export function drawGate(c: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t;
  const tier = o.tier;
  drawGear(c, o, false);
  // 背後天門光輪
  c.save();
  c.globalCompositeOperation = "lighter";
  c.translate(0, -2);
  const rings = 1 + Math.min(2, tier);
  for (let r = 0; r &lt; rings; r++) {
    c.save();
    c.rotate(t * (0.6 - r * 0.25) * (r % 2 ? -1 : 1));
    c.strokeStyle = `rgba(255,210,74,${0.65 - r * 0.15})`;
    c.lineWidth = 1.6;
    const R = 44 + r * 8;
    c.beginPath(); c.arc(0, 0, R, 0, 6.3); c.stroke();
    for (let i = 0; i &lt; 16; i++) {
