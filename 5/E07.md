function glitchOverlay(ctx: CanvasRenderingContext2D, a: number, t: number) {
  ctx.save();
  ctx.globalCompositeOperation = "lighter";
  for (let i = 0; i &lt; 14; i++) {
    const y = ((Math.floor(t * 30) * 37 + i * 71) % H);
    ctx.fillStyle = i % 2 ? hexA(G, 0.25 * a) : hexA(M, 0.25 * a);
    ctx.fillRect(((i * 53 + Math.floor(t * 40)) % 80) - 40, y, W, 4 + (i % 3) * 3);
  }
  ctx.restore();
}

function ultStep(g: Game, k: K, dt: number) {
  const u = k.u, lv = g.lvl("ult"), dm = g.dmgMult();
  for (const b of g.eb.items) if (b.alive && u &lt; 6.5) b.slow = 0.35;
  if (u >= 2 && u &lt; 5) {
    // 為每架敵機開啟駭入視窗
    for (const e of live(g)) if (!k.wins.some((w) => w.uid === e.uid) && k.wins.length &lt; 24) k.wins.push({ e, uid: e.uid, p: 0, boss: e.boss || !!e.def.mid });
    for (const w of k.wins) w.p = Math.min(1, w.p + dt * (lv >= 3 ? 0.55 : 0.36) * (w.boss ? 0.8 : 1) * rand(0.6, 1.4));
  }
  if (u >= 5 && !k.done) {
    k.done = true;
    for (const w of k.wins) {
      if (!w.e.alive || w.e.dead || w.e.uid !== w.uid) continue;
      if (w.boss) { g.damageEnemy(w.e, 250 * dm * (lv >= 5 ? 1.5 : 1)); g.fx.text(w.e.x, w.e.y, "DELETED", M, 26, 1.2); }
      else hack(g, w.e);
    }
    g.later(1.5, () => { for (const e of [...K(g).hackList]) { if (e.alive && !e.dead) { g.aoe(e.x, e.y, 60, 40 * dm, { pal: "jade", small: true }); e.freeze = 0; g.damageEnemy(e, e.hp + 1); } } });
    g.flash(G, 0.5); g.sfx("ultgo");
    bigText(g, "存取完成", "ACCESS GRANTED", G, 1.0, H * 0.35);
  }
  if (u >= 6.5) {
    const passes = lv >= 7 ? 2 : 1;
    const f = clamp((u - 6.5) / (3 / passes), 0, passes);
    const prev = k.band;
    const pf = f % 1, dir = Math.floor(f) % 2 === 0;
    k.band = dir ? pf * (H + 80) - 40 : H + 40 - pf * (H + 80);
    if (prev >= 0) {
      const y0 = Math.min(prev, k.band) - 30, y1 = Math.max(prev, k.band) + 30;
      for (const e of live(g)) if (e.y >= y0 && e.y &lt;= y1) g.damageEnemy(e, 40 * dm, { noText: true });
      for (const b of g.eb.items) if (b.alive && b.y >= y0 && b.y &lt;= y1) { b.alive = false; g.score += 10; }
    }
  }
}

function drawRain(g: Game, ctx: CanvasRenderingContext2D, k: K) {
  const u = k.u, t = g.time;
  const a = Math.min(1, u / 0.5, (9.5 - u) / 0.6);
  ctx.save();
  ctx.font = "14px monospace"; ctx.textAlign = "center";

    const head = ((t * (180 + (c * 37) % 120) + c * 113) % (H + 300)) - 100;
    for (let i = 0; i &lt; 16; i++) {
      const y = head - i * 16;
      if (y &lt; -16 || y > H) continue;
      ctx.fillStyle = i === 0 ? `rgba(220,255,230,${a})` : `rgba(57,255,136,${a * (1 - i / 16) * 0.7})`;
      ctx.fillText(GLYPH[(c * 7 + i * 3 + Math.floor(t * 10)) % GLYPH.length], x, y);
    }

  ctx.fillStyle = `rgba(0,0,0,${0.18 * a})`;
  for (let y = 0; y &lt; H; y += 4) ctx.fillRect(0, y, W, 1);
  ctx.restore();

interface K { fx: number; fy: number; ward: number; wardMax: number; u: number; tick: number; fx2: number; fy2: number; lanceX: number; flare: boolean; ray: number }
const K = (g: Game) => g.p.k as K;

function mirrors(g: Game) {
  const p = g.p, n = g.lvl("weapon") >= 3 ? 6 : 4, k = K(g);
  const out: { x: number; y: number }[] = [];
  const spread = k.ward > 0 ? 2.6 : 1.9;
  const R = k.ward > 0 ? (g.lvl("active") >= 3 ? 95 : 70) : 40;
  for (let i = 0; i &lt; n; i++) {
    const a = -Math.PI / 2 + (i - (n - 1) / 2) * (spread / Math.max(1, n - 1)) + Math.sin(g.time * 2 + i) * 0.03;
    out.push({ x: p.x + Math.cos(a) * R, y: p.y + 6 + Math.sin(a) * R });
  }
  return out;
}
