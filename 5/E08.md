function bigMirrors(k: K) {
  const out: { x: number; y: number }[] = [];
  const u = k.u;
  const drop = easeOut(clamp(u / 1.2, 0, 1));
  const align = clamp((u - 6) / 0.6, 0, 1);
  for (let i = 0; i &lt; 12; i++) {
    const f = i / 11;
    const ax = 30 + f * (W - 60), ay = 70 + Math.sin(f * Math.PI) * -30 + 60;
    const lx = k.lanceX - 50 + (i % 4) * 33, ly = 40 + Math.floor(i / 4) * 30;
    const x = ax + (lx - ax) * easeInOut(align), y = (ay - 200 * (1 - drop)) + (ly - ay) * easeInOut(align);
    out.push({ x, y });
  }
  return out;
}
function ultStep(g: Game, k: K, dt: number) {
  const u = k.u, lv = g.lvl("ult"), dm = g.dmgMult();
  if (u >= 1.2 && u &lt; 6) {
    let best: Enemy | null = null, bs = -1;
    for (const e of live(g)) { let s = e.maxHp; for (const o of live(g)) if (Math.hypot(o.x - e.x, o.y - e.y) &lt; 90) s += 30; if (s > bs) { bs = s; best = e; } }
    const tx = best ? best.x : W / 2 + Math.sin(u) * 150, ty = best ? best.y : H * 0.35;
    const sp = lv >= 3 ? 380 : 260, d = Math.hypot(tx - k.fx2, ty - k.fy2);
    if (d > 1) { const s = Math.min(d, sp * dt); k.fx2 += ((tx - k.fx2) / d) * s; k.fy2 += ((ty - k.fy2) / d) * s; }
    k.tick -= dt;
    if (k.tick &lt;= 0) {
      k.tick = 0.05;
      g.aoe(k.fx2, k.fy2, 70, 18 * dm, { noFx: true, noText: true });
      for (const m of bigMirrors(k)) lineHit(g, m.x, m.y, k.fx2, k.fy2, 10, 3 * dm);
      g.clearBullets(k.fx2, k.fy2, 80, "score");
    }
  }
  if (u >= 6 && u &lt; 8.5) {
    const f = clamp((u - 6.4) / 2.0, 0, 1);
    const prev = k.lanceX;
    k.lanceX = -80 + easeInOut(f) * (W + 160);
    const w = lv >= 5 ? 170 : 120;
    k.tick -= dt;
    if (k.tick &lt;= 0 && u > 6.4) {
      k.tick = 0.05;
      lineHit(g, k.lanceX, 0, k.lanceX, H, w, 40 * dm);
      lineClear(g, k.lanceX, 0, k.lanceX, H, w);
    }
    void prev;
  }
  if (u >= 8.5 && !k.flare) {
    k.flare = true;
    g.aoe(W / 2, H / 2, 1500, 200 * dm, { pal: "gold" });
    if (lv >= 7) g.clearBullets(W / 2, H / 2, 3000, "score");
    g.flash("#ffffff", 1); g.shake(18, 0.7); g.sfx("ultgo");
    bigText(g, "正午審判", "HIGH NOON", O, 1.0, H * 0.4);
  }
}
function drawUlt(g: Game, ctx: CanvasRenderingContext2D, k: K) {
  const u = k.u, t = g.time, lv = g.lvl("ult");
  const ms = bigMirrors(k);
  ctx.save();
  ctx.globalCompositeOperation = "lighter";
  if (u >= 1.2 && u &lt; 6.1) {
    for (const m of ms) {
      ctx.strokeStyle = hexA(Y, 0.3); ctx.lineWidth = 12;
      ctx.beginPath(); ctx.moveTo(m.x, m.y); ctx.lineTo(k.fx2, k.fy2); ctx.stroke();
      ctx.strokeStyle = hexA("#ffffff", 0.7); ctx.lineWidth = 2; ctx.stroke();
    }
    glowAt(ctx, k.fx2, k.fy2, 150, O, 0.6); glowAt(ctx, k.fx2, k.fy2, 70, "#ffffff", 0.9);
    for (let i = 0; i &lt; 3; i++) { ctx.strokeStyle = hexA(Y, 0.5); ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(k.fx2, k.fy2, 74 + i * 14 + Math.sin(t * 15 + i) * 4, 0, TAU); ctx.stroke(); }
  }
  if (u >= 6.3 && u &lt; 8.6) {
    const w = lv >= 5 ? 170 : 120;
    const gr = ctx.createLinearGradient(k.lanceX - w, 0, k.lanceX + w, 0);
    gr.addColorStop(0, "rgba(255,157,59,0)"); gr.addColorStop(0.35, "rgba(255,227,107,0.6)"); gr.addColorStop(0.5, "rgba(255,255,255,0.95)"); gr.addColorStop(0.65, "rgba(255,227,107,0.6)"); gr.addColorStop(1, "rgba(255,157,59,0)");
    ctx.fillStyle = gr; ctx.fillRect(k.lanceX - w, 0, w * 2, H);
  }
  if (u >= 8.5) {
    const f = clamp((u - 8.5) / 1, 0, 1);
    glowAt(ctx, W / 2, H * 0.3, 500 * (1 + f), "#ffffff", 1 - f);
    for (let i = 1; i &lt;= 5; i++) glowAt(ctx, W / 2 + (W / 2 - 100) * (i / 5) * (1 - 2 * 0.5), H * 0.3 + i * 90, 20 + i * 10, i % 2 ? Y : O, (1 - f) * 0.6);
  }
  ctx.restore();
  // 巨鏡
  for (const m of ms) {
    ctx.save(); ctx.translate(m.x, m.y);
    ctx.rotate(Math.atan2(k.fy2 - m.y, k.fx2 - m.x) + Math.PI / 2);
    ctx.fillStyle = "#3a2a10"; ctx.beginPath(); for (let i = 0; i &lt; 6; i++) { const b = (i / 6) * TAU; ctx.lineTo(Math.cos(b) * 20, Math.sin(b) * 11); } ctx.closePath(); ctx.fill();
    const gr = ctx.createLinearGradient(-20, 0, 20, 0); gr.addColorStop(0, "#ffffff"); gr.addColorStop(0.5, "#cfe8ff"); gr.addColorStop(1, Y);

    ctx.restore();
  }
  void rand;
}
