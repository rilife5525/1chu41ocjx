function halftone(ctx: CanvasRenderingContext2D, x0: number, y0: number, x1: number, y1: number, col: string, step = 5) {
  ctx.fillStyle = col;
  for (let y = y0; y &lt; y1; y += step) for (let x = x0 + (((y - y0) / step) % 2 ? step / 2 : 0); x &lt; x1; x += step) {
    const k = 1.5 - ((y - y0) / (y1 - y0)) * 1.2;
    ctx.beginPath(); ctx.arc(x, y, Math.max(0.3, k), 0, 7); ctx.fill();
  }
}
