// 零位 GLITCH：銳角隱形三角翼、RGB 色散、像素錯位切片、全息六角環
import { type ShipDrawOpts, mirror, fillPoly, lin, flame, drawGear, path, type Pt } from "./shipUtil";
import { glow, hexA } from "./sprites";

const G = "#39ff88", M = "#ff3df0";
const HULL: Pt[] = mirror([[0, -44], [5, -26], [10, -10], [38, 18], [36, 24], [14, 18], [8, 28], [0, 22]]);

export function drawGlitch(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t;
  drawGear(ctx, o, false);
  // 全息六角環
  ctx.save(); ctx.globalCompositeOperation = "lighter";
  ctx.strokeStyle = hexA(G, 0.35); ctx.lineWidth = 1;
  ctx.rotate(t * 0.6);
  ctx.beginPath(); for (let i = 0; i &lt; 6; i++) { const a = (i / 6) * Math.PI * 2; ctx.lineTo(Math.cos(a) * 40, Math.sin(a) * 40); } ctx.closePath(); ctx.stroke();
  ctx.restore();
  flame(ctx, -6, 24, 4, 16 + o.thrust * 16, t, "#e0ffe8", G);
  flame(ctx, 6, 24, 4, 16 + o.thrust * 16, t + 0.2, "#e0ffe8", M);
  ctx.save();
  ctx.scale(1 - Math.abs(o.bank) * 0.14, 1);
  // RGB 色散殘影
  const j = Math.sin(t * 13) > 0.85 ? 4 : 1.5;
  ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.5;
  ctx.translate(-j, 0); path(ctx, HULL); ctx.fillStyle = hexA(M, 0.6); ctx.fill();
  ctx.translate(j * 2, 0); path(ctx, HULL); ctx.fillStyle = hexA(G, 0.6); ctx.fill();
  ctx.restore();
  fillPoly(ctx, HULL, lin(ctx, 0, -44, 0, 28, [[0, "#d0ffe0"], [0.2, "#1f5a3a"], [0.6, "#08140e"], [1, "#030806"]]), G, 1.2);
  // 電路刻線
  ctx.strokeStyle = hexA(G, 0.8); ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(0, -36); ctx.lineTo(0, -12); ctx.lineTo(10, -2); ctx.lineTo(26, 14); ctx.moveTo(0, -12); ctx.lineTo(-10, -2); ctx.lineTo(-26, 14); ctx.stroke();
  for (const [x, y] of [[26, 14], [-26, 14], [10, -2], [-10, -2]]) { ctx.fillStyle = M; ctx.fillRect(x - 1.5, y - 1.5, 3, 3); }
  // 天線刺
  ctx.strokeStyle = hexA("#ffffff", 0.8); ctx.beginPath(); ctx.moveTo(4, -24); ctx.lineTo(8, -40); ctx.moveTo(-4, -24); ctx.lineTo(-8, -40); ctx.stroke();
  // 座艙（像素）
  for (let i = 0; i &lt; 3; i++) for (let k = 0; k &lt; 2; k++) { ctx.fillStyle = (i + k + Math.floor(t * 6)) % 3 ? G : "#ffffff"; ctx.fillRect(-3 + k * 3, -22 + i * 3, 3, 3); }
  if (o.tier >= 1) fillPoly(ctx, [[36, 24], [42, 30], [34, 28]], M);
  if (o.tier >= 1) fillPoly(ctx, [[-36, 24], [-42, 30], [-34, 28]], M);
  if (o.tier >= 2) { ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.drawImage(glow(G), -24, -8, 48, 48); ctx.restore(); }
  ctx.restore();
  // 像素錯位切片
  const seed = Math.floor(t * 8);
  if ((seed * 7919) % 5 === 0 || o.tier >= 3) {
    ctx.save(); ctx.globalCompositeOperation = "lighter";
    for (let i = 0; i &lt; 3; i++) {
      const y = ((seed * 31 + i * 17) % 60) - 36;
      ctx.fillStyle = i % 2 ? hexA(M, 0.5) : hexA(G, 0.5);
      ctx.fillRect(-30 + ((seed + i * 13) % 12) - 6, y, 60, 2);
    }
    ctx.restore();
  }
  drawGear(ctx, o, true);
}

// 日鏡 HELIOSTAT：日晷指針機身、放射日芒尾翼、太陽圓盤紋章（鏡片由技能層繪製）
import { type ShipDrawOpts, mirror, fillPoly, lin, rad, flame, drawGear } from "./shipUtil";
import { glow, hexA } from "./sprites";

export function drawHeliostat(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t, Y = "#ffe36b", O = "#ff9d3b", D = "#2a1a04";
  drawGear(ctx, o, false);
  // 日芒
  ctx.save(); ctx.globalCompositeOperation = "lighter";
  ctx.translate(0, 14); ctx.rotate(t * 0.4);
  const n = o.tier >= 2 ? 16 : 12;
  for (let i = 0; i &lt; n; i++) {
    const a = (i / n) * Math.PI * 2, L = i % 2 ? 26 : 36;
    ctx.fillStyle = hexA(i % 2 ? O : Y, 0.4);
    ctx.beginPath(); ctx.moveTo(Math.cos(a - 0.08) * 14, Math.sin(a - 0.08) * 14); ctx.lineTo(Math.cos(a) * L, Math.sin(a) * L); ctx.lineTo(Math.cos(a + 0.08) * 14, Math.sin(a + 0.08) * 14); ctx.fill();
  }
  ctx.restore();
  flame(ctx, 0, 30, 6, 18 + o.thrust * 18, t, "#ffffff", O);
  ctx.save();
  ctx.scale(1 - Math.abs(o.bank) * 0.14, 1);
  // 側翼（細長三角）
  for (const s of [-1, 1]) {
    ctx.save(); ctx.scale(s, 1);
    fillPoly(ctx, [[6, -4], [34, 14], [32, 20], [8, 16]], lin(ctx, 6, -4, 34, 20, [[0, "#fff4c8"], [0.4, "#c08a20"], [1, D]]), Y, 1);
    fillPoly(ctx, [[10, 8], [26, 16], [10, 14]], hexA("#ffffff", 0.6));
    if (o.tier >= 1) fillPoly(ctx, [[34, 14], [42, 12], [36, 20]], Y);
    ctx.restore();
  }
  // 日晷指針機身
  const body = mirror([[0, -48], [3, -30], [7, -6], [9, 16], [6, 30], [0, 32]]);
  fillPoly(ctx, body, lin(ctx, 0, -48, 0, 32, [[0, "#ffffff"], [0.2, Y], [0.6, "#b0761a"], [1, D]]), "#fff4c8", 1.2);
  // 太陽圓盤
  ctx.fillStyle = rad(ctx, 0, 12, 10, [[0, "#ffffff"], [0.5, Y], [1, O]]);
  ctx.beginPath(); ctx.arc(0, 12, 8, 0, Math.PI * 2); ctx.fill();
  ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.6 + Math.sin(t * 5) * 0.3; ctx.drawImage(glow(Y), -18, -6, 36, 36); ctx.restore();
  // 座艙
  ctx.fillStyle = rad(ctx, 0, -16, 6, [[0, "#ffffff"], [1, "#ffcf6b"]]);
  ctx.beginPath(); ctx.ellipse(0, -16, 3, 7, 0, 0, Math.PI * 2); ctx.fill();
  ctx.restore();
  if (o.tier >= 3) { ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.strokeStyle = hexA(Y, 0.6); ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(0, 12, 42, 0, Math.PI * 2); ctx.stroke(); ctx.restore(); }
  drawGear(ctx, o, true);
}

// 星錘 MORNINGSTAR：緊湊裝甲機身、中央絞盤齒環、勾爪翼、萊姆能量線（鏈錘由技能層繪製）
import { type ShipDrawOpts, mirror, fillPoly, lin, rad, flame, drawGear } from "./shipUtil";
import { glow, hexA } from "./sprites";

export function drawMorningstar(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t, L = "#b6ff4a", S = "#d8e2ff", D = "#141a10";
  drawGear(ctx, o, false);
  flame(ctx, -12, 26, 6, 18 + o.thrust * 20, t, "#f4ffe0", L);
  flame(ctx, 12, 26, 6, 18 + o.thrust * 20, t + 0.3, "#f4ffe0", L);
  ctx.save();
  ctx.scale(1 - Math.abs(o.bank) * 0.12, 1);
  // 勾爪翼
  for (const s of [-1, 1]) {
    ctx.save(); ctx.scale(s, 1);
    ctx.beginPath();
    ctx.moveTo(12, -10);
    ctx.bezierCurveTo(30, -12, 40, 0, 38, 16);
    ctx.lineTo(42, 24); ctx.lineTo(34, 22);
    ctx.bezierCurveTo(30, 12, 22, 10, 14, 16);
    ctx.closePath();
    ctx.fillStyle = lin(ctx, 12, -12, 40, 24, [[0, "#6a7488"], [0.5, "#3a4052"], [1, D]]); ctx.fill();
    ctx.strokeStyle = S; ctx.lineWidth = 1; ctx.stroke();
    ctx.strokeStyle = L; ctx.lineWidth = 1.4; ctx.beginPath(); ctx.moveTo(16, 4); ctx.quadraticCurveTo(28, 2, 34, 14); ctx.stroke();
    if (o.tier >= 1) fillPoly(ctx, [[38, 16], [46, 12], [42, 24]], L);
    if (o.tier >= 2) fillPoly(ctx, [[14, -4], [22, -18], [20, -6]], hexA(L, 0.9));
    ctx.restore();
  }
  // 機身
  const body = mirror([[0, -38], [8, -30], [14, -12], [14, 12], [10, 26], [0, 30]]);
  fillPoly(ctx, body, lin(ctx, 0, -38, 0, 30, [[0, "#eef3ff"], [0.3, "#8a94aa"], [0.75, "#3a4052"], [1, D]]), S, 1.2);
  // 裝甲接縫
  ctx.strokeStyle = hexA(D, 0.8); ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(-12, -6); ctx.lineTo(12, -6); ctx.moveTo(-12, 18); ctx.lineTo(12, 18); ctx.stroke();
  // 絞盤齒環
  ctx.save(); ctx.translate(0, 6); ctx.rotate(t * 3);
  ctx.fillStyle = "#2a3020"; ctx.beginPath(); ctx.arc(0, 0, 9, 0, Math.PI * 2); ctx.fill();
  ctx.strokeStyle = L; ctx.lineWidth = 2;
  for (let i = 0; i &lt; 8; i++) { const a = (i / 8) * Math.PI * 2; ctx.beginPath(); ctx.moveTo(Math.cos(a) * 7, Math.sin(a) * 7); ctx.lineTo(Math.cos(a) * 11, Math.sin(a) * 11); ctx.stroke(); }
  ctx.restore();
  ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.globalAlpha = 0.7 + Math.sin(t * 6) * 0.3; ctx.drawImage(glow(L), -10, -4, 20, 20); ctx.restore();
  // 座艙
  ctx.fillStyle = rad(ctx, 0, -22, 7, [[0, "#ffffff"], [0.5, "#e6ffc0"], [1, L]]);
  ctx.beginPath(); ctx.ellipse(0, -20, 4.5, 7, 0, 0, Math.PI * 2); ctx.fill();
  ctx.restore();
  if (o.tier >= 3) { ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.strokeStyle = hexA(L, 0.6); ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(0, 6, 22, -t * 2, -t * 2 + Math.PI * 1.2); ctx.stroke(); ctx.restore(); }
  drawGear(ctx, o, true);
}
