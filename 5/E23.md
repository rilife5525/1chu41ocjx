/** 放射光芒 */
export function lightRays(ctx: CanvasRenderingContext2D, x: number, y: number, n: number, len: number, rot: number, color: string, a: number) {
  ctx.save();
  ctx.globalCompositeOperation = "lighter";
  ctx.translate(x, y);
  ctx.rotate(rot);
  ctx.fillStyle = color;
  ctx.globalAlpha = a;
  for (let i = 0; i &lt; n; i++) {
    ctx.rotate((Math.PI * 2) / n);
    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.lineTo(len, -len * 0.035);
    ctx.lineTo(len, len * 0.035);
    ctx.closePath();
    ctx.fill();
  }
  ctx.restore();
}
/** 粗體描邊大字 */
export function bigText(ctx: CanvasRenderingContext2D, str: string, x: number, y: number, size: number, color: string, a = 1, stroke = "#12081e", rot = 0) {
  ctx.save();
  ctx.globalAlpha = a;
  ctx.translate(x, y);
  ctx.rotate(rot);
  ctx.font = `900 ${size}px "Noto Serif TC","Noto Sans TC","Microsoft JhengHei",serif`;
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.lineJoin = "round";
  ctx.lineWidth = size * 0.16;
  ctx.strokeStyle = stroke;
  ctx.strokeText(str, 0, 0);
  ctx.fillStyle = color;
  ctx.fillText(str, 0, 0);
  ctx.restore();
}

// 詰將：棋路布陣
// 一句話機制：飛駒以「一步一頓」的棋路前進，被標記的敵機擊墜化為持駒，再從戰場任一處打回；主動技布下飛車與角行，奧義把戰場化為將棋盤。

type PKind = "rook" | "bishop" | "knight" | "dragon";

const GLYPH: Record&lt;string, string> = { rook: "飛", bishop: "角", knight: "桂", dragon: "龍", lance: "香", gold: "金", silver: "銀", king: "王" };

/** 畫一枚將棋駒（五角木牌） */
function tile(c: CanvasRenderingContext2D, x: number, y: number, s: number, ch: string, col = "#ffc94a", fill = "#2a2347", a = 1) {
  c.save();
  c.translate(x, y);
  c.globalAlpha = a;
  c.shadowColor = col;
  c.shadowBlur = 14;
  c.fillStyle = fill;
  c.strokeStyle = col;
  c.lineWidth = Math.max(2, s * 0.06);
  c.beginPath();
  c.moveTo(0, -s * 0.62);
  c.lineTo(s * 0.42, -s * 0.28);
  c.lineTo(s * 0.52, s * 0.55);
  c.lineTo(-s * 0.52, s * 0.55);
  c.lineTo(-s * 0.42, -s * 0.28);
  c.closePath();
  c.fill();
  c.stroke();
  c.shadowBlur = 0;
  c.fillStyle = col;
  c.font = `900 ${Math.round(s * 0.62)}px "Noto Serif TC", serif`;
  c.textAlign = "center";
  c.textBaseline = "middle";
  c.fillText(ch, 0, s * 0.14);
  c.restore();
}

/** 沿線段傷害敵機並消除敵彈 */
function lineHit(g: Game, x1: number, y1: number, x2: number, y2: number, w: number, dmg: number) {
  for (const e of enemyList(g)) if (segDist(e.x, e.y, x1, y1, x2, y2) &lt; e.r + w / 2) g.damageEnemy(e, dmg, { noHitFx: false });
  for (let i = g.eb.items.length - 1; i >= 0; i--) {
    const b = g.eb.items[i];
    if (segDist(b.x, b.y, x1, y1, x2, y2) &lt; w / 2 + 6) g.eb.kill(i);
  }
}
const L = 1400;
function pieceFire(g: Game, pc: Piece, thick: number, dm: number, lines: Line[]) {
  const dmg = 24 * dm * (1 + g.lvl("active") * 0.08);
  const add = (x1: number, y1: number, x2: number, y2: number, col: string) => {
    lines.push({ x1, y1, x2, y2, t: 0, w: thick, col });
    lineHit(g, x1, y1, x2, y2, thick, dmg);
  };
  if (pc.kind === "rook" || pc.kind === "dragon") {
    add(pc.x, -50, pc.x, H + 50, "#ffe08a");
    add(-50, pc.y, W + 50, pc.y, "#ffe08a");
  }
  if (pc.kind === "bishop" || pc.kind === "dragon") {
    add(pc.x - L, pc.y - L, pc.x + L, pc.y + L, "#ff6a86");
    add(pc.x - L, pc.y + L, pc.x + L, pc.y - L, "#ff6a86");
  }
  g.fx.ring(pc.x, pc.y, 6, 54, 0.4, "#ffc94a", 3);
  g.sfx("laser", 0.12);
}

/** 持駒打入：桂馬 L 形跳殺（Lv7 升格龍王，附十字貫穿） */
function dropMove(g: Game, dragon: boolean) {
  const tg: Enemy | null = strongest(g);
  const c = enemyCenter(g);
  const tx = tg ? tg.x : c.x;
  const ty = tg ? tg.y : c.y;
  const sx = clamp(tx + (Math.random() &lt; 0.5 ? -1 : 1) * 90, 40, W - 40);
  const sy = Math.max(40, ty - 190);
  let u = 0;
  const lv = g.lvl("passive");
  const dm = g.dmgMult();
  g.sfx("lock");
  const fx = { alive: true, layer: 1 as const, update(dt: number) { u += dt * 2.6; if (u >= 1) { fx.alive = false; land(); } }, draw(ctx: CanvasRenderingContext2D) {
    const q = clamp(u, 0, 1);
    const x = q &lt; 0.5 ? sx + (tx - sx) * (q / 0.5) : tx;
    const y = q &lt; 0.5 ? sy : sy + (ty - sy) * ((q - 0.5) / 0.5);
    tile(ctx, x, y - Math.sin(q * Math.PI) * 24, 46, dragon ? "龍" : "桂", dragon ? "#ff6a86" : "#ffc94a");
  } };
  g.addFx(fx);
  function land() {
    g.aoe(tx, ty, 82 + lv * 8, 72 * dm * (1 + lv * 0.1), { pal: "gold" });
    g.fx.ring(tx, ty, 10, 110, 0.5, "#ffc94a", 5);
    g.shake(6, 0.2);
    if (lv >= 5) g.heal(3);
    if (dragon) {
      lineHit(g, tx, -50, tx, H + 50, 36, 50 * dm);
      lineHit(g, -50, ty, W + 50, ty, 36, 50 * dm);
      g.fx.ring(tx, ty, 10, 260, 0.6, "#ff6a86", 6);
    }
  }
}

export const gambitKit: ShipKit = {
  init(g) {
    g.p.k = { hand: 0, pieces: [] as Piece[], dur: 0, t: 0, thick: 26, lines: [] as Line[], n: 0 };
  },
  update(g, dt) {
    const k = g.p.k;
    const lines: Line[] = k.lines;
    for (let i = lines.length - 1; i >= 0; i--) { lines[i].t += dt; if (lines[i].t > 0.32) lines.splice(i, 1); }
    if (k.dur &lt;= 0) return;
    k.t += dt;
    const dm = g.dmgMult();
    const lv = g.lvl("active");
    for (const pc of k.pieces as Piece[]) {
      if (pc.kind === "knight") {
        pc.jump -= dt;
        if (pc.jump &lt;= 0) {
          pc.jump = 1.5;
          const tg = strongest(g);
          if (tg) {
            g.fx.ring(pc.x, pc.y, 6, 40, 0.3, "#ffc94a", 3);
            pc.x = clamp(tg.x + 70, 40, W - 40);
            pc.y = clamp(tg.y - 110, 80, H - 200);
            g.aoe(tg.x, tg.y, 80, 58 * dm, { pal: "gold", small: true });
            g.fx.ring(tg.x, tg.y, 8, 90, 0.4, "#ffc94a", 4);
            g.sfx("lock");
          }
        }
      } else {
        pc.fire -= dt;
        if (pc.fire &lt;= 0) {
          pc.fire = 0.9;
          pieceFire(g, pc, k.thick, dm * (lv >= 7 ? 1.4 : 1), lines);
        }
      }
    }
    if (k.t >= k.dur) { k.dur = 0; k.pieces.length = 0; }
  },
  fire(g) {
    const p = g.p;
    const k = p.k;
    const lv = g.lvl("weapon");
    const dm = g.dmgMult();
    p.fireT += fireGap(g, 0.42);
    k.n++;
    const offs = lv >= 3 ? [-24, 0, 24] : [-15, 15];
    const step = lv >= 5 ? 1500 : 1100;
    for (const ox of offs) g.spawnPB(p.x + ox, p.y - 26, -Math.PI / 2, 0, B + (lv >= 5 ? 4 : 3), 15 * dm * (lv >= 5 ? 1.3 : 1), { kind: 21, a: step, selfMove: 1, pierce: 1, life: 2.1, r: 13 });
    g.sfx("shoot3", 0.08);
  },
  updateBullet(g, b: PBullet, dt) {
    if (b.kind !== 21) return;
    const cyc = b.t % 0.26;
    if (cyc &lt; 0.09) b.y -= b.a * dt;
    // Lv7：行滿三步，「成」為貫穿巨駒
    if (g.lvl("weapon") >= 7 && b.c === 0 && b.t > 0.78) {
      b.c = 1;
      b.style = B + 4;
      b.pierce = 3;
      b.dmg *= 1.6;
      b.r = 18;
      b.scale = 1.15;
      g.fx.ring(b.x, b.y, 4, 26, 0.3, "#ff6a86", 3);
    }
  },
  onHit(g, b, e) {
    if (b.kind === 21 && e.mark === 0) { e.mark = 1; g.fx.text(e.x, e.y - e.r - 10, "王手", "#ff6a86", 13, 0.5); }
  },
  onKill(g, e) {
    const k = g.p.k;
    if (e.mark > 0) {
      k.hand++;
      g.fx.text(e.x, e.y, "持駒", "#ffc94a", 14, 0.7);
      const lv = g.lvl("passive");
      const need = lv >= 3 ? 2 : 3;
      if (k.hand >= need) {
        k.hand -= need;
        dropMove(g, lv >= 7);
      }
    }
  },
  castActive(g) {
    const p = g.p;
    const k = p.k;
    const lv = g.lvl("active");
    const y0 = Math.max(260, p.y - 230);
    let rx = p.x;
    let bx = W - p.x;
    if (Math.abs(rx - bx) &lt; 110) { rx = clamp(p.x - 110, 50, W - 50); bx = clamp(p.x + 110, 50, W - 50); }
    k.pieces = [{ kind: "rook", x: rx, y: y0, fire: 0.1, jump: 0 }, { kind: "bishop", x: bx, y: y0, fire: 0.5, jump: 0 }] as Piece[];
    if (lv >= 3) k.pieces.push({ kind: "knight", x: W / 2, y: y0 + 90, fire: 0, jump: 0.6 });
    if (lv >= 7) k.pieces.push({ kind: "dragon", x: W / 2, y: 210, fire: 0.3, jump: 0 });
    k.dur = 7 + (lv >= 5 ? 3 : 0);
    k.t = 0;
    k.thick = lv >= 7 ? 40 : 26;
    g.toast("布陣：飛車・角行", "#ffc94a");
