function ultStep(g: Game, k: K, dt: number) {
  const u = k.u, p = g.p, lv = g.lvl("ult"), dm = g.dmgMult();
  if (u &lt; 1.2) {
    const f = easeInOut(u / 1.2);
    const tt = twinTarget(g);
    k.tx = tt.x + (p.x - tt.x) * f; k.ty = tt.y + (p.y - tt.y) * f;
    if (u + dt >= 1.2) { k.fused = true; p.form = 1; g.flash("#ffffff", 0.7); g.sfx("morph"); g.shake(10, 0.4); }
  } else if (u &lt; 7.5) {
    k.fused = true; p.form = 1;
    k.tick -= dt;
    if (k.tick &lt;= 0) {
      k.tick = 0.08;
      const w = lv >= 3 ? 100 : 70;
      lineHit(g, p.x, p.y - 30, p.x, -20, w, 14 * dm);
      g.clearBullets(0, 0, 0, "score", p.x, w, p.y / 2, p.y);
    }
    if (lv >= 5) { k.novaT -= dt; if (k.novaT &lt;= 0) { k.novaT = 1.5; g.aoe(p.x, p.y, 160, 60 * dm, { pal: "white" }); g.fx.ring(p.x, p.y, 20, 180, 0.5, P, 5); g.fx.ring(p.x, p.y, 10, 160, 0.5, B, 3); } }
  } else {
    if (k.fused) {
      k.fused = false; p.form = 0;
      k.tx = p.x; k.ty = p.y;
      g.sfx("morph");
    }
    const tt = twinTarget(g);
    k.tx += (tt.x - k.tx) * Math.min(1, dt * 8); k.ty += (tt.y - k.ty) * Math.min(1, dt * 8);
    if (u > 8.1 && !k.final) {
      k.final = true;
      const mul = lv >= 7 ? 2 : 1;
      const ax = p.x, ay = p.y, bx = k.tx, by = k.ty;
      const ex1 = bx + (bx - ax) * 0.2, ey1 = -40, ex2 = ax + (ax - bx) * 0.2, ey2 = -40;
      lineHit(g, ax, ay, ex1, ey1, 60, 150 * dm * mul); lineHit(g, bx, by, ex2, ey2, 60, 150 * dm * mul);
      lineClear(g, ax, ay, ex1, ey1, 60); lineClear(g, bx, by, ex2, ey2, 60);
      (k as unknown as { xb: number[] }).xb = [ax, ay, ex1, ey1, bx, by, ex2, ey2, g.time];
      g.flash("#ffffff", 0.8); g.shake(16, 0.6); g.sfx("ultgo");
      bigText(g, "雙星．交錯", "X-CROSS", B, 1.0, H * 0.35);
    }
  }
}
function drawUlt(g: Game, ctx: CanvasRenderingContext2D, k: K) {
  const u = k.u, p = g.p, t = g.time, lv = g.lvl("ult");
  ctx.save();
  ctx.globalCompositeOperation = "lighter";
  if (u &lt; 1.3) {
    // 螺旋交纏軌跡
    for (let i = 0; i &lt; 30; i++) {
      const f = i / 30, a = t * 10 + f * 10;
      const cx = p.x + (k.tx - p.x) * 0.5, cy = p.y - f * 160;
      glowAt(ctx, cx + Math.cos(a) * 40 * (1 - u / 1.3), cy, 6, P, 1 - f);
      glowAt(ctx, cx - Math.cos(a) * 40 * (1 - u / 1.3), cy, 6, B, 1 - f);
    }
  }
  if (k.fused) {
    const w = lv >= 3 ? 100 : 70;
    for (const [col, ph] of [[P, 0], [B, Math.PI]] as [string, number][]) {
      ctx.strokeStyle = hexA(col, 0.85); ctx.lineWidth = 10;
      ctx.beginPath();
      for (let y = p.y - 30; y > -20; y -= 8) { const x = p.x + Math.sin((p.y - y) * 0.03 - t * 14 + ph) * w * 0.4; y === p.y - 30 ? ctx.moveTo(x, y) : ctx.lineTo(x, y); }
      ctx.stroke();
      ctx.strokeStyle = "rgba(255,255,255,0.9)"; ctx.lineWidth = 3; ctx.stroke();
    }
    const gr = ctx.createLinearGradient(p.x - w / 2, 0, p.x + w / 2, 0);
    gr.addColorStop(0, "rgba(255,122,217,0)"); gr.addColorStop(0.5, "rgba(255,255,255,0.25)"); gr.addColorStop(1, "rgba(90,216,255,0)");
    ctx.fillStyle = gr; ctx.fillRect(p.x - w / 2, 0, w, p.y - 30);
    glowAt(ctx, p.x, p.y - 30, 50, "#ffffff", 0.8);
  }
  const xb = (k as unknown as { xb?: number[] }).xb;
  if (xb && t - xb[8] &lt; 0.7) {
    const f = (t - xb[8]) / 0.7;
    for (const [i, c] of [[0, P], [4, B]] as [number, string][]) {
      ctx.strokeStyle = hexA(c, 0.8 * (1 - f)); ctx.lineWidth = 60 * (1 - f * 0.5); ctx.lineCap = "round";
      ctx.beginPath(); ctx.moveTo(xb[i], xb[i + 1]); ctx.lineTo(xb[i + 2], xb[i + 3]); ctx.stroke();
      ctx.strokeStyle = hexA("#ffffff", 1 - f); ctx.lineWidth = 12; ctx.stroke();
    }
  }
  ctx.restore();
}

function addInf(g: Game, e: Enemy, v: number) {
  const pl = g.lvl("passive");

}
