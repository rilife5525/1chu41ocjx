const Y = "#ffe500", M = "#ff2f8e", C = "#19c6ff", K = "#111111";
const WORDS = [
  { ch: "砰", col: M }, { ch: "咻", col: C }, { ch: "轟", col: Y }, { ch: "啪", col: "#ff8a3d" }, { ch: "嘩", col: "#b07cff" }, { ch: "哈", col: "#ffffff" },
];
interface Bub { t: number; dur: number; cap: number }
interface Slam { x: number; y: number; t: number }
interface K { i: number; mile: number; prevCombo: number; panel: number; bub: Bub | null; u: number; tick: number; slams: Slam[]; stamp: boolean; shattered: boolean }
const KK = (g: Game) => g.p.k as K;

function scale(g: Game) { return 1 + Math.min(0.4, g.combo * 0.004); }
function word(g: Game, x: number, y: number, a: number, w: number, spd: number, dmg: number, o: Partial&lt;PBullet> = {}) {
  g.spawnPB(x, y, a, spd, -1, dmg, { kind: 20, a: w, b: scale(g), c: w === 2 ? 0.55 : 0, r: 10, rot: rand(-0.3, 0.3), ...o });
}
function boom(g: Game, x: number, y: number) {
  const r = g.lvl("weapon") >= 5 ? 100 : 70;
  g.aoe(x, y, r, 30 * g.dmgMult() * scale(g), { pal: "gold" });
  g.fx.ring(x, y, 10, r, 0.35, K, 6);
  g.fx.text(x, y, "轟！", Y, 28, 0.5);
  g.shake(4, 0.12);
}

export const comicKit: ShipKit = {
  init(g) { g.p.k = { i: 0, mile: 0, prevCombo: 0, panel: 0, bub: null, u: -1, tick: 0, slams: [], stamp: false, shattered: false } as K; },
  update(g, dt) {
    const k = KK(g), p = g.p, pl = g.lvl("passive"), dm = g.dmgMult();
    // 集中線：連擊里程碑觸發分格閃擊
    const step = pl >= 3 ? 20 : 25;
    const mile = Math.floor(g.combo / step);
    if (mile > k.mile && g.combo > 0) {
      k.panel = 0.55;
      for (const e of live(g)) if (e.y > 0 && e.y &lt; H) g.damageEnemy(e, 30 * dm);
      if (pl >= 5) g.clearBullets(W / 2, H / 2, 3000, "score");
      g.shake(6, 0.2); g.sfx("boom_m", 0.2);
    }
    k.mile = mile;
    if (pl >= 7 && k.prevCombo > 10 && g.combo === 0) { g.combo = Math.floor(k.prevCombo / 2); g.comboT = 1.5; }
    k.prevCombo = g.combo;
    if (k.panel > 0) k.panel -= dt;
    // 對話框
    const b = k.bub;
    if (b) {
      b.t += dt;
      const al = g.lvl("active");
      const bw = al >= 3 ? 280 : 220, bh = al >= 3 ? 150 : 120, bx = p.x, by = p.y - 40 - bh / 2 - 30;
      for (const e of g.eb.items) if (e.alive && Math.abs(e.x - bx) &lt; bw / 2 && Math.abs(e.y - by) &lt; bh / 2) { e.alive = false; b.cap++; }
      if (b.t >= b.dur) {
        const n = Math.max(12, Math.min(60, b.cap)) * (al >= 5 ? 2 : 1);
        for (let i = 0; i &lt; n; i++) word(g, bx + rand(-bw / 3, bw / 3), by + rand(-bh / 3, bh / 3), -Math.PI / 2 + rand(-1.0, 1.0), Math.floor(rand(0, 6)), rand(480, 680), 10 * dm, { src: 2 });
        for (const pb of g.pb.items) if (pb.alive && pb.kind === 20 && pb.src === 2 && pb.a === 0) pb.a = 5;
        if (al >= 7) for (const e of live(g)) if (Math.hypot(e.x - bx, e.y - by) &lt; 200 && !e.boss) e.freeze = Math.max(e.freeze, 1.2);
        g.fx.text(bx, by, "哈！！", K, 44, 0.8);
        g.shake(8, 0.3); g.sfx("ultgo", 0.2);
        k.bub = null;
      }
    }
    if (p.formT > 0) { k.u += dt; ultStep(g, k, dt); }
    if (tickUlt(g, dt)) { k.u = -1; k.slams = []; }
  },
  fire(g) {
    const p = g.p, k = KK(g), lv = g.lvl("weapon");
    const n = lv >= 7 ? 5 : 4;
    const w = k.i++ % n;
    const dm = g.dmgMult() * scale(g);
    const lanes = lv >= 3 ? [-10, 10] : [0];
    for (const o of lanes) {
      const x = p.x + o, y = p.y - 26;
      if (w === 0) word(g, x, y, -Math.PI / 2, 0, 620, 12 * dm);
      else if (w === 1) word(g, x, y, -Math.PI / 2, 1, 1100, 9 * dm, { pierce: 3 });
      else if (w === 2) word(g, x, y, -Math.PI / 2, 2, 360, 6 * dm);
      else if (w === 3) for (const a of [-0.3, 0, 0.3]) word(g, x, y, -Math.PI / 2 + a, 3, 650, 8 * dm);
      else for (let i = 0; i &lt; 8; i++) word(g, x, y, (i / 8) * TAU, 4, 520, 6 * dm);
    }
    p.fireT += iv(g, 0.17);
    g.sfx(["shoot1", "shoot3", "shoot4", "shoot2", "shoot5"][w], 0.05);
  },
  updateBullet(g, b: PBullet, dt) {
    if (b.kind !== 20) return;
    b.x += b.vx * dt; b.y += b.vy * dt;
    if (b.a === 2) { b.c -= dt; if (b.c &lt;= 0) { boom(g, b.x, b.y); b.alive = false; } }
  },
  onHit(g, b, e, dmg) {
    if (b.kind !== 20) return;
    if (b.a === 0) g.aoe(b.x, b.y, 40, dmg * 0.7, { small: true, noText: true, pal: "red", skip: e.uid });
    else if (b.a === 2) boom(g, b.x, b.y);
    else if (b.a === 3 && !e.boss && !e.def.mid) { e.y -= 12; e.freeze = Math.max(e.freeze, 0.15); }
  },
  castActive(g) {
    KK(g).bub = { t: 0, dur: 3, cap: 0 };
    g.sfx("whoosh");
  },
  castUlt(g) {
    const k = KK(g);
    startUlt(g, 9.5);
    k.u = 0; k.tick = 0; k.slams = []; k.stamp = false; k.shattered = false;
    ultBackdrop(g, 9.5, 0.45, 0.7);
    mkFx(g, 2, () => g.p.formT > 0, (ctx) => drawUlt(g, ctx, k));
  },
  drawOver(g, ctx) {
    const k = KK(g), p = g.p;
    // 文字彈
    ctx.save();
    ctx.textAlign = "center"; ctx.textBaseline = "middle"; ctx.lineJoin = "round";
    for (const b of g.pb.items) {

      ctx.fillStyle = wd.col; ctx.fillText(wd.ch, 0, 0);
      ctx.restore();
    }
    ctx.restore();
    // 對話框
    const bb = k.bub;
    if (bb) {
      const al = g.lvl("active");
      const bw = al >= 3 ? 280 : 220, bh = al >= 3 ? 150 : 120, bx = p.x, by = p.y - 40 - bh / 2 - 30;
      const pop = easeOut(Math.min(1, bb.t / 0.2)) * (bb.t > bb.dur - 0.15 ? 1.15 : 1);

      ctx.fillStyle = "rgba(255,255,255,0.92)"; ctx.strokeStyle = K; ctx.lineWidth = 4;
      ctx.beginPath(); ctx.ellipse(0, 0, bw / 2, bh / 2, 0, 0, TAU); ctx.fill(); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(-10, bh / 2 - 6); ctx.lineTo(0, bh / 2 + 34); ctx.lineTo(16, bh / 2 - 8); ctx.closePath(); ctx.fill(); ctx.stroke();
      ctx.font = `900 22px "Noto Sans TC"`; ctx.textAlign = "center"; ctx.textBaseline = "middle";

      ctx.fillStyle = K; ctx.font = `900 16px "Noto Sans TC"`; ctx.fillText(`吞下 ${bb.cap} 發`, 0, 0);
      ctx.restore();
    }
    // 分格閃擊
    if (k.panel > 0) {

    }
  },
  preview() { return { mode: 0, aux: 0 }; },

// ── 奧義：完結篇．跨頁大爆發 ──────────────────────
function ultStep(g: Game, k: K, dt: number) {
  const u = k.u, lv = g.lvl("ult"), dm = g.dmgMult();
  if (u &lt; 3.4) for (const b of g.eb.items) if (b.alive) b.slow = 0.25;
  if (u >= 3 && !k.shattered) {
    k.shattered = true;
    for (let i = 0; i &lt; 70; i++) g.fx.shard(rand(W), rand(80, 880), rand(-300, 300), rand(-300, 300), rand(0.5, 1), rand(6, 14), i % 3 ? "#ffffff" : Y, 300);
    g.flash("#ffffff", 0.8); g.sfx("boom_l"); g.shake(12, 0.4);
  }
  if (u >= 3.4 && u &lt; 7.5) {
    k.tick -= dt;
    if (k.tick &lt;= 0) {
      k.tick = lv >= 3 ? 0.28 : 0.4;
      const l = live(g);
      const e = l.length ? l[Math.floor(Math.random() * l.length)] : null;
      const x = e ? e.x : rand(80, W - 80), y = e ? e.y : rand(100, H * 0.55);
      k.slams.push({ x, y, t: 0 });
      g.later(0.12, () => {
        const R = lv >= 5 ? 140 : 110;
        g.aoe(x, y, R, 90 * dm, { pal: "gold" });
        g.fx.ring(x, y, 20, R * 1.3, 0.4, K, 10); g.fx.ring(x, y, 10, R, 0.35, M, 5);
        g.shake(9, 0.2); g.sfx("boom_m", 0.05);
      });
    }
  }
  for (const s of k.slams) s.t += dt;
  k.slams = k.slams.filter((s) => s.t &lt; 0.9);
  if (u >= 8.2 && !k.stamp) {
    k.stamp = true;
    g.aoe(W / 2, H / 2, 1500, 300 * dm * (lv >= 7 ? 2 : 1), { pal: "red" });
    g.clearBullets(W / 2, H / 2, 3000, "score");
    for (let i = 0; i &lt; 60; i++) { const a = rand(TAU), s = rand(200, 900); g.fx.shard(W / 2, H * 0.42, Math.cos(a) * s, Math.sin(a) * s, rand(0.5, 1.2), rand(4, 12), i % 2 ? K : "#e8102e", 200); }
    g.flash("#ff2f4a", 0.6); g.shake(22, 0.8); g.sfx("ultgo");
  }
}
function halftone(ctx: CanvasRenderingContext2D, col: string, a: number, t: number) {
  ctx.save(); ctx.globalAlpha = a; ctx.fillStyle = col;
  for (let y = 0; y &lt; H; y += 14) for (let x = 0; x &lt; W; x += 14) {
    const d = Math.hypot(x - W / 2, y - H * 0.4) / 500;
    const r = Math.max(0.5, (1 - d) * 5 + Math.sin(t * 4 + d * 10) * 1);
