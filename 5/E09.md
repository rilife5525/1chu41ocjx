function drawStar(ctx: CanvasRenderingContext2D, x: number, y: number, r: number, rot: number, hot: boolean) {
  ctx.save();
  ctx.translate(x, y);
  ctx.globalCompositeOperation = "lighter";
  glowAt(ctx, 0, 0, r * (hot ? 3 : 2), L, hot ? 0.8 : 0.45);
  ctx.globalCompositeOperation = "source-over";
  ctx.rotate(rot);
  ctx.fillStyle = "#c8d0e0";
  for (let i = 0; i &lt; 10; i++) {
    const a = (i / 10) * TAU;
    ctx.beginPath(); ctx.moveTo(Math.cos(a - 0.18) * r * 0.85, Math.sin(a - 0.18) * r * 0.85); ctx.lineTo(Math.cos(a) * r * 1.45, Math.sin(a) * r * 1.45); ctx.lineTo(Math.cos(a + 0.18) * r * 0.85, Math.sin(a + 0.18) * r * 0.85); ctx.fill();
  }
  const g = ctx.createRadialGradient(-r * 0.3, -r * 0.3, r * 0.1, 0, 0, r);
  g.addColorStop(0, "#ffffff"); g.addColorStop(0.4, "#9aa4b8"); g.addColorStop(1, "#2a3040");
  ctx.fillStyle = g; ctx.beginPath(); ctx.arc(0, 0, r, 0, TAU); ctx.fill();
  ctx.strokeStyle = L; ctx.lineWidth = Math.max(1, r * 0.08);
  ctx.beginPath(); ctx.moveTo(-r * 0.5, -r * 0.2); ctx.lineTo(-r * 0.1, r * 0.1); ctx.lineTo(r * 0.4, -r * 0.1); ctx.moveTo(-r * 0.1, r * 0.1); ctx.lineTo(0, r * 0.6); ctx.stroke();
  ctx.restore();
}

const PX = W / 2, PY = -420, PL = 900, SR = 130;
function ultStep(g: Game, k: K, dt: number) {
  const u = k.u, lv = g.lvl("ult"), dm = g.dmgMult();
  const swings = lv >= 3 ? 4 : 3;
  const A = 1.0;
  if (u &lt; 1.2) {
    const f = easeOut(u / 1.2);
    const tx = PX - Math.sin(A) * PL, ty = PY + Math.cos(A) * PL;
    k.sx = PX + (tx - PX) * f; k.sy = -300 + (ty + 300) * f;
  } else if (u &lt; 7) {
    const half = 5.8 / swings;
    const th = -A * Math.cos((Math.PI * (u - 1.2)) / half);
    k.sx = PX + Math.sin(th) * PL; k.sy = PY + Math.cos(th) * PL;
    k.tick -= dt;
    if (k.tick &lt;= 0) {
