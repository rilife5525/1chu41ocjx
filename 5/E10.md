function drawPendulum(g: Game, ctx: CanvasRenderingContext2D, k: K, ghosts: { x: number; y: number }[]) {
  const u = k.u, t = g.time;
  if (u > 8.6) {
    // 隕坑殘光
    const f = clamp((u - 8.3) / 0.7, 0, 1);
    ctx.save(); ctx.globalCompositeOperation = "lighter"; glowAt(ctx, PX, H * 0.5, 300 * (1 - f * 0.5), L, 0.6 * (1 - f)); ctx.restore();
    return;
  }
  ctx.save();
  // 鏈條
  const px = k.sy &lt; 130 && u > 7 ? PX : PX, py = PY;
  ctx.strokeStyle = hexA(S, 0.8); ctx.lineWidth = 4;
  const n = 30;
  for (let i = 0; i &lt; n; i++) {
    const f = i / n, x = px + (k.sx - px) * f, y = py + (k.sy - py) * f;
    ctx.beginPath(); ctx.ellipse(x, y, 6, 3, Math.atan2(k.sy - py, k.sx - px) + (i % 2 ? Math.PI / 2 : 0), 0, TAU); ctx.stroke();
  }
  ctx.globalCompositeOperation = "lighter";
  ctx.strokeStyle = hexA(L, 0.4); ctx.lineWidth = 8;
  ctx.beginPath(); ctx.moveTo(px, py); ctx.lineTo(k.sx, k.sy); ctx.stroke();
  // 動態殘影
  ghosts.forEach((gh, i) => { ctx.globalAlpha = (i / ghosts.length) * 0.25; glowAt(ctx, gh.x, gh.y, SR * 1.2, L, (i / ghosts.length) * 0.3); });
  ctx.globalAlpha = 1;
  ctx.restore();
  drawStar(ctx, k.sx, k.sy, SR * 0.72, t * 1.5, true);
  // 巨星裂縫光
  ctx.save(); ctx.globalCompositeOperation = "lighter";
  for (let i = 0; i &lt; 6; i++) { const a = t + i; glowAt(ctx, k.sx + Math.cos(a) * SR * 0.4, k.sy + Math.sin(a) * SR * 0.4, 20, L, 0.5); }
  ctx.restore();
}
