// 霓彈：粉紅三角翼＋側掛緩衝艙，進化會增加霓虹滾邊與環繞鋼珠
import { drawGear, flame, fillPoly, mirror, lin, rad, ring, type ShipDrawOpts } from "./shipUtil";

export function drawBumper(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t, tier = o.tier;
  drawGear(ctx, o, false);
  // 尾焰（雙粉焰＋中央水藍焰）
  for (const s of [-1, 1]) flame(ctx, s * 9, 30, 24 + o.thrust * 16, 9, t, "#ffffff", "#ff3d7f", "rgba(255,61,127,0)");
  flame(ctx, 0, 34, 18 + o.thrust * 12, 6, t + 1, "#ffffff", "#35f0ff");
  // 環繞鋼珠（第 4 階）
  if (tier >= 3) {
    for (let i = 0; i &lt; 3; i++) {
      const a = t * 2.4 + (i * 6.2832) / 3;
      const x = Math.cos(a) * 46, y = Math.sin(a) * 18 + 4;
      ctx.fillStyle = rad(ctx, x - 2, y - 2, 0, 6, [[0, "#fff"], [0.5, "#9fd8ff"], [1, "#2a2f66"]]);
      ctx.beginPath(); ctx.arc(x, y, 5, 0, 7); ctx.fill();
    }
  }
  // 機翼＋緩衝艙
  mirror(ctx, () => {
    fillPoly(ctx, [[7, -10], [42, 20], [46, 33], [28, 29], [10, 22]], lin(ctx, 7, -10, 46, 33, [[0, "#ff7aa8"], [0.6, "#c2185b"], [1, "#4a0c3a"]]), "#ffffff", 1.2);
    ctx.strokeStyle = "#35f0ff"; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(11, -3); ctx.lineTo(41, 23); ctx.stroke();
    if (tier >= 1) {
      ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.strokeStyle = "rgba(255,61,127,.8)"; ctx.lineWidth = 4;
      ctx.beginPath(); ctx.moveTo(11, -3); ctx.lineTo(41, 23); ctx.stroke(); ctx.restore();
    }
    // 緩衝艙：圓形保險桿
    const px = 29, py = 12;
    ctx.fillStyle = rad(ctx, px - 2, py - 2, 0, 11, [[0, "#ffffff"], [0.45, "#35f0ff"], [1, "#12306a"]]);
    ctx.beginPath(); ctx.arc(px, py, 9, 0, 7); ctx.fill();
    ring(ctx, px, py, 10.5, "#ffffff", 1.6);
    const pulse = 0.5 + 0.5 * Math.sin(t * 6);
    ctx.save(); ctx.globalCompositeOperation = "lighter"; ring(ctx, px, py, 12 + pulse * 3, `rgba(255,61,127,${0.6 - pulse * 0.4})`, 2); ctx.restore();
    // 前鴨翼（第 3 階）
    if (tier >= 2) fillPoly(ctx, [[7, -24], [19, -14], [8, -12]], "#35f0ff", "#ffffff", 1);
  });
  // 機身
  fillPoly(ctx, [[0, -46], [6, -24], [9, 4], [7, 28], [0, 35], [-7, 28], [-9, 4], [-6, -24]],
    lin(ctx, -9, 0, 9, 0, [[0, "#7a1450"], [0.5, "#ffd0e2"], [1, "#7a1450"]]), "#ffffff", 1.3);
  ctx.strokeStyle = "rgba(8,6,15,.5)"; ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(0, -30); ctx.lineTo(0, 30); ctx.stroke();
  // 座艙
  ctx.fillStyle = lin(ctx, 0, -22, 0, 4, [[0, "#e8ffff"], [1, "#1a74d9"]]);
  ctx.beginPath(); ctx.ellipse(0, -10, 4.4, 11, 0, 0, 7); ctx.fill();
  ctx.fillStyle = "rgba(255,255,255,.8)"; ctx.beginPath(); ctx.ellipse(-1.4, -14, 1.3, 5, 0, 0, 7); ctx.fill();
  // 鼻尖霓虹
  ctx.save(); ctx.globalCompositeOperation = "lighter"; ctx.fillStyle = "#ff3d7f";
  ctx.beginPath(); ctx.arc(0, -40, 3 + Math.sin(t * 8), 0, 7); ctx.fill(); ctx.restore();
  drawGear(ctx, o, true);
}

// 回聲：前掠翼攔截機，尾部拖出半透明殘響輪廓，機頭前方有聲波弧線
import { drawGear, flame, fillPoly, mirror, lin, rad, type ShipDrawOpts } from "./shipUtil";

function hull(ctx: CanvasRenderingContext2D, o: ShipDrawOpts, fill: boolean) {
  const tier = o.tier;
  mirror(ctx, () => {
    const wing: [number, number][] = [[7, 2], [46, -16], [48, -9], [14, 20]];
    if (fill) fillPoly(ctx, wing, lin(ctx, 7, 2, 48, -9, [[0, "#1b2a66"], [0.6, "#3df5d0"], [1, "#d6fff6"]]), "#e8fffb", 1.2);
    else { ctx.beginPath(); ctx.moveTo(7, 2); ctx.lineTo(46, -16); ctx.lineTo(48, -9); ctx.lineTo(14, 20); ctx.closePath(); ctx.stroke(); }
    if (fill && tier >= 1) fillPoly(ctx, [[14, 4], [38, -8], [16, 14]], "rgba(138,92,255,.7)");
    if (fill && tier >= 2) fillPoly(ctx, [[8, 24], [24, 30], [10, 34]], "#8a5cff", "#e8fffb", 1);
  });
  if (fill) {
    fillPoly(ctx, [[0, -46], [6, -24], [8, 8], [6, 28], [0, 33], [-6, 28], [-8, 8], [-6, -24]],
      lin(ctx, -8, 0, 8, 0, [[0, "#0b1030"], [0.5, "#4a5bd0"], [1, "#0b1030"]]), "#3df5d0", 1.4);
  } else {
    ctx.beginPath(); ctx.moveTo(0, -46); ctx.lineTo(6, -24); ctx.lineTo(8, 8); ctx.lineTo(6, 28); ctx.lineTo(0, 33); ctx.lineTo(-6, 28); ctx.lineTo(-8, 8); ctx.lineTo(-6, -24); ctx.closePath(); ctx.stroke();
  }
}

export function drawEcho(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const t = o.t, tier = o.tier;
  drawGear(ctx, o, false);
  // 殘響輪廓（往後拖曳）
  ctx.save();
  ctx.strokeStyle = "#8a5cff"; ctx.lineWidth = 1.5;
  for (let i = 2; i >= 1; i--) {
    ctx.save(); ctx.translate(Math.sin(t * 3 + i) * 2, i * (9 + o.thrust * 5));
    ctx.globalAlpha = 0.34 / i; hull(ctx, o, false); ctx.restore();
  }
  ctx.restore();
  // 環形噴口
  ctx.save();
