const B = EXT_BASE_CZ;
interface Note { x: number; y: number; taken: boolean; t: number }

/** 畫一顆音符（八分音符） */
function drawNote(c: CanvasRenderingContext2D, x: number, y: number, s: number, col: string, a = 1) {
  c.save();
  c.translate(x, y);
  c.globalAlpha = a;
  c.fillStyle = col;
  c.beginPath(); c.ellipse(-s * 0.1, s * 0.5, s * 0.36, s * 0.26, -0.4, 0, 6.3); c.fill();
  c.fillRect(s * 0.17, -s * 0.75, s * 0.1, s * 1.28);
  c.beginPath(); c.moveTo(s * 0.27, -s * 0.75); c.quadraticCurveTo(s * 0.9, -s * 0.55, s * 0.55, -s * 0.05); c.quadraticCurveTo(s * 0.7, -s * 0.5, s * 0.27, -s * 0.45); c.fill();
  c.restore();
}

function pulse(g: Game) {
  const p = g.p;
  const k = p.k;
  const lv = g.lvl("passive");
  const r = 95 + lv * 14 + (lv >= 5 ? 35 : 0);
  const dm = g.dmgMult();
  g.fx.ring(p.x, p.y, 12, r, 0.45, "#ff7ab8", 4);
  g.fx.ring(p.x, p.y, 6, r * 0.8, 0.5, "#ffc94a", 2);
  const n = g.aoe(p.x, p.y, r, 12 * dm * (1 + lv * 0.15), { pal: "gold", noFx: true, noText: true });
  g.clearBullets(p.x, p.y, r * 0.65, "none");
  k.pc++;
  if (lv >= 3 && n > 0 && k.pc % 3 === 0) g.heal(2);
  if (lv >= 7 && k.pc % 3 === 0) for (let i = 0; i &lt; 8; i++) g.spawnPB(p.x, p.y - 10, (i / 8) * 6.283, 420, B + 1, 11 * dm, { homing: 5, life: 2.2, src: 2 });
  g.sfx("shoot2", 0.08);
}

function grandChord(g: Game, big: boolean) {
  const dm = g.dmgMult();
  g.flash("#ffc94a", 0.45);
  g.shake(9, 0.4);
  g.fx.ring(W / 2, H / 2, 20, 700, 0.8, "#ffc94a", 8);
  g.fx.ring(W / 2, H / 2, 20, 540, 0.7, "#ff7ab8", 5);
  for (const e of enemyList(g)) g.damageEnemy(e, (big ? 90 : 70) * dm * (1 + g.lvl("active") * 0.12));
  g.clearBullets(W / 2, H / 2, 99999, "coin");
  g.sfx("evolve");
}
