function updateBumps(g: Game, dt: number) {
  const k = g.p.k, bumps: Bump[] = k.bumps, lv = g.lvl("active");
  for (let i = bumps.length - 1; i >= 0; i--) {
    const m = bumps[i];
    m.t -= dt; m.pulse = Math.max(0, m.pulse - dt * 4);
    if (m.t &lt;= 0) { g.fx.explosion(m.x, m.y, 16, { core: "#fff", mid: "#ff3d7f", edge: "#35f0ff", smoke: "#1a0933" }); bumps.splice(i, 1); continue; }
    for (const b of g.eb.items) {
      if (!b.alive) continue;
      const dx = b.x - m.x, dy = b.y - m.y, rr = 28 + b.r;
      if (dx * dx + dy * dy &lt; rr * rr) { b.alive = false; m.pulse = 1; g.score += 5; g.fx.spark(b.x, b.y, dx * 4, dy * 4, 0.2, 2, "#35f0ff", 2); }
    }
    if (lv >= 5) {
      m.shoot -= dt;
      if (m.shoot &lt;= 0) {
        const e = g.nearest(m.x, m.y, 700, (q) => q.y > 0);
        if (e) { m.shoot = 0.6; g.spawnPB(m.x, m.y, Math.atan2(e.y - m.y, e.x - m.x), 760, S(1), 9, { homing: 0.5, src: 2 }); }
      }
    }
  }
  if (lv >= 7 && bumps.length > 1) {
    k.webT = (k.webT ?? 0) - dt;
    if (k.webT &lt;= 0) {
      k.webT = 0.15;
      for (let i = 0; i &lt; bumps.length; i++) {
        const a = bumps[i], c = bumps[(i + 1) % bumps.length];
        for (const e of g.en.items) if (!e.dead && e.alive && segDist(e.x, e.y, a.x, a.y, c.x, c.y) &lt; 14 + e.r * 0.6) g.damageEnemy(e, 6 * g.dmgMult(), { noText: true, noHitFx: true });
      }
    }
  }
}

export const bumperKit: ShipKit = {
  init(g) { g.p.k = { stack: 0, stackT: 0, fever: 0, feverBall: 0, bumps: [] as Bump[], ballAt: 0.5, volley: 0, rallies: 0 }; },

  update(g, dt) {
    const k = g.p.k;
    if (k.stack > 0 && k.fever &lt;= 0) { k.stackT -= dt; if (k.stackT &lt;= 0) { k.stack--; k.stackT = 1; } }
    if (k.fever > 0) {
      k.fever -= dt; g.addUlt(dt * 3);
      if (k.fever &lt;= 0) { k.fever = 0; k.stack = Math.floor(maxStack(g) / 2); }
      if (g.lvl("passive") >= 7) { k.feverBall -= dt; if (k.feverBall &lt;= 0) { k.feverBall = 0.5; launchBall(g); } }
    }
    updateBumps(g, dt);
  },

  fire(g) {
    const p = g.p, k = p.k;
    const mul = 1 + 0.03 * k.stack;
    k.volley++;
    for (const sx of [-1, 1]) g.spawnPB(p.x + sx * 10, p.y - 30, UP + sx * 0.025, 920, S(1), 7 * mul);
    if (k.volley % 3 === 0) for (const sx of [-1, 1]) g.spawnPB(p.x + sx * 18, p.y - 20, UP + sx * 0.16, 860, S(1), 5 * mul);
    const maxB = g.lvl("weapon") >= 3 ? 3 : 2;
    if (g.t >= k.ballAt && countBalls(g) &lt; maxB) { launchBall(g); k.ballAt = g.t + 1.1 / p.s.rate; }
    p.fireT += 0.115 / (p.s.rate * (1 + 0.04 * k.stack));
  },

  updateBullet(g, b, dt) {
    if (b.style !== S(0)) return;
    const p = g.p, k = p.k;
    b.vy += 430 * dt;
    const sp = Math.hypot(b.vx, b.vy);
    if (sp > 1000) { b.vx *= 1000 / sp; b.vy *= 1000 / sp; }
    b.x += b.vx * dt; b.y += b.vy * dt;
    if (b.x &lt; 18) { b.x = 18; b.vx = Math.abs(b.vx); g.fx.hit(b.x, b.y, "#35f0ff"); }
    if (b.x > W - 18) { b.x = W - 18; b.vx = -Math.abs(b.vx); g.fx.hit(b.x, b.y, "#35f0ff"); }
    if (b.y &lt; 56) { b.y = 56; b.vy = Math.abs(b.vy) * 0.92; g.fx.hit(b.x, b.y, "#ff3d7f"); }
    if (b.c > 0) { b.c -= dt; if (b.c &lt;= 0) b.hits.length = 0; }
    for (const m of k.bumps as Bump[]) {
      const dx = b.x - m.x, dy = b.y - m.y, d = Math.hypot(dx, dy) || 1, rr = 28 + b.r;
      if (d &lt; rr) {
        const nx = dx / d, ny = dy / d, dot = b.vx * nx + b.vy * ny;
        if (dot &lt; 0) { b.vx -= 2 * dot * nx; b.vy -= 2 * dot * ny; }
        const s = Math.max(560, Math.hypot(b.vx, b.vy) * 1.1), l = Math.hypot(b.vx, b.vy) || 1;
        b.vx = (b.vx / l) * s; b.vy = (b.vy / l) * s;
        b.x = m.x + nx * rr; b.y = m.y + ny * rr; m.pulse = 1; b.a += 1;
        g.aoe(m.x, m.y, 74, 11 * g.dmgMult(), { pal: "cyan", small: true, noText: true });
        g.fx.ring(m.x, m.y, 20, 80, 0.3, "#35f0ff", 3); audio.sfx("hit", 0.05);
      }
    }
    if (b.vy > 0) {
      const dx = b.x - p.x, dy = b.y - (p.y - 12), d = Math.hypot(dx, dy);
      if (d &lt; 56 && b.y &lt; p.y + 24) bat(g, b, d);
    }
    if (b.y > H + 30) {
      b.alive = false; k.stack = Math.max(0, k.stack - 2); k.stackT = 1;
      g.fx.text(clamp(b.x, 40, W - 40), H - 60, "MISS", "#ff3d7f", 18, 0.8);
    }
    if (b.x === b.x && Math.random() &lt; 0.7) g.fx.glowP(b.x, b.y, 0, 0, 0.22, 17, "#ff3d7f", 3, 0, 0.6);
    b.dmg = b.b * (1 + 0.18 * Math.min(b.a, 10)) * (1 + 0.03 * k.stack) * (k.fever > 0 ? 1.3 : 1);
  },

  onHit(g, b, e) {

    b.vx = vx; b.vy = vy; b.x = e.x + nx * (e.r * 0.9 + b.r + 3); b.y = e.y + ny * (e.r * 0.9 + b.r + 3);
    b.c = 0.14; b.a += 0.5; g.shake(2, 0.08);
