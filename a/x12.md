// 關卡選擇：Boss 設定圖卡片、難度、解鎖、最高紀錄、無限模式
import { useState } from "react";
import { bossArtStyle } from "../assets";
import { DIFFS } from "../game/constants";
import { BOSSES, ENEMIES } from "../game/data/enemies";
import { STAGES } from "../game/data/stages";
import { SHIPS, shipTier } from "../game/data/ships";
import { useSave } from "../game/save";
import { Btn, Currency, FullscreenBtn, ShipCanvas } from "./ui";
import { Icon } from "./icons";

export function StageSelect({ onBack, onHangar, onStart, initial = 1 }: { onBack: () => void; onHangar: () => void; onStart: (stageId: number, diff: number) => void; initial?: number }) {
  const save = useSave();
  const [sel, setSel] = useState(initial);
  const ship = SHIPS[save.sel];
  const tier = shipTier(save.ships[save.sel].lv);
  const cleared = (n: number, d = 0) => !!save.stages[String(n)]?.clear?.[d];
  const stageOpen = (n: number) => n === 1 || cleared(n - 1);
  const diffOpen = (n: number, d: number) => (n === 0 ? d === 0 || (d === 1 && cleared(3)) || (d === 2 && cleared(6)) : d === 0 || cleared(n, d - 1));
  // 難度：記住上輪選的（save.lastDiff）；對當前關未解鎖時回落「簡單」
  const [diff, setDiff] = useState(() => {
    const d = save.lastDiff ?? 0;
    return d > 0 && !diffOpen(initial, d) ? 0 : d;
  });
  const st = sel > 0 ? STAGES[sel - 1] : null;
  const boss = st ? BOSSES[st.boss] : null;
  const hi = sel === 0 ? save.endless.hi : save.stages[String(sel)]?.hi?.[diff] || 0;
  const open = sel === 0 || stageOpen(sel);
  const dOk = diffOpen(sel, diff);
  const pick = (n: number) => {
    setSel(n);
    if (!diffOpen(n, diff)) setDiff(0);
  };

  return (
    <div className="screen bg-[var(--ink)]">
      <div className="absolute inset-0 halftone opacity-40" />
      <div className="absolute -right-32 top-0 bottom-0 w-[55%] bg-[var(--red)]/20 -skew-x-12" />
      <div className="relative h-full flex flex-col p-3 sm:p-5 gap-3">
        {/* 標題列 */}
        <div className="flex items-center gap-3">
          <button onClick={onBack} className="w-10 h-10 grid place-items-center bg-black border-2 border-white -skew-x-6 hover:bg-[var(--red)]">
            <Icon name="back" />
          </button>
          <div>
            <div className="disp text-4xl sm:text-5xl leading-none title-jag !text-[40px] sm:!text-[52px]" style={{ transform: "rotate(-2deg)" }}>SORTIE</div>
          </div>
          <div className="text-sm tracking-[0.4em] text-white/60 font-black hidden sm:block">出擊・選擇任務</div>
          <div className="ml-auto flex items-center gap-2">
            <Currency coins={save.coins} cores={save.cores} className="hidden sm:flex" />
            <FullscreenBtn />
          </div>
        </div>

        <div className="flex-1 min-h-0 flex flex-col lg:flex-row gap-3">
          {/* 卡片區 */}
          <div className="shrink-0 lg:shrink lg:flex-1 min-h-0 grid grid-cols-3 lg:grid-cols-4 gap-2 sm:gap-3 content-start auto-rows-fr">
            {STAGES.map((s, i) => {
              const ok = stageOpen(s.id);
              const on = sel === s.id;
              return (
                <button
                  key={s.id}
                  onClick={() => pick(s.id)}
                  className={`relative overflow-hidden text-left aspect-[5/4] lg:aspect-[4/3] border-2 transition-transform ${on ? "border-[var(--red)] scale-[1.03] z-10" : "border-white/30 hover:border-white"}`}
                  style={{ animation: `slideInU .4s ${i * 0.06}s both`, clipPath: "polygon(0 0, 100% 0, 100% calc(100% - 14px), calc(100% - 14px) 100%, 0 100%)" }}
                >
                  <div className="absolute inset-0 bg-cover" style={{ ...bossArtStyle(BOSSES[s.boss].art), filter: ok ? "none" : "grayscale(1) brightness(0.4)" }} />
                  <div className="absolute inset-0 bg-gradient-to-t from-black via-black/30 to-transparent" />
                  {on && <div className="absolute inset-0 border-4 border-[var(--red)]/70 pointer-events-none" />}

                  </div>
                  {!ok && (
                    <div className="absolute inset-0 grid place-items-center bg-black/50">
                      <Icon name="lockicon" size={30} />
                    </div>
                  )}
                </button>
              );
            })}
            {/* 無限模式 */}
            <button
              onClick={() => pick(0)}
              className={`relative overflow-hidden text-left h-[72px] lg:h-auto border-2 col-span-3 lg:col-span-2 ${sel === 0 ? "border-[var(--red)] scale-[1.01]" : "border-white/30 hover:border-white"}`}

            >

                <Icon name="infinity" size={90} />

/** 手機端暫停時的側邊抽屜。
 *  左側＝資訊頁（跟手 peek 驅動，開右滑／關左滑，手勢在暫停覆蓋層）；
 *  右側＝立繪頁（開＝純按鈕；關＝按頂端鈕或右滑，右滑手勢掛在面板本體，跟左側隔離避免打架）。
 *  peek＝「開出程度」（0=全收、w=全開），左側收在左、右側收在右。 */
function PhoneDrawer({ side, open, peek, dragging, onOpenChange, ship, shipId, stage, diff, fps, maxW, panelProps }: {
  side: "left" | "right";
  open: boolean;
  peek?: number | null; // 跟手開出程度（0=全收、w=全開）；null=未在拖，依 open 定位
  dragging?: boolean;   // 跟手中：關掉過場，讓抽屜貼手
  onOpenChange: (o: boolean) => void;
  ship: { color: string; color2: string; en: string; name: string; pilot: string };
  shipId: ShipId;
  stage: { id: number; name: string } | null;
  diff: number;
  fps: number;
  maxW: number; // 戰鬥區寬度，抽屜寬度以其為準
  panelProps?: { onPointerDown?: (e: RPointerEvent<HTMLDivElement>) => void; onPointerMove?: (e: RPointerEvent<HTMLDivElement>) => void; onPointerUp?: (e: RPointerEvent<HTMLDivElement>) => void; onPointerCancel?: (e: RPointerEvent<HTMLDivElement>) => void };
}) {
  const w = Math.min(320, Math.round(maxW * 0.85));
  // 左側收左（peek-w）；右側收右（w-peek）；未在跟手時依 open 定位
  const tx = side === "left"
    ? (peek != null ? peek - w : open ? 0 : -w)
    : (peek != null ? w - peek : open ? 0 : w);

  return (
    <div
      className={`absolute top-0 bottom-0 z-[45] flex flex-col justify-center ${open || dragging ? "pointer-events-auto" : "pointer-events-none"}`}
      style={{
        [side]: 0,
        width: w,
        transform: `translateX(${tx}px)`,
        transition: dragging ? "none" : "transform .22s cubic-bezier(.3,.9,.4,1)",
        background: "rgba(8,8,14,0.97)",
        [side === "left" ? "borderRight" : "borderLeft"]: "2px solid var(--red)",
        boxShadow: side === "left" ? "8px 0 40px rgba(0,0,0,0.8)" : "-8px 0 40px rgba(0,0,0,0.8)",
        // 掛本體手勢時允許垂直捲透、水平交給 JS 跟手
        ...(panelProps ? { touchAction: "pan-y" as const } : null),
      } as CSSProperties}
      {...panelProps}
    >
      {/* 收合把手：抽屜頂端，純圖示、無外框、紅呼吸 */}
      {open && (
        <button
          onClick={() => onOpenChange(false)}
          aria-label="收合側欄"
          className="absolute top-3 px-2 text-[20px] font-black leading-none active:opacity-60"
          style={{
            [side === "left" ? "right" : "left"]: 12,
            color: "#ff3040",
            animation: "breatheRed 1.6s ease-in-out infinite",
          }}
        >
          {side === "left" ? "◀" : "▶"}
        </button>
      )}
      {side === "left" ? (
        /* 資訊頁：關卡、難度、操作、FPS（可隨時滑動拉出） */
        <div className="mx-auto w-[88%] panel jag p-3 my-3 border-l-4 border-[var(--red)]">
          <div className="disp text-xl leading-none">{stage ? `STAGE ${stage.id}` : "ENDLESS"}</div>
          <div className="font-black mt-1">{stage ? stage.name : "無限模式"}</div>
          <div className="text-[11px] text-white/60 mt-0.5">{DIFFS[diff].name}難度</div>
          <div className="space-y-1 mt-3 text-[12px] text-white/80">
            <div><b className="text-white">移動</b> 拖曳</div>
            <div><b className="text-[var(--cyan)]">主動技</b> 左下按鈕</div>
            <div><b className="text-[#c58bff]">奧義</b> 右下按鈕</div>
            <div><b>暫停</b> 右上按鈕</div>
          </div>
          <div className="mt-3 text-[11px] text-white/45 tracking-widest">FPS {fps}</div>
        </div>
      ) : (
        /* 立繪頁：機師立繪＋機體名／駕駛（僅按鈕開關，名字字放大） */
        <div className="mx-auto w-[88%] flex flex-col items-center gap-2.5 px-2 my-3">
          <div className="relative w-full aspect-[1/2] jag-torn overflow-hidden">
            <Portrait id={shipId} variant="battle" className="absolute inset-0" style={{ filter: `drop-shadow(0 0 20px ${ship.color}99)` }} />
          </div>
          <div className="w-full bg-black/85 px-3 py-2.5 border-l-4" style={{ borderColor: ship.color }}>
            <div className="disp text-[26px] leading-tight" style={{ color: ship.color2 }}>{ship.en}</div>
            <div className="text-[20px] font-black text-white/85 mt-1">{ship.name}｜{ship.pilot}</div>
          </div>
        </div>
      )}
    </div>
  );
}

export function GameScreen({ shipId, stageId, diff, onExit, onRetry, onNext }: Props) {
  const save = useSave();
  const wrapRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const gameRef = useRef<Game | null>(null);
  const [dim, setDim] = useState({ w: 360, h: 640, wide: false });
  const [hud, setHud] = useState<HudData | null>(null);
  const [cards, setCards] = useState<CardChoice[] | null>(null);
  const [boss, setBoss] = useState<{ b: BossDef; stage: number; k: number } | null>(null);
  const [banner, setBanner] = useState<{ title: string; sub: string; color: string; k: number } | null>(null);
  const [evo, setEvo] = useState<{ name: string; skill: string; tier: number; k: number } | null>(null);
  const [state, setState] = useState<GameState>("playing");
  const [result, setResult] = useState<RunResult | null>(null);
  const [showSettings, setShowSettings] = useState(false);
  const [confirmQuit, setConfirmQuit] = useState(false);
  const [hint, setHint] = useState(!save.tutorialSeen);
  // 手機側邊抽屜：左＝資訊頁（可滑動跟手、也可按按鈕）、右＝立繪頁（純按鈕，不跟手避免打架）
  const [drawerL, setDrawerL] = useState(false);
  const [drawerR, setDrawerR] = useState(false);
  // 左側跟手拖出進度（0=全收、w=全開），null=未在拖
  const [peekL, setPeekL] = useState<number | null>(null);
  // 右側（立繪頁）跟手收合：openness（0=全收、w=全開），開著時右滑往外推出收合，null=未在拖
  const [peekR, setPeekR] = useState<number | null>(null);
  // 手勢狀態（僅控制左側資訊頁；照 chat.html 的跟手＋速度 EMA）
  const swipeRef = useRef<{
    id: number; sx: number; sy: number;
    engaged: boolean; dir: "open" | "close";
    lx: number; lt: number; v: number | null; peek: number;
  } | null>(null);
  // 右側立繪頁本體手勢：只在開著時往右滑收合，跟左側隔離（方向相反、互不觸發）
  const rswipeRef = useRef<{
    id: number; sx: number; sy: number;
    engaged: boolean;
    lx: number; lt: number; v: number | null; peek: number;
  } | null>(null);
  const [draggingL, setDraggingL] = useState(false);
  const [draggingR, setDraggingR] = useState(false);
  // 抽屜寬度（跟 PhoneDrawer 內同式：min(320, maxW*0.85)），手勢 clamp 用
  const drawerW = Math.min(320, Math.round(dim.w * 0.85));

  // 右側立繪頁本體手勢：開著時右滑（推出畫面外）跟手收合
  const rPanelProps = {
    onPointerDown: (e: RPointerEvent<HTMLDivElement>) => {
      if (!drawerR || dim.wide) { rswipeRef.current = null; return; }
      rswipeRef.current = {
        id: e.pointerId, sx: e.clientX, sy: e.clientY, engaged: false,
        lx: e.clientX, lt: e.timeStamp, v: null, peek: drawerW,
      };
      e.currentTarget.setPointerCapture(e.pointerId);
    },
    onPointerMove: (e: RPointerEvent<HTMLDivElement>) => {
      const s = rswipeRef.current;
      if (!s || e.pointerId !== s.id) return;
      const dx = e.clientX - s.sx;
      const dy = e.clientY - s.sy;
      { const dtm = Math.max(8, e.timeStamp - s.lt); const inst = (e.clientX - s.lx) / dtm; s.v = s.v == null ? inst : s.v * 0.6 + inst * 0.4; }
      s.lx = e.clientX; s.lt = e.timeStamp;
      const horiz = Math.abs(dx) > Math.abs(dy) * 1.2;
      if (!s.engaged) {
        if (Math.abs(dy) > 12 && !horiz) { rswipeRef.current = null; setDraggingR(false); setPeekR(null); return; }
        // 右滑（dx>0）往外推出＝收合；左滑不跟手
        if (dx > 10 && horiz) { s.engaged = true; setDraggingR(true); }
        else return;
      }
      s.peek = Math.max(0, Math.min(drawerW, drawerW - dx));
      setPeekR(s.peek);
    },
    onPointerUp: (e: RPointerEvent<HTMLDivElement>) => {
      const s = rswipeRef.current;
      rswipeRef.current = null;
      setDraggingR(false);
      if (!s || e.pointerId !== s.id) { setPeekR(null); return; }
      const v = (s.v != null && e.timeStamp - s.lt <= 120) ? s.v : s.peek / Math.max(1, e.timeStamp - s.lt);
      // 收超過半、或快速右掃 → 收合，否則彈回全開
      const willClose = s.peek < drawerW / 2 || v > 0.4;
      setDrawerR(!willClose);
      setPeekR(null);
    },
    onPointerCancel: () => { rswipeRef.current = null; setDraggingR(false); setPeekR(null); },
  };

  // 非暫停時自動收合抽屜（例如點「繼續戰鬥」恢復遊戲）
  useEffect(() => {
    if (state !== "paused") {
      setDrawerL(false);
      setDrawerR(false);
      setPeekL(null);
      setPeekR(null);
      setDraggingL(false);
      setDraggingR(false);
      swipeRef.current = null;
      rswipeRef.current = null;
    }
  }, [state]);
  const ship = SHIPS[shipId];
  const stage = stageId > 0 ? STAGES[stageId - 1] : null;

  // 版面：9:16 戰鬥區自動置中，寬螢幕顯示左右資訊欄
  useLayoutEffect(() => {
    const el = wrapRef.current!;
    const calc = () => {
      const vw = el.clientWidth;
      const vh = el.clientHeight;
      // PC 視窗模式（非全螢幕）遊戲框上下各留 10px，避免底端貼到 taskbar 被蓋；
      // 手機觸控 / 全螢幕維持全高貼邊（照 000B-2 的 letterbox 留白，但只作用 PC）
      const pc = window.matchMedia("(pointer: fine)").matches;
      const m = pc && !document.fullscreenElement ? 10 : 0;
      let h = vh - m * 2;
      let w = (h * 540) / 960;
      if (w > vw) {
        w = vw;
        h = Math.min((w * 960) / 540, vh - m * 2);
      }
      setDim({ w: Math.floor(w), h: Math.floor(h), wide: vw - w >= 460 });
    };
    calc();
    const ro = new ResizeObserver(calc);
    ro.observe(el);
    window.addEventListener("orientationchange", calc);
    window.addEventListener("resize", calc);
    window.addEventListener("fullscreenchange", calc);
    return () => {
      ro.disconnect();
      window.removeEventListener("orientationchange", calc);
      window.removeEventListener("resize", calc);
      window.removeEventListener("fullscreenchange", calc);
    };
  }, []);

  useEffect(() => {
    gameRef.current?.resize(dim.w, dim.h);
  }, [dim]);

  // 建立遊戲
  useEffect(() => {
    audio.init();
    const canvas = canvasRef.current!;
    const g = new Game({
      ship: shipId,
      stageId,
      diff,
      save: getData(),
      canvas,
      cb: {
        onCards: (c) => setCards(c),
        onState: (s) => setState(s),
        onOver: (r) => setTimeout(() => setResult(r), 500),
        onClear: (r) => setTimeout(() => setResult(r), 300),
        onBossIntro: (b, st) => {
          const k = Date.now();
          setBoss({ b, stage: st, k });
          setTimeout(() => setBoss((cur) => (cur && cur.k === k ? null : cur)), 3400);
        },
        onBanner: (title, sub, color = "#fff") => {
          const k = Date.now() + Math.random();
          setBanner({ title, sub, color, k });
          setTimeout(() => setBanner((cur) => (cur && cur.k === k ? null : cur)), 2300);
        },
        onEvolve: (i) => {
          const k = Date.now();
          setEvo({ ...i, k });
          setTimeout(() => setEvo((cur) => (cur && cur.k === k ? null : cur)), 2200);
        },
      },
    });
    gameRef.current = g;
    g.resize(dim.w, dim.h);
    g.start();
    audio.playBgm(stage ? stage.music : 1);
    const iv = window.setInterval(() => setHud({ ...g.hud }), 85);
    const vis = () => {
      if (document.hidden) g.pause();
    };
    document.addEventListener("visibilitychange", vis);
    if (!save.tutorialSeen) {
      mutate((d) => {
        d.tutorialSeen = true;
      });
      setTimeout(() => setHint(false), 6500);
    }
    return () => {
      window.clearInterval(iv);
      document.removeEventListener("visibilitychange", vis);
      g.stop();
      audio.setTheme(0);
      gameRef.current = null;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [shipId, stageId, diff]);

  // 卡片鍵盤選擇
  useEffect(() => {
    if (!cards) return;
    const h = (e: KeyboardEvent) => {
      const n = parseInt(e.key);
      if (n >= 1 && n <= cards.length) pick(cards[n - 1].id);
    };
    window.addEventListener("keydown", h);
    return () => window.removeEventListener("keydown", h);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [cards]);

  const pick = (id: string) => {
    gameRef.current?.pickCard(id);
    setCards(null);
  };
  const g = gameRef.current;
  const k = dim.w / 540;
  const playing = state === "playing" && !cards;
  const ultcut = state === "ultcut"; // cutin 期間暫-hide 中央提示，避免蓋住立繪臉

  return (
    <div ref={wrapRef} className="screen bg-black flex items-center justify-center">
      {/* 背景氛圍 */}
      <div className="absolute inset-0 opacity-60" style={{ background: `radial-gradient(circle at 50% 50%, ${ship.color}22, #000 70%)` }} />
      <div className="absolute inset-0 halftone opacity-20" />

      {dim.wide && (
        <div className="relative flex-1 min-w-0 h-full flex items-center justify-center pointer-events-none">
          <div className="relative h-[86%] aspect-[1/2] anim-up">
            <div className="absolute inset-0 bg-black/60 jag-torn translate-x-3 translate-y-3" />
            <Portrait id={shipId} variant="battle" className="absolute inset-0 jag-torn" style={{ filter: `drop-shadow(0 0 26px ${ship.color}77)` }} />
            <div className="absolute left-2 right-2 bottom-3 bg-black/85 px-3 py-2 border-l-4" style={{ borderColor: ship.color }}>
              <div className="disp text-3xl leading-none" style={{ color: ship.color2 }}>{ship.en}</div>
              <div className="text-xs text-white/70">{ship.pilot}</div>
            </div>
          </div>
        </div>
      )}
