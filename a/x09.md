// 戰機繪製分派入口
import type { ShipId } from "../types";
import { SHIPS } from "../data/ships";
import { drawCrow, drawJade, drawLance } from "./shipDrawA";
import { drawNoir, drawPrism, drawVolt } from "./shipDrawB";
import type { ShipDrawOpts } from "./shipUtil";

export type { ShipDrawOpts } from "./shipUtil";

export function makeOpts(id: ShipId, t: number, tier: number, gear: number[] = [0, 0, 0, 0, 0, 0]): ShipDrawOpts {
  const d = SHIPS[id];
  return { t, tier, thrust: 1, form: 0, morph: 0, gear, bank: 0, color: d.color, color2: d.color2 };
}

/** 於目前座標原點繪製戰機（機首朝上） */
const PORT_SHIPS_076 = new Set<string>([]);

function drawShipBase(ctx: CanvasRenderingContext2D, id: ShipId, o: ShipDrawOpts) {
  const s = PORT_SHIPS_076.has(id) ? 0.76 : 1;
  if (s !== 1) ctx.save();
  if (s !== 1) ctx.scale(s, s);
  switch (id) {
    case "crow": drawCrow(ctx, o); break;
    case "lance": drawLance(ctx, o); break;
    case "jade": drawJade(ctx, o); break;
    case "volt": drawVolt(ctx, o); break;
    case "noir": drawNoir(ctx, o); break;
    case "prism": drawPrism(ctx, o); break;
  }
  if (s !== 1) ctx.restore();
}

export function drawShip(ctx: CanvasRenderingContext2D, id: ShipId, o: ShipDrawOpts) {
  drawShipBase(ctx, id, o);
}

// 戰機繪製（上）：佔位骨架 —— 外觀血肉已清空，三個機體統一畫中性機身輪廓。
// 新增／改寫外觀時照此結構：開 drawGear(ctx,o,false)、於中心原點（機首朝上）繪製、結 drawGear(ctx,o,true)；進化用 o.tier，形態用 o.form。
import { TAU } from "../constants";
import { fillPoly, flame, lin, mirror, drawGear } from "./shipUtil";
import type { ShipDrawOpts } from "./shipUtil";

const OUT = "rgba(10,4,10,0.9)";

/** 中性佔位機身：只證明接線可用，不含任何機體特定造型 */
function placeholderShip(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  const { t, thrust, tier } = o;
  drawGear(ctx, o, false);
  // 尾焰
  flame(ctx, 0, 28, 30 * thrust, 4.5, t, "#ffffff", o.color2);
  // 主翼（左右對稱，隨進化微張）
  const wy = 16 + Math.min(tier, 3) * 2;
  mirror(ctx, () => {
    fillPoly(ctx, [[4, 0], [14 + wy, 26], [14 + wy, 30], [6, 24]], lin(ctx, 4, 0, 14 + wy, 30, [[0, o.color2], [1, o.color]]), OUT, 1);
  });
  // 機身
  fillPoly(ctx, [[0, -34], [5, -14], [6.5, 8], [4, 28], [0, 34], [-4, 28], [-6.5, 8], [-5, -14]], lin(ctx, -7, 0, 7, 0, [[0, o.color], [0.5, o.color2], [1, o.color]]), OUT, 1.2);
  // 座艙
  ctx.fillStyle = o.color2;
  ctx.beginPath();
  ctx.ellipse(0, -12, 2.6, 7, 0, 0, TAU);
  ctx.fill();
  ctx.strokeStyle = OUT;
  ctx.lineWidth = 1;
  ctx.stroke();
  drawGear(ctx, o, true);
}

export function drawCrow(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  placeholderShip(ctx, o);
}

export function drawLance(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  placeholderShip(ctx, o);
}

export function drawJade(ctx: CanvasRenderingContext2D, o: ShipDrawOpts) {
  placeholderShip(ctx, o);
}

/** 建立並註冊技能特效：update 回傳 false 代表結束 */
export function mkFx(g: Game, layer: 0 | 1 | 2, update: (dt: number) => boolean | void, draw: (ctx: CanvasRenderingContext2D) => void): Fx {
  const fx: Fx = {
    alive: true,
    layer,
    update: (dt) => {
      if (update(dt) === false) fx.alive = false;
    },
    draw,
  };
  g.addFx(fx);
  return fx;
}

/** 短暫閃電線 */
export function boltFx(g: Game, x1: number, y1: number, x2: number, y2: number, color: string, life = 0.16, w = 2.6) {
  let t = 0;
  const seed = Math.random() * 100;
  mkFx(
    g,
    1,
    (dt) => {
      t += dt;
      return t < life;
    },
    (ctx) => {
      ctx.globalCompositeOperation = "lighter";
      ctx.globalAlpha = Math.max(0, 1 - t / life);
      bolt(ctx, x1, y1, x2, y2, seed + Math.floor(t * 40), 7, 9, color, w, "#ffffff");
      ctx.globalAlpha = 1;
      ctx.globalCompositeOperation = "source-over";
    }
  );
}

export function drawBars(ctx: CanvasRenderingContext2D, a: number, W: number, H: number) {

}

/** 點到線段距離（點 (px,py) 到線段 (x0,y0)-(x1,y1)） */
export function segDist(px: number, py: number, x0: number, y0: number, x1: number, y1: number): number {
  const dx = x1 - x0, dy = y1 - y0;
  const l2 = dx * dx + dy * dy;
  let t = l2 > 0 ? ((px - x0) * dx + (py - y0) * dy) / l2 : 0;
  t = Math.max(0, Math.min(1, t));
  return Math.hypot(px - (x0 + dx * t), py - (y0 + dy * t));
}

/** 特效貼圖（來源 fx 圖集 API 的保留殼，dst 無 fx 圖集）
 *  idx===3：來源 來源機體D 奧義魔法陣貼圖（霓虹紫魔法陣）→ dst 用 canvas 程式重繪
 *  其他 idx：no-op（dst 無對應貼圖）
 *  繪製參數跟來源 drawCell 語意一致：以 (cx,cy) 中心、size 邊長、rot 旋轉、alpha 透明度、mode 混合模式。
 *  來源機體D 兩層反向旋轉：外層 size×1、內層 size×0.62，rot 相反號，重疊出旋渦感。 */
export function fxTex(ctx: CanvasRenderingContext2D, idx: number, cx: number, cy: number, size: number, rot = 0, alpha = 1, mode: GlobalCompositeOperation = "screen") {
  if (idx !== 3) return; // 來源 fx 圖集只有第 3 格（魔法陣）被 來源機體D 用到，其他格不搬
  if (alpha <= 0.01) return;
  const r = size / 2;
  ctx.save();
  ctx.globalAlpha = Math.min(1, alpha);
  ctx.globalCompositeOperation = mode;
  ctx.translate(cx, cy);
  ctx.rotate(rot);
  ctx.scale(r / 100, r / 100); // 把 size 歸一化到 100 單位，內部幾何用 0~100 座標
  // ── 外環符號帶（主紫 #c76bff）──
  ctx.strokeStyle = "rgba(199,107,255,0.6)";
  ctx.lineWidth = 1.6;
  ctx.beginPath();
  ctx.arc(0, 0, 92, 0, Math.PI * 2);
  ctx.stroke();
  ctx.strokeStyle = "rgba(255,95,168,0.45)";
  ctx.lineWidth = 1.2;
  ctx.beginPath();
  ctx.arc(0, 0, 80, 0, Math.PI * 2);
  ctx.stroke();
  // 外環符號（16 個，每 22.5° 一個，主紫 rune）
  ctx.strokeStyle = "rgba(199,107,255,0.85)";
  ctx.lineWidth = 1;
  for (let i = 0; i < 16; i++) {
    const a = (i / 16) * Math.PI * 2;
    const x = Math.cos(a) * 86, y = Math.sin(a) * 86;
    ctx.save();
    ctx.translate(x, y);
    ctx.rotate(a + Math.PI / 2);
    ctx.beginPath();
    ctx.moveTo(0, -4);
    ctx.lineTo(0, 4);
    ctx.moveTo(0, -4);
    ctx.lineTo(3, -1);
    ctx.stroke();
    ctx.restore();
  }
  // ── 六角星（兩個反向三角，粉紅 #ff5fa8）──
  ctx.strokeStyle = "rgba(255,95,168,0.8)";
  ctx.lineWidth = 1.6;
  for (const flip of [0, 1]) {
    ctx.beginPath();
    for (let i = 0; i < 3; i++) {
      const a = -Math.PI / 2 + flip * Math.PI + (i / 3) * Math.PI * 2;
      const x = Math.cos(a) * 58, y = Math.sin(a) * 58;
      if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    }
    ctx.closePath();
    ctx.stroke();
  }
  // ── 八角星（兩個反向正方形，主紫 #c76bff）──
  ctx.strokeStyle = "rgba(199,107,255,0.9)";
  ctx.lineWidth = 1.2;
  for (const rot2 of [Math.PI / 8, 0]) {
    ctx.beginPath();
    for (let i = 0; i < 4; i++) {
      const a = -Math.PI / 2 + rot2 + (i / 4) * Math.PI * 2;
      const x = Math.cos(a) * 34, y = Math.sin(a) * 34;
      if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
    }
    ctx.closePath();
    ctx.stroke();
  }
  // ── 放射線（12 條，主紫，時鐘刻度感）──
  ctx.strokeStyle = "rgba(199,107,255,0.55)";
  ctx.lineWidth = 0.8;
  for (let i = 0; i < 12; i++) {
    const a = (i / 12) * Math.PI * 2;
    ctx.beginPath();
    ctx.moveTo(Math.cos(a) * 42, Math.sin(a) * 42);
    ctx.lineTo(Math.cos(a) * 78, Math.sin(a) * 78);
    ctx.stroke();
  }
  // ── 中心光點（粉紅 #ff5fa8，符機體主色、不純白）──
  ctx.strokeStyle = "rgba(255,95,168,0.85)";
  ctx.lineWidth = 1.4;
  ctx.beginPath();
  ctx.arc(0, 0, 10, 0, Math.PI * 2);
  ctx.stroke();
  ctx.fillStyle = "rgba(255,95,168,0.6)";
  ctx.beginPath();
  ctx.arc(0, 0, 4, 0, Math.PI * 2);
  ctx.fill();
  ctx.restore();
}

  if (a <= 0.005) return;

  ctx.globalAlpha = Math.min(1, a);
  ctx.fillStyle = color;

}

export function liveEnemies(g: Game): Enemy[] {
  const out: Enemy[] = [];

}
